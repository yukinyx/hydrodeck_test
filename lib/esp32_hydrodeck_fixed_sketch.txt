class WifiProvisioningCallbacks : public NimBLECharacteristicCallbacks {
  void onWrite(NimBLECharacteristic* characteristic, NimBLEConnInfo& connInfo) override {
    String payload = characteristic->getValue().c_str();
    int separator = payload.indexOf('\n');
    if (separator <= 0) return;

    String newSsid = payload.substring(0, separator);
    String newPassword = payload.substring(separator + 1);
    newSsid.trim();
    newPassword.trim();
    if (newSsid.length() == 0 || newSsid.length() >= sizeof(ssid) || newPassword.length() >= sizeof(password)) return;

    strncpy(previousSsid, ssid, sizeof(previousSsid));
    strncpy(previousPassword, password, sizeof(previousPassword));
    newSsid.toCharArray(ssid, sizeof(ssid));
    newPassword.toCharArray(password, sizeof(password));
    provisioningPending = true;
    provisioningStartedAt = 0;
  }
};

class BleServerCallbacks : public NimBLEServerCallbacks {
  void onDisconnect(NimBLEServer* server, NimBLEConnInfo& connInfo, int reason) override {
    delay(100);
    NimBLEDevice::startAdvertising();
  }
};

void setupBleProvisioning() {
  NimBLEDevice::init(BLE_DEVICE_NAME);
  NimBLEServer* bleServer = NimBLEDevice::createServer();
  bleServer->setCallbacks(new BleServerCallbacks());
  NimBLEService* bleService = bleServer->createService(BLE_SERVICE_UUID);
  wifiCharacteristic = bleService->createCharacteristic(
    BLE_WIFI_CHARACTERISTIC_UUID,
    NIMBLE_PROPERTY::WRITE
  );
  statusCharacteristic = bleService->createCharacteristic(
    BLE_STATUS_CHARACTERISTIC_UUID,
    NIMBLE_PROPERTY::READ | NIMBLE_PROPERTY::NOTIFY
  );
  statusCharacteristic->setValue("READY");
  wifiCharacteristic->setCallbacks(new WifiProvisioningCallbacks());
  bleService->start();
  NimBLEAdvertising* advertising = NimBLEDevice::getAdvertising();
  advertising->addServiceUUID(BLE_SERVICE_UUID);
  NimBLEDevice::startAdvertising();
}

int getMedianNum(int values[], int length) {
  int bTab[length];
  for (byte i = 0; i < length; i++) {
    bTab[i] = values[i];
  }
  int i, j, bTemp;
  for (j = 0; j < length - 1; j++) {
    for (i = 0; i < length - j - 1; i++) {
      if (bTab[i] > bTab[i + 1]) {
        bTemp = bTab[i];
        bTab[i] = bTab[i + 1];
        bTab[i + 1] = bTemp;
      }
    }
  }
  if ((length & 1) > 0) {
    bTemp = bTab[(length - 1) / 2];
  } else {
    bTemp = (bTab[length / 2] + bTab[length / 2 - 1]) / 2;
  }
  return bTemp;
}

float readPhSensor(int pin, float sensorSlope) {
  int measurings = 0;
  int samples = 10;
  for (int i = 0; i < samples; i++) {
    measurings += analogRead(pin);
    delay(10);
  }
  float voltage = (v_ref / 4095.0f) * (float)measurings / (float)samples;
  float ph = 7.0 + ((2.5 - voltage) / sensorSlope);
  if (ph < 0.0) ph = 0.0;
  if (ph > 14.0) ph = 14.0;
  return ph;
}

// Actuator control functions
void setPump(int pumpNum, bool turnOn) {
  int pin = -1;
  switch (pumpNum) {
    case 1: pin = PUMP1_PIN; break;
    case 2: pin = PUMP2_PIN; break;
    case 3: pin = PUMP3_PIN; break;
    case 4: pin = PUMP4_PIN; break;
    case 5: pin = PUMP5_PIN; break;
    case 6: pin = PUMP6_PIN; break;
  }
  if (pin != -1) {
    digitalWrite(pin, turnOn ? PUMP_ON : PUMP_OFF);
  }
}

// HTTP API Control Handlers
void handleControl() {
  if (server.hasArg("pump") && server.hasArg("state")) {
    int p = server.arg("pump").toInt();
    bool state = (server.arg("state") == "1" || server.arg("state") == "on");
    if (p >= 1 && p <= 6) {
      setPump(p, state);
      server.send(200, "text/plain", "OK");
      return;
    }
  } else if (server.hasArg("light") && server.hasArg("state")) {
    bool state = (server.arg("state") == "1" || server.arg("state") == "on");
    setGrowLight(state);
    server.send(200, "text/plain", "OK");
    return;
  }
  server.send(400, "text/plain", "Bad Request");
}

void processFirebaseResult(AsyncResult& firebaseResult) {
  if (!firebaseResult.isResult()) return;
  if (firebaseResult.isError()) {
    Serial.print("Firebase error [");
    Serial.print(firebaseResult.uid());
    Serial.print("] code: ");
    Serial.print(firebaseResult.error().code());
    Serial.print(" message: ");
    Serial.println(firebaseResult.error().message());
  }
  if (firebaseResult.isEvent()) {
    Serial.print("Firebase event: ");
    Serial.println(firebaseResult.eventLog().message());
  }
  if (firebaseResult.isDebug()) {
    Serial.print("Firebase debug: ");
    Serial.println(firebaseResult.debug());
  }
}

void initializeFirebase() {
  if (firebaseInitialized || WiFi.status() != WL_CONNECTED) return;
  ssl_client.setInsecure();
  ssl_client.setConnectionTimeout(1000);
  ssl_client.setHandshakeTimeout(5);
  initializeApp(aClient, app, getAuth(user_auth), processFirebaseResult, "authTask");
  app.getApp<RealtimeDatabase>(Database);
  Database.url(DATABASE_URL);
  firebaseInitialized = true;
  Serial.println("Firebase authentication starting...");
}

void setup() {
  Serial.begin(115200);

  // Initialize Actuators (Default OFF)
  pinMode(PUMP1_PIN, OUTPUT); digitalWrite(PUMP1_PIN, PUMP_OFF);
  pinMode(PUMP2_PIN, OUTPUT); digitalWrite(PUMP2_PIN, PUMP_OFF);
  pinMode(PUMP3_PIN, OUTPUT); digitalWrite(PUMP3_PIN, PUMP_OFF);
  pinMode(PUMP4_PIN, OUTPUT); digitalWrite(PUMP4_PIN, PUMP_OFF);
  pinMode(PUMP5_PIN, OUTPUT); digitalWrite(PUMP5_PIN, PUMP_OFF);
  pinMode(PUMP6_PIN, OUTPUT); digitalWrite(PUMP6_PIN, PUMP_OFF);
  pinMode(GROW_LIGHT_PIN, OUTPUT); digitalWrite(GROW_LIGHT_PIN, LIGHT_OFF);

  // Initialize Sensors
  pinMode(WATER_SENSOR_PIN1, INPUT_PULLUP);
  pinMode(WATER_SENSOR_PIN2, INPUT_PULLUP);
  pinMode(PH_PIN1, INPUT);
  pinMode(PH_PIN2, INPUT);
  pinMode(TDS_PIN1, INPUT);
  pinMode(TDS_PIN2, INPUT);

  analogReadResolution(12);
  analogSetPinAttenuation(PH_PIN1, ADC_11db);
  analogSetPinAttenuation(PH_PIN2, ADC_11db);
  analogSetPinAttenuation(TDS_PIN1, ADC_11db);
  analogSetPinAttenuation(TDS_PIN2, ADC_11db);

  // Temperature Sensors Setup
  sensors1.begin();
  sensors1.requestTemperatures();
  sensors2.begin();
  sensors2.requestTemperatures();
  delay(750);

  float tempReading1 = sensors1.getTempCByIndex(0);
  if (tempReading1 != DEVICE_DISCONNECTED_C && tempReading1 > -50.0) {
    currentTemperature1 = tempReading1;
  }
  float tempReading2 = sensors2.getTempCByIndex(0);
  if (tempReading2 != DEVICE_DISCONNECTED_C && tempReading2 > -50.0) {
    currentTemperature2 = tempReading2;
  }
  sensors1.setWaitForConversion(false);
  sensors2.setWaitForConversion(false);

  // Wi-Fi and BLE Setup
  wifiPreferences.begin("wifi", false);
  String savedSsid = wifiPreferences.getString("ssid", ssid);
  String savedPassword = wifiPreferences.getString("password", password);
  savedSsid.toCharArray(ssid, sizeof(ssid));
  savedPassword.toCharArray(password, sizeof(password));
  setupBleProvisioning();

  WiFi.begin(ssid, password);
  Serial.print("Connecting to Wi-Fi");
  unsigned long wifiStart = millis();
  while (WiFi.status() != WL_CONNECTED && millis() - wifiStart < 30000) {
    delay(500);
    Serial.print('.');
  }
  if (WiFi.status() == WL_CONNECTED) {
    Serial.print("\nESP32 IP: ");
    Serial.println(WiFi.localIP());
    if (MDNS.begin("hydrodeck")) Serial.println("mDNS: hydrodeck.local");
    initializeFirebase();
  } else {
    Serial.println("\nWi-Fi connection failed. BLE provisioning remains available.");
  }

  server.on("/", HTTP_GET, handleRootIdent);
  server.on("/status", HTTP_GET, handleCombinedStatus);
  server.on("/control", HTTP_GET, handleControl); // Control via URL e.g. /control?pump=1&state=1
  server.begin();
}

void loop() {
  app.loop();
  server.handleClient();
  unsigned long now = millis();

  // Wi-Fi Management
  if (WiFi.status() != WL_CONNECTED && !provisioningPending && now - lastWifiRetry >= wifiRetryInterval) {
    lastWifiRetry = now;
    if (firebaseInitialized) {
      firebaseInitialized = false;
      Serial.println("Wi-Fi lost; Firebase will re-authenticate after reconnect.");
    }
    Serial.println("Retrying Wi-Fi connection...");
    WiFi.disconnect(false, false);
    WiFi.begin(ssid, password);
  }
  if (WiFi.status() == WL_CONNECTED) {
    initializeFirebase();
  }

  // BLE Provisioning Handler
  if (provisioningPending) {
    if (provisioningStartedAt == 0) {
      WiFi.disconnect(false, false);
      WiFi.begin(ssid, password);
      provisioningStartedAt = now;
    } else if (WiFi.status() == WL_CONNECTED) {
      wifiPreferences.putString("ssid", ssid);
      wifiPreferences.putString("password", password);
      statusCharacteristic->setValue("WIFI_CONNECTED");
      statusCharacteristic->notify();
      provisioningPending = false;
      restartAfterProvisioning = true;
      restartAt = now + 1500;
      Serial.println("Wi-Fi provisioning successful.");
    } else if (now - provisioningStartedAt >= provisioningTimeout) {
      strncpy(ssid, previousSsid, sizeof(ssid));
      strncpy(password, previousPassword, sizeof(password));
      WiFi.disconnect(false, false);
      WiFi.begin(ssid, password);
      statusCharacteristic->setValue("WIFI_FAILED");
      statusCharacteristic->notify();
      provisioningPending = false;
      Serial.println("Wi-Fi provisioning failed.");
    }
  }

  if (restartAfterProvisioning && now >= restartAt) {
    ESP.restart();
  }

  // Sampling TDS Sensor 1 & Sensor 2
  static unsigned long sampleTime = 0;
  if (now - sampleTime > 40U) {
    sampleTime = now;
    analogBuffer1[analogBufferIndex1] = analogRead(TDS_PIN1);
    analogBufferIndex1 = (analogBufferIndex1 + 1) % SCOUNT;

    analogBuffer2[analogBufferIndex2] = analogRead(TDS_PIN2);
    analogBufferIndex2 = (analogBufferIndex2 + 1) % SCOUNT;
  }

  // Calculate TDS Values
  static unsigned long tdsTime = 0;
  if (now - tdsTime > 800U) {
    tdsTime = now;

    // Buffer 1
    for (int i = 0; i < SCOUNT; i++) analogBufferTemp1[i] = analogBuffer1[i];
    float avgV1 = getMedianNum(analogBufferTemp1, SCOUNT) * (float)VREF / 4096.0;
    avgV1 = constrain(avgV1, 0.0f, (float)VREF);
    float compCoeff1 = 1.0 + 0.02 * (currentTemperature1 - 25.0);
    float compV1 = avgV1 / compCoeff1;
    float tdsVal1 = (133.42 * compV1 * compV1 * compV1 - 255.86 * compV1 * compV1 + 857.39 * compV1) * 0.5;
    currentTdsValue1 = max(0.0f, tdsVal1) * tdsCalibrationFactor;

    // Buffer 2
    for (int i = 0; i < SCOUNT; i++) analogBufferTemp2[i] = analogBuffer2[i];
    float avgV2 = getMedianNum(analogBufferTemp2, SCOUNT) * (float)VREF / 4096.0;
    avgV2 = constrain(avgV2, 0.0f, (float)VREF);
    float compCoeff2 = 1.0 + 0.02 * (currentTemperature2 - 25.0);
    float compV2 = avgV2 / compCoeff2;
    float tdsVal2 = (133.42 * compV2 * compV2 * compV2 - 255.86 * compV2 * compV2 + 857.39 * compV2) * 0.5;
    currentTdsValue2 = max(0.0f, tdsVal2) * tdsCalibrationFactor;
  }

  // Reading pH Sensors
  if (now - lastPhCheck >= phInterval) {
    lastPhCheck = now;
    currentPhValue1 = readPhSensor(PH_PIN1, slope1);
    currentPhValue2 = readPhSensor(PH_PIN2, slope2);
  }

  // Reading Temperature Sensors
  if (now - lastTempCheck >= tempInterval) {
    lastTempCheck = now;
    
    float reading1 = sensors1.getTempCByIndex(0);
    if (reading1 != DEVICE_DISCONNECTED_C && reading1 > -50.0) currentTemperature1 = reading1;
    sensors1.requestTemperatures();

    float reading2 = sensors2.getTempCByIndex(0);
    if (reading2 != DEVICE_DISCONNECTED_C && reading2 > -50.0) currentTemperature2 = reading2;
    sensors2.requestTemperatures();
  }

  // Syncing with Firebase Realtime Database
  if (firebaseInitialized && app.ready() && now - lastFirebasePush >= firebaseInterval) {
    lastFirebasePush = now;

    // Set 1 Data
    Database.set<float>(aClient, "/hydrodeck/set1/temp", currentTemperature1);
    Database.set<float>(aClient, "/hydrodeck/set1/ph", currentPhValue1);
    Database.set<float>(aClient, "/hydrodeck/set1/tds", currentTdsValue1);
    Database.set<String>(aClient, "/hydrodeck/set1/water", digitalRead(WATER_SENSOR_PIN1) == HIGH ? "full" : "low");

    // Set 2 Data
    Database.set<float>(aClient, "/hydrodeck/set2/temp", currentTemperature2);
    Database.set<float>(aClient, "/hydrodeck/set2/ph", currentPhValue2);
    Database.set<float>(aClient, "/hydrodeck/set2/tds", currentTdsValue2);
    Database.set<String>(aClient, "/hydrodeck/set2/water", digitalRead(WATER_SENSOR_PIN2) == HIGH ? "full" : "low");

    // Actuators Data Status Sync
    Database.set<bool>(aClient, "/hydrodeck/actuators/pump1", digitalRead(PUMP1_PIN) == PUMP_ON);
    Database.set<bool>(aClient, "/hydrodeck/actuators/pump2", digitalRead(PUMP2_PIN) == PUMP_ON);
    Database.set<bool>(aClient, "/hydrodeck/actuators/pump3", digitalRead(PUMP3_PIN) == PUMP_ON);
    Database.set<bool>(aClient, "/hydrodeck/actuators/pump4", digitalRead(PUMP4_PIN) == PUMP_ON);
    Database.set<bool>(aClient, "/hydrodeck/actuators/pump5", digitalRead(PUMP5_PIN) == PUMP_ON);
    Database.set<bool>(aClient, "/hydrodeck/actuators/pump6", digitalRead(PUMP6_PIN) == PUMP_ON);
    Database.set<bool>(aClient, "/hydrodeck/actuators/grow_light", digitalRead(GROW_LIGHT_PIN) == LIGHT_ON);

    if (aClient.lastError().code() != 0) {
      Serial.print("Firebase update failed, code: ");
      Serial.println(aClient.lastError().code());
    } else {
      Serial.println("Firebase update sent for dual sensor sets and actuators.");
    }
  }
}

Arduino IDE setup:
- Install Mobizt's FirebaseClient library.
- Install/select the Espressif ESP32 board package. Preferences is included with it.
- Install the NimBLE-Arduino library by h2int. The sketch uses NimBLE to reduce RAM usage.
- Install OneWire and DallasTemperature libraries.
- In Arduino IDE choose Tools > Partition Scheme > Huge APP (3MB No OTA/1MB SPIFFS).
- The default 1.31 MB app partition cannot hold Firebase HTTPS/authentication and BLE together.
- The app sends Wi-Fi credentials over BLE as SSID, a newline, then password.
- The ESP32 sends WIFI_CONNECTED or WIFI_FAILED over the status characteristic.
- The ESP32 advertises as ESP32-HYDRODECK. Keep it powered while provisioning.
- Remove a conflicting library that also installs FirebaseClient.h.
- Select an ESP32 board.
- Ensure Firebase Realtime Database rules allow the Firebase account above.
- The Flutter app's factory PIN is 6967; that PIN is app access control only.
*/
