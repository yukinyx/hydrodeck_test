int getMedianNum(int values[], int length) {
  int bTab[length];
  for (byte i = 0; i < length; i++) {
    bTab[i] = values[i];
  }
  int i, j, bTemp;
  for (j = 0; j < length - 1; j++) {
    for (i = 0; i < length - j - 1; i++) {
      if (bTab[i] > bTab[i + 1]) {
        bTemp = bTab[i];
        bTab[i] = bTab[i + 1];
        bTab[i + 1] = bTemp;
      }
    }
  }
  if ((length & 1) > 0) {
    bTemp = bTab[(length - 1) / 2];
  } else {
    bTemp = (bTab[length / 2] + bTab[length / 2 - 1]) / 2;
  }
  return bTemp;
}

float readPhSensor(int pin, float sensorSlope) {
  int measurings = 0;
  int samples = 10;
  for (int i = 0; i < samples; i++) {
    measurings += analogRead(pin);
  }
  float voltage = (v_ref / 4095.0f) * (float)measurings / (float)samples;
  float ph = 7.0 + ((2.5 - voltage) / sensorSlope);
  if (ph < 0.0) ph = 0.0;
  if (ph > 14.0) ph = 14.0;
  return ph;
}

// Actuator control functions
void setPump(int pumpNum, bool turnOn) {
  int pin = -1;
  switch (pumpNum) {
    case 1: pin = waterPump1Pin; break;
    case 2: pin = phUp1Pin; break;
    case 3: pin = phDown1Pin; break;
    case 4: pin = nutrientPump1Pin; break;
    case 5: pin = waterPump2Pin; break;
    case 6: pin = phUp2Pin; break;
    case 7: pin = phDown2Pin; break;
    case 8: pin = nutrientPump2Pin; break;
  }
  if (pin != -1) {
    digitalWrite(pin, turnOn ? PUMP_ON : PUMP_OFF);
  }
}

// Parse PH_ADJUST:<bed>:<direction> command format
void parsePhAdjustCommand(const String& command) {
  // Expected format: "PH_ADJUST:<bed>:<direction>"
  // Example: "PH_ADJUST:1:UP" or "PH_ADJUST:2:DOWN"
  if (!command.startsWith("PH_ADJUST:")) return;

  String payload = command.substring(10); // Remove "PH_ADJUST:" prefix
  int firstColon = payload.indexOf(':');
  if (firstColon == -1) return;

  String bedStr = payload.substring(0, firstColon);
  String direction = payload.substring(firstColon + 1);

  int bed = bedStr.toInt();
  if (bed < 1 || bed > 2) return;

  String actuatorName;
  if (direction == "UP") {
    actuatorName = (bed == 1) ? "phUp1" : "phUp2";
  } else if (direction == "DOWN") {
    actuatorName = (bed == 1) ? "phDown1" : "phDown2";
  } else {
    return; // Invalid direction
  }

  // Check if pH is actually in range that needs adjustment
  bool shouldActivate = false;
  if (bed == 1) {
    if (direction == "UP" && currentPhValue1 < minPhThreshold) shouldActivate = true;
    if (direction == "DOWN" && currentPhValue1 > maxPhThreshold) shouldActivate = true;
  } else { // bed == 2
    if (direction == "UP" && currentPhValue2 < minPhThreshold) shouldActivate = true;
    if (direction == "DOWN" && currentPhValue2 > maxPhThreshold) shouldActivate = true;
  }

  if (!shouldActivate) return;

  // Set durations based on bed
  unsigned long durationMs = bed == 1 ? BED1_PUMP_DURATION_MS : BED2_PUMP_DURATION_MS;

  // Find pump number
  int pumpNum = pumpNumberForActuator(actuatorName);
  if (pumpNum == 0) return;

  // Check if pump is already active
  if (pumpRuns[pumpNum - 1].active) return;

  // Check for conflicting pH pumps (can't have both up and down active at same time)
  if ((pumpNum == 2 || pumpNum == 3) && (pumpRuns[1].active || pumpRuns[2].active)) return;
  if ((pumpNum == 6 || pumpNum == 7) && (pumpRuns[5].active || pumpRuns[6].active)) return;

  // Start the pump run
  PumpRunState& run = pumpRuns[pumpNum - 1];
  run.active = true;
  run.stopWhenFull = false;
  run.hasDeadline = true;
  run.stopAt = millis() + durationMs;
  setPump(pumpNum, true);
  publishActuatorState(actuatorName, true);
}

String consumedCommandStatus(const String& requestId) {
  const String entries = commandPreferences.getString("recentIds", "");
  int start = 0;
  while (start < entries.length()) {
    const int separator = entries.indexOf(',', start);
    const int end = separator < 0 ? entries.length() : separator;
    const String entry = entries.substring(start, end);
    const int statusSeparator = entry.indexOf('=');
    const String storedId = statusSeparator < 0 ? entry : entry.substring(0, statusSeparator);
    if (storedId == requestId) {
      return statusSeparator < 0 ? "completed" : entry.substring(statusSeparator + 1);
    }
    if (separator < 0) break;
    start = separator + 1;
  }
  return "";
}

bool wasCommandConsumed(const String& requestId) {
  return consumedCommandStatus(requestId).length() > 0;
}

bool isValidCommandRequestId(const String& requestId) {
  // Current app IDs are Firebase push IDs or the cmd_* fallback format.
  return requestId.startsWith("cmd_") ||
      (requestId.startsWith("-") && requestId.length() >= 16);
}

void rememberConsumedCommand(const String& requestId, const String& status = "accepted") {
  if (requestId.length() == 0) return;
  String entries = commandPreferences.getString("recentIds", "");
  String retained;
  int start = 0;
  while (start < entries.length()) {
    const int separator = entries.indexOf(',', start);
    const int end = separator < 0 ? entries.length() : separator;
    const String entry = entries.substring(start, end);
    const int statusSeparator = entry.indexOf('=');
    const String storedId = statusSeparator < 0 ? entry : entry.substring(0, statusSeparator);
    if (storedId != requestId && entry.length() > 0) {
      if (retained.length() > 0) retained += ",";
      retained += entry;
    }
    if (separator < 0) break;
    start = separator + 1;
  }
  if (retained.length() > 0) retained += ",";
  retained += requestId + "=" + status;
  while (retained.length() > 700) {
    const int separator = retained.indexOf(',');
    if (separator < 0) break;
    retained.remove(0, separator + 1);
  }
  commandPreferences.putString("recentIds", retained);
}

void publishActuatorState(const String& actuator, bool state) {
  if (!firebaseInitialized || !app.ready()) return;
  Database.set<bool>(aClient, "/hydrodeck/actuators/" + actuator, state);
}

void publishCommandAck(const String& requestId, const String& status) {
  rememberConsumedCommand(requestId, status);
  if (requestId.length() == 0 || !firebaseInitialized || !app.ready()) return;
  Database.set<String>(aClient, "/hydrodeck/commandAcks/" + requestId, status);
}

String pumpStartRejectionReason(int pumpNum) {
  if (pumpNum < 1 || pumpNum > 8) return "invalid_pump";
  if (pumpRuns[pumpNum - 1].active) return "pump_already_active";
  if ((pumpNum == 1 && pumpRuns[4].active) || (pumpNum == 5 && pumpRuns[0].active)) {
    return "other_bed_water_pump_active";
  }
  if ((pumpNum == 2 && pumpRuns[2].active) || (pumpNum == 3 && pumpRuns[1].active) ||
      (pumpNum == 6 && pumpRuns[6].active) || (pumpNum == 7 && pumpRuns[5].active)) {
    return "opposite_ph_pump_active";
  }
  const bool chemicalPump = pumpNum == 2 || pumpNum == 3 || pumpNum == 4 ||
      pumpNum == 6 || pumpNum == 7 || pumpNum == 8;
  if (chemicalPump) {
    for (int index = 0; index < 8; index++) {
      const int otherPump = index + 1;
      const bool otherIsChemical = otherPump == 2 || otherPump == 3 || otherPump == 4 ||
          otherPump == 6 || otherPump == 7 || otherPump == 8;
      if (otherPump != pumpNum && otherIsChemical && pumpRuns[index].active) {
        return "chemical_mutex_busy_by_" + String(remoteActuatorNames[index]);
      }
    }
  }
  return "";
}

bool startPumpRun(
    int pumpNum,
    unsigned long durationMs,
    bool stopWhenFull,
    const String& requestId = "",
    String* rejectionReason = nullptr) {
  const String reason = pumpStartRejectionReason(pumpNum);
  if (rejectionReason != nullptr) *rejectionReason = reason;
  if (reason.length() > 0) return false;
  PumpRunState& run = pumpRuns[pumpNum - 1];
  run.active = true;
  run.stopWhenFull = stopWhenFull;
  run.hasDeadline = durationMs > 0;
  run.stopAt = durationMs == 0 ? 0 : millis() + durationMs;
  activeCommandIds[pumpNum - 1] = requestId;
  setPump(pumpNum, true);
  publishActuatorState(remoteActuatorNames[pumpNum - 1], true);
  publishCommandAck(requestId, "running");
  return true;
}

void serviceAutomaticPhControl(int bed, float phValue, unsigned long now) {
  const int upPump = bed == 1 ? 2 : 6;
  const int downPump = bed == 1 ? 3 : 7;
  const unsigned long pulseDuration =
      bed == 1 ? BED1_PUMP_DURATION_MS : BED2_PUMP_DURATION_MS;
  unsigned long& lowSince = bed == 1 ? bed1PhLowSince : bed2PhLowSince;
  unsigned long& highSince = bed == 1 ? bed1PhHighSince : bed2PhHighSince;
  unsigned long& lastAttemptLogAt =
      bed == 1 ? bed1PhLastAttemptLogAt : bed2PhLastAttemptLogAt;

  if (phValue < minPhThreshold) {
    highSince = 0;
    if (lowSince == 0) lowSince = now;
    const unsigned long secondsBelowMin = (now - lowSince) / 1000UL;
    if (now - lowSince >= phDelayThreshold) {
      String rejectionReason;
      const bool started = startPumpRun(
        upPump,
        pulseDuration,
        false,
        "",
        &rejectionReason
      );
      if (started || lastAttemptLogAt == 0 ||
          now - lastAttemptLogAt >= phAutoRetryLogInterval) {
        Serial.printf(
          "Auto pH-up bed=%d pH=%.2f min=%.2f max=%.2f secondsBelowMin=%lu result=%s reason=%s\n",
          bed,
          phValue,
          minPhThreshold,
          maxPhThreshold,
          secondsBelowMin,
          started ? "accepted" : "rejected",
          started ? "none" : rejectionReason.c_str()
        );
        lastAttemptLogAt = now;
      }
      if (started) lowSince = now;
    }
  } else if (phValue > maxPhThreshold) {
    lowSince = 0;
    if (highSince == 0) highSince = now;
    const unsigned long secondsAboveMax = (now - highSince) / 1000UL;
    if (now - highSince >= phDelayThreshold) {
      String rejectionReason;
      const bool started = startPumpRun(
        downPump,
        pulseDuration,
        false,
        "",
        &rejectionReason
      );
      if (started || lastAttemptLogAt == 0 ||
          now - lastAttemptLogAt >= phAutoRetryLogInterval) {
        Serial.printf(
          "Auto pH-down bed=%d pH=%.2f min=%.2f max=%.2f secondsAboveMax=%lu result=%s reason=%s\n",
          bed,
          phValue,
          minPhThreshold,
          maxPhThreshold,
          secondsAboveMax,
          started ? "accepted" : "rejected",
          started ? "none" : rejectionReason.c_str()
        );
        lastAttemptLogAt = now;
      }
      if (started) highSince = now;
    }
  } else {
    if (phValue >= minPhThreshold + phControlHysteresis) lowSince = 0;
    if (phValue <= maxPhThreshold - phControlHysteresis) highSince = 0;
  }
}

void stopPumpRun(int pumpNum) {
  if (pumpNum < 1 || pumpNum > 8) return;
  PumpRunState& run = pumpRuns[pumpNum - 1];
  if (!run.active) return;
  setPump(pumpNum, false);
  const String requestId = activeCommandIds[pumpNum - 1];
  activeCommandIds[pumpNum - 1] = "";
  run.active = false;
  run.stopWhenFull = false;
  run.hasDeadline = false;
  run.stopAt = 0;
  publishActuatorState(remoteActuatorNames[pumpNum - 1], false);
  publishCommandAck(requestId, "completed");
}

bool executePumpRequest(const String& actuator, bool enforceManualRules, const String& requestId = "") {
  const int pumpNum = pumpNumberForActuator(actuator);
  if (pumpNum == 0) return false;

  unsigned long durationMs = 0;
  bool stopWhenFull = false;
  int* manualShortRunCounter = nullptr;
  if (actuator == "phUp1" || actuator == "phDown1") {
    if ((actuator == "phUp1" && currentPhValue1 > maxPhThreshold) ||
      (actuator == "phDown1" && currentPhValue1 < minPhThreshold)) return false;
    durationMs = BED1_PUMP_DURATION_MS;
  } else if (actuator == "phUp2" || actuator == "phDown2") {
    if ((actuator == "phUp2" && currentPhValue2 > maxPhThreshold) ||
      (actuator == "phDown2" && currentPhValue2 < minPhThreshold)) return false;
    durationMs = BED2_PUMP_DURATION_MS;
  } else if (actuator == "nutrient1") {
    durationMs = BED1_PUMP_DURATION_MS;
  } else if (actuator == "nutrient2") {
    durationMs = BED2_PUMP_DURATION_MS;
  } else if (actuator == "waterPump1" || actuator == "waterPump2") {
    if (pumpRuns[pumpNum - 1].active) return false;
    const bool isFull = waterSensorFullForPump(pumpNum);
    if (isFull) {
      durationMs = pumpNum <= 4 ? BED1_PUMP_DURATION_MS : BED2_PUMP_DURATION_MS;
      if (enforceManualRules) {
        int& count = actuator == "waterPump1" ? manualWaterShortRuns1 : manualWaterShortRuns2;
        if (count >= 2) return false;
        manualShortRunCounter = &count;
      }
    } else {
      stopWhenFull = true;
    }
  }
  const bool started = startPumpRun(pumpNum, durationMs, stopWhenFull, requestId);
  if (started && manualShortRunCounter != nullptr) (*manualShortRunCounter)++;
  return started;
}

void servicePumpRuns() {
  const unsigned long now = millis();
  for (int pumpNum = 1; pumpNum <= 8; pumpNum++) {
    PumpRunState& run = pumpRuns[pumpNum - 1];
    if (!run.active) continue;
    if (run.stopWhenFull && waterSensorFullForPump(pumpNum)) {
      stopPumpRun(pumpNum);
    } else if (run.hasDeadline && (long)(now - run.stopAt) >= 0) {
      stopPumpRun(pumpNum);
    }
  }
}

void processCommandCleanupResult(AsyncResult& result) {
  if (pendingCommandCleanupWrites > 0) pendingCommandCleanupWrites--;
  if (result.isError()) {
    Serial.print("Command queue cleanup failed, code: ");
    Serial.println(result.error().code());
  }
  result.clear();
}

void removeQueuedCommandChild(const String& path) {
  pendingCommandCleanupWrites++;
  Database.remove(
    aClient,
    path,
    processCommandCleanupResult,
    "commandCleanup"
  );
}

void finishRemoteCommand(const String& previousStatus) {
  const String requestId = pendingRemoteCommandId;
  const String actuator = pendingRemoteActuator;
  const String commandValue = pendingRemoteCommandValue;

  if (previousStatus.length() > 0) {
    bool commandIsRunning = false;
    for (int index = 0; index < 8; index++) {
      if (activeCommandIds[index] == requestId) commandIsRunning = true;
    }
    if (commandIsRunning) {
      publishCommandAck(requestId, "running");
    } else {
      publishCommandAck(requestId, "completed");
    }
    Serial.printf(
      "Command requestId=%s actuator=%s result=%s duplicate=1 leftover_id_removed=1\n",
      requestId.c_str(),
      actuator.c_str(),
      commandIsRunning ? "running" : "completed"
    );
  } else {
    publishCommandAck(requestId, "accepted");
    if (actuator == "growLight") {
      if (commandValue == "0" || commandValue == "1") {
        const bool requestedState = commandValue == "1";
        setGrowLight(requestedState);
        publishActuatorState(actuator, requestedState);
        publishCommandAck(requestId, "completed");
        Serial.printf(
          "Command requestId=%s actuator=%s result=completed\n",
          requestId.c_str(),
          actuator.c_str()
        );
      } else {
        publishCommandAck(requestId, "rejected");
        Serial.printf(
          "Command requestId=%s actuator=%s result=rejected reason=invalid_value\n",
          requestId.c_str(),
          actuator.c_str()
        );
      }
    } else {
      const int pumpNum = pumpNumberForActuator(actuator);
      String rejectionReason;
      const bool started = commandValue == "1" && executePumpRequest(actuator, true, requestId);
      if (!started) {
        rejectionReason = commandValue != "1"
            ? "invalid_value"
            : pumpStartRejectionReason(pumpNum);
        if (rejectionReason.length() == 0) rejectionReason = "threshold_or_manual_safety";
        publishCommandAck(requestId, "rejected");
      }
      Serial.printf(
        "Command requestId=%s actuator=%s result=%s reason=%s\n",
        requestId.c_str(),
        actuator.c_str(),
        started ? "accepted" : "rejected",
        started ? "none" : rejectionReason.c_str()
      );
    }
  }

  removeQueuedCommandChild(pendingRemoteCommandPath);
  pendingRemoteActuator = "";
  pendingRemoteCommandId = "";
  pendingRemoteCommandValue = "";
  pendingRemoteCommandPath = "";
  remoteCommandPending = false;
}

bool processQueuedCommand(
    const String& actuator,
    const String& requestId,
    const String& commandValue,
    const String& commandPath) {
  if (!isValidCommandRequestId(requestId)) {
    removeQueuedCommandChild(commandPath);
    Serial.printf(
      "Command requestId=%s actuator=%s result=removed reason=invalid_id\n",
      requestId.c_str(),
      actuator.c_str()
    );
    return true;
  }

  pendingRemoteActuator = actuator;
  pendingRemoteCommandId = requestId;
  pendingRemoteCommandValue = commandValue;
  pendingRemoteCommandPath = commandPath;
  const String previousStatus = consumedCommandStatus(requestId);
  finishRemoteCommand(previousStatus);
  return previousStatus.length() > 0;
}

void processRemoteActuatorResult(AsyncResult& result) {
  remoteCommandPending = false;
  if (result.isError() || !result.available()) {
    return;
  }

  const String payload = result.to<RealtimeDatabaseResult>().to<String>();
  DynamicJsonDocument commandDocument(2048);
  const DeserializationError parseError = deserializeJson(commandDocument, payload);
  if (!parseError && commandDocument.is<JsonObject>()) {
    JsonObject allActuatorQueues = commandDocument.as<JsonObject>();
    for (JsonPair actuatorEntry : allActuatorQueues) {
      const String actuator = actuatorEntry.key().c_str();
      bool knownActuator = false;
      for (int index = 0; index < 9; index++) {
        if (actuator == remoteActuatorNames[index]) knownActuator = true;
      }
      if (!knownActuator) continue;

      const String queuePath = "/hydrodeck/commands/" + actuator;
      JsonVariant queueValue = actuatorEntry.value();
      if (queueValue.is<JsonObject>()) {
        JsonObject commandMap = queueValue.as<JsonObject>();
        for (JsonPair command : commandMap) {
          const String requestId = command.key().c_str();
          String commandValue;
          if (command.value().is<const char*>()) {
            commandValue = command.value().as<const char*>();
          }
          const String commandPath = queuePath + "/" + requestId;
          const bool continueDraining = processQueuedCommand(
            actuator,
            requestId,
            commandValue,
            commandPath
          );
          if (!continueDraining) break;
        }
      } else if (queueValue.is<const char*>()) {
        const String legacyValue = queueValue.as<const char*>();
        if (legacyValue.length() == 0 || legacyValue == "null") continue;
        String requestId;
        String commandValue;
        if (actuator == "growLight") {
          const int separator = legacyValue.lastIndexOf('|');
          if (separator > 0) {
            requestId = legacyValue.substring(0, separator);
            commandValue = legacyValue.substring(separator + 1);
          }
        } else {
          requestId = legacyValue;
          commandValue = "1";
        }
        processQueuedCommand(actuator, requestId, commandValue, queuePath);
      }
    }
  }
  result.clear();
}

void requestRemoteActuatorCommand() {

  remoteCommandPending = true;
  Database.get(aClient, "/hydrodeck/commands", processRemoteActuatorResult, false, "remoteCommands");
}

void processPhLimitResult(AsyncResult& result) {
  phLimitRequestPending = false;
  if (!result.isError() && result.available()) {
    const String payload = result.to<RealtimeDatabaseResult>().to<String>();
    DynamicJsonDocument settingsDocument(512);
    const DeserializationError parseError = deserializeJson(settingsDocument, payload);
    if (!parseError && settingsDocument.is<JsonObject>()) {
      JsonObject settings = settingsDocument.as<JsonObject>();
      const JsonVariant minimumValue = settings["minPh"];
      const JsonVariant maximumValue = settings["maxPh"];
      const bool minimumIsNumber = minimumValue.is<float>() || minimumValue.is<int>();
      const bool maximumIsNumber = maximumValue.is<float>() || maximumValue.is<int>();
      if (minimumIsNumber && maximumIsNumber) {
        const float minimum = minimumValue.as<float>();
        const float maximum = maximumValue.as<float>();
        if (minimum > 0.0f && minimum < maximum && maximum <= 14.0f &&
            (minimum != minPhThreshold || maximum != maxPhThreshold)) {
          minPhThreshold = minimum;
          maxPhThreshold = maximum;
          Serial.printf(
            "pH thresholds updated min=%.2f max=%.2f; bed1 pH=%.2f bed2 pH=%.2f\n",
            minPhThreshold,
            maxPhThreshold,
            currentPhValue1,
            currentPhValue2
          );
        }
      }
    }
    result.clear();
  }
}

String currentManilaIntervalKey(const struct tm& localTime) {
  char key[16];
  const int slotMinute = localTime.tm_min < 30 ? 0 : 30;
  snprintf(
    key,
    sizeof(key),
    "%04d%02d%02d_%02d%02d",
    localTime.tm_year + 1900,
    localTime.tm_mon + 1,
    localTime.tm_mday,
    localTime.tm_hour,
    slotMinute
  );
  return String(key);
}

  snapshotCheckPending = false;
  if (result.isError() || !result.available()) return;
  const String currentValue = result.to<RealtimeDatabaseResult>().to<String>();
  result.clear();
  if (currentValue.length() > 0 && currentValue != "null") {
    lastIntervalSnapshotIdentity = pendingIntervalSnapshotIdentity;
    commandPreferences.putString("lastSnapshotIdentity", lastIntervalSnapshotIdentity);
    return;
  }

  const unsigned long nowMs = millis();
    if (!temperatureSensorValid1 || !temperatureSensorValid2 ||
      !phSensorValid1 || !phSensorValid2 || !tdsSensorValid1 || !tdsSensorValid2 ||
      nowMs - lastTemperatureValidAt1 > 5000UL ||
      nowMs - lastTemperatureValidAt2 > 5000UL ||
      nowMs - lastPhValidAt1 > 5000UL || nowMs - lastPhValidAt2 > 5000UL ||
      nowMs - lastTdsValidAt1 > 5000UL || nowMs - lastTdsValidAt2 > 5000UL) return;

  const time_t capturedAt = time(nullptr);
  String snapshotJson = "{";
  snapshotJson += "\"intervalKey\":\"" + pendingIntervalSnapshotKey + "\",";
  snapshotJson += "\"batchNumber\":" + String(activeBatchNumber) + ",";
  snapshotJson += "\"capturedAt\":" + String((long)capturedAt) + ",";
  snapshotJson += "\"source\":\"esp32\",";

  snapshotJson += "\"tds\":" + String(currentTdsValue1, 0) + ",";
  snapshotJson += "\"water\":\"" + String(digitalRead(WATER_SENSOR_PIN1) == HIGH ? "full" : "low") + "\"},";
  snapshotJson += "\"bed2\":{\"temp\":" + String(currentTemperature2, 1) + ",";
  snapshotJson += "\"ph\":" + String(currentPhValue2, 2) + ",";
  snapshotJson += "\"tds\":" + String(currentTdsValue2, 0) + ",";
  snapshotJson += "\"water\":\"" + String(digitalRead(WATER_SENSOR_PIN2) == HIGH ? "full" : "low") + "\"}}";

  snapshotWritePending = true;
  Database.set<object_t>(
    aClient,
    pendingIntervalSnapshotPath,
    object_t(snapshotJson.c_str()),
    processIntervalSnapshotWrite,
    "intervalSnapshotWrite"
  );
}

void requestIntervalSnapshot() {
  if (!firebaseInitialized || !app.ready() || activeBatchNumber <= 0 ||
      snapshotCheckPending || snapshotWritePending) return;

  struct tm localTime;
  if (!getLocalTime(&localTime, 0) || localTime.tm_year < (2024 - 1900)) return;
  if (localTime.tm_min != 0 && localTime.tm_min != 30) return;
  const String intervalKey = currentManilaIntervalKey(localTime);
  const String intervalIdentity = intervalKey + "|" + String(activeBatchNumber);
  if (intervalIdentity == lastIntervalSnapshotIdentity) return;

  pendingIntervalSnapshotKey = intervalKey;
  pendingIntervalSnapshotIdentity = intervalIdentity;
  pendingIntervalSnapshotPath = "/hydrodeck/intervalSnapshots/" + intervalKey +
      "/sessions/" + String(activeBatchNumber);
  snapshotCheckPending = true;
  Database.get(
    aClient,
    pendingIntervalSnapshotPath,
    processIntervalSnapshotCheck,
    false,
    "intervalSnapshotCheck"
  );
}

  newSsid.trim();

void maintainProvisioningHotspot() {
  if (WiFi.getMode() == WIFI_AP_STA && WiFi.softAPIP() == apIp) return;

  const unsigned long now = millis();
  if (lastHotspotAttempt != 0 && now - lastHotspotAttempt < hotspotRetryInterval) return;
  lastHotspotAttempt = now;

  if (WiFi.getMode() != WIFI_AP_STA) {
    if (!WiFi.mode(WIFI_AP_STA)) {
      Serial.println("Failed to enable AP+STA mode; hotspot startup will retry.");
      return;
    }
    delay(100);
  }

  WiFi.setSleep(false);
  WiFi.softAPdisconnect(false);
  if (!WiFi.softAPConfig(apIp, apGateway, apSubnet)) {
    Serial.println("Failed to configure the ESP32 hotspot IP; startup will retry.");
    return;
  }

  const bool started = WiFi.softAP(AP_SSID, AP_PASSWORD, 1, false, 4);
  if (!started || WiFi.softAPIP() != apIp) {
    Serial.println("Failed to start ESP32-HYDRODECK hotspot; startup will retry.");
    return;
  }
  Serial.print("ESP32 hotspot is broadcasting: ");
  Serial.print(AP_SSID);
  Serial.print(" at ");
  Serial.println(WiFi.softAPIP());
}

// HTTP API Control Handlers
void handleControl() {
  // Handle traditional actuator control: /control?actuator=<name>&state=<0|1>
  if (server.hasArg("actuator") && server.hasArg("state")) {
    String actuator = server.arg("actuator");
    bool state = (server.arg("state") == "1" || server.arg("state") == "on");
    if (actuator == "growLight") {
      const String requestId = server.arg("requestId");
      if (requestId.length() > 0 && !isValidCommandRequestId(requestId)) {
        server.send(400, "text/plain", "Invalid request ID");
        return;
      }
      if (requestId.length() > 0 && wasCommandConsumed(requestId)) {
        server.send(200, "text/plain", "Duplicate request ignored");
        return;
      }
      if (requestId.length() > 0) rememberConsumedCommand(requestId, "accepted");
      setGrowLight(state);
      publishActuatorState(actuator, state);
      if (requestId.length() > 0) publishCommandAck(requestId, "completed");
      server.send(200, "text/plain", "OK");
      return;
    }
    const int pumpNum = pumpNumberForActuator(actuator);
    if (pumpNum > 0) {
      const String requestId = server.arg("requestId");
      if (!state) {
        stopPumpRun(pumpNum);
        server.send(200, "text/plain", "OK");
        return;
      }
      if (requestId.length() > 0 && !isValidCommandRequestId(requestId)) {
        server.send(400, "text/plain", "Invalid request ID");
        return;
      }
      if (requestId.length() > 0 &&
          wasCommandConsumed(requestId)) {
        server.send(200, "text/plain", "Duplicate request ignored");
        return;
      }
      if (requestId.length() > 0) rememberConsumedCommand(requestId);
      if (!executePumpRequest(actuator, true, requestId)) {
        publishCommandAck(requestId, "rejected");
        server.send(409, "text/plain", "Pump busy, threshold blocked, or water safety limit reached");
        return;
      }
      server.send(200, "text/plain", "OK");
      return;
    }
  }

  // Handle command-based control: /control?command=<command>
  // Used for PH_ADJUST:<bed>:<direction> format from Flutter app
  if (server.hasArg("command")) {
    String command = server.arg("command");
    if (command.startsWith("PH_ADJUST:")) {
      parsePhAdjustCommand(command);
      server.send(200, "text/plain", "OK");
      return;
    }
    // Could add other command formats here if needed
  }

  server.send(400, "text/plain", "Bad Request");
}

  ssl_client.setConnectionTimeout(1000);
  ssl_client.setHandshakeTimeout(5);

    configTzTime("PHT-8", "pool.ntp.org", "time.nist.gov");

  initializeApp(aClient, app, getAuth(user_auth), processFirebaseResult, "authTask");

  Database.url(DATABASE_URL);
  firebaseInitialized = true;
  Serial.println("Firebase authentication starting...");
}

void setup() {
  Serial.begin(115200);

  // Initialize Actuators (Default OFF)
  configureSafeOutput(waterPump1Pin, PUMP_OFF);
  configureSafeOutput(phUp1Pin, PUMP_OFF);
  configureSafeOutput(phDown1Pin, PUMP_OFF);
  configureSafeOutput(nutrientPump1Pin, PUMP_OFF);
  configureSafeOutput(waterPump2Pin, PUMP_OFF);
  configureSafeOutput(phUp2Pin, PUMP_OFF);
  configureSafeOutput(phDown2Pin, PUMP_OFF);
  configureSafeOutput(nutrientPump2Pin, PUMP_OFF);
  configureSafeOutput(growLightPin, LIGHT_OFF);

  // Initialize Sensors
  pinMode(WATER_SENSOR_PIN1, INPUT_PULLUP);
  pinMode(WATER_SENSOR_PIN2, INPUT_PULLUP);
  pinMode(PH_PIN1, INPUT);
  pinMode(PH_PIN2, INPUT);
  pinMode(TDS_PIN1, INPUT);
  pinMode(TDS_PIN2, INPUT);

  analogReadResolution(12);
  analogSetPinAttenuation(PH_PIN1, ADC_11db);
  analogSetPinAttenuation(PH_PIN2, ADC_11db);
  analogSetPinAttenuation(TDS_PIN1, ADC_11db);
  analogSetPinAttenuation(TDS_PIN2, ADC_11db);

  // Temperature Sensors Setup
  sensors1.begin();
  sensors1.requestTemperatures();
  sensors2.begin();
  sensors2.requestTemperatures();
  delay(750);

  float tempReading1 = sensors1.getTempCByIndex(0);
  if (tempReading1 != DEVICE_DISCONNECTED_C && tempReading1 > -50.0) {
    currentTemperature1 = tempReading1;
    temperatureSensorValid1 = true;
    lastTemperatureValidAt1 = millis();
  }
  float tempReading2 = sensors2.getTempCByIndex(0);
  if (tempReading2 != DEVICE_DISCONNECTED_C && tempReading2 > -50.0) {
    currentTemperature2 = tempReading2;
    temperatureSensorValid2 = true;
    lastTemperatureValidAt2 = millis();
  }
  sensors1.setWaitForConversion(false);
  sensors2.setWaitForConversion(false);

  // Wi-Fi and hotspot setup
  wifiPreferences.begin("wifi", false);
  commandPreferences.begin("pumpcmd", false);
  setGrowLight(commandPreferences.getBool("growLightOn", false));
  lastIntervalSnapshotIdentity = commandPreferences.getString("lastSnapshotIdentity", "");
  String savedSsid = wifiPreferences.getString("ssid", ssid);
  String savedPassword = wifiPreferences.getString("password", password);
  savedSsid.toCharArray(ssid, sizeof(ssid));
  savedPassword.toCharArray(password, sizeof(password));
  maintainProvisioningHotspot();

  WiFi.begin(ssid, password);
  wifiConnectInProgress = true;
  wifiConnectStarted = millis();
  Serial.println("Connecting to home Wi-Fi in the background; hotspot remains available.");

  server.on("/", HTTP_GET, handleRootIdent);
  server.on("/provision", HTTP_GET, handleProvision);
  server.on("/settings", HTTP_GET, handleSettings);
  server.on("/status", HTTP_GET, handleCombinedStatus);
  server.on("/control", HTTP_GET, handleControl); // Control via URL e.g. /control?actuator=phUp1&state=1 OR /control?command=PH_ADJUST:1:UP
  server.begin();
}

void loop() {
  app.loop();
  server.handleClient();
  unsigned long now = millis();

  maintainProvisioningHotspot();
  if (WiFi.status() == WL_CONNECTED && !mdnsStarted) {
    mdnsStarted = MDNS.begin("hydrodeck");
    if (mdnsStarted) Serial.println("mDNS: hydrodeck.local");
  }

  // Apply replacement Wi-Fi credentials without disabling the setup AP.
  if (provisioningPending) {
    if (!wifiConnectInProgress) {
      WiFi.disconnect(false, false);
      WiFi.begin(ssid, password);
      wifiConnectInProgress = true;
      wifiConnectStarted = now;
      Serial.println("Trying the newly provisioned Wi-Fi network...");
    } else if (WiFi.status() == WL_CONNECTED && WiFi.SSID() == String(ssid)) {
      provisioningPending = false;
      wifiConnectInProgress = false;
      initializeFirebase();
      Serial.println("Replacement Wi-Fi connected; setup hotspot remains available.");
    } else if (now - wifiConnectStarted >= wifiConnectTimeout) {
      wifiConnectInProgress = false;
      provisioningPending = false;
      lastWifiRetry = now;
      Serial.println("New Wi-Fi attempt timed out; setup hotspot remains available.");
    }
  } else if (WiFi.status() == WL_CONNECTED) {
    wifiConnectInProgress = false;
    initializeFirebase();
  } else {
    if (firebaseInitialized) {
      firebaseInitialized = false;
      Serial.println("Wi-Fi lost; Firebase will re-authenticate after reconnect.");
    }
    if (wifiConnectInProgress && now - wifiConnectStarted >= wifiConnectTimeout) {
      wifiConnectInProgress = false;
    }
    if (!wifiConnectInProgress && now - lastWifiRetry >= wifiRetryInterval) {
      lastWifiRetry = now;
      WiFi.begin(ssid, password);
      wifiConnectInProgress = true;
      wifiConnectStarted = now;
      Serial.println("Retrying saved Wi-Fi credentials...");
    }
  }

  if (now - lastRemoteCommandPoll >= remoteCommandInterval) {
    lastRemoteCommandPoll = now;
    requestRemoteActuatorCommand();
  }

  if (!remoteCommandPending && pendingCommandCleanupWrites == 0) {
    if (now - lastPhLimitPoll >= phLimitPollInterval) {
      lastPhLimitPoll = now;
      requestPhLimit();
    }

    if (now - lastActiveBatchPoll >= activeBatchPollInterval) {
      lastActiveBatchPoll = now;
      requestActiveBatchNumber();
    }
    requestIntervalSnapshot();

    if (firebaseInitialized && app.ready() &&
        now - lastFirebaseHeartbeat >= firebaseHeartbeatInterval) {
      lastFirebaseHeartbeat = now;
      Database.set<object_t>(
        aClient,
        "/hydrodeck/lastSeen",
        object_t("{\".sv\":\"timestamp\"}")
      );
    }
  }

  // Sampling TDS Sensor 1 & Sensor 2
  static unsigned long sampleTime = 0;
  if (now - sampleTime > 40U) {
    sampleTime = now;
    analogBuffer1[analogBufferIndex1] = analogRead(TDS_PIN1);
    analogBufferIndex1 = (analogBufferIndex1 + 1) % SCOUNT;

    analogBuffer2[analogBufferIndex2] = analogRead(TDS_PIN2);
    analogBufferIndex2 = (analogBufferIndex2 + 1) % SCOUNT;
  }

  // Calculate TDS Values
  static unsigned long tdsTime = 0;
  if (now - tdsTime > 800U) {
    tdsTime = now;

    // Buffer 1
    for (int i = 0; i < SCOUNT; i++) analogBufferTemp1[i] = analogBuffer1[i];
    float avgV1 = getMedianNum(analogBufferTemp1, SCOUNT) * (float)VREF / 4096.0;
    avgV1 = constrain(avgV1, 0.0f, (float)VREF);
    float compCoeff1 = 1.0 + 0.02 * (currentTemperature1 - 25.0);
    float compV1 = avgV1 / compCoeff1;
    float tdsVal1 = (133.42 * compV1 * compV1 * compV1 - 255.86 * compV1 * compV1 + 857.39 * compV1) * 0.5;
    currentTdsValue1 = max(0.0f, tdsVal1) * tdsCalibrationFactor;
    tdsSensorValid1 = true;
    lastTdsValidAt1 = now;

    // Buffer 2
    for (int i = 0; i < SCOUNT; i++) analogBufferTemp2[i] = analogBuffer2[i];
    float avgV2 = getMedianNum(analogBufferTemp2, SCOUNT) * (float)VREF / 4096.0;
    avgV2 = constrain(avgV2, 0.0f, (float)VREF);
    float compCoeff2 = 1.0 + 0.02 * (currentTemperature2 - 25.0);
    float compV2 = avgV2 / compCoeff2;
    float tdsVal2 = (133.42 * compV2 * compV2 * compV2 - 255.86 * compV2 * compV2 + 857.39 * compV2) * 0.5;
    currentTdsValue2 = max(0.0f, tdsVal2) * tdsCalibrationFactor;
    tdsSensorValid2 = true;
    lastTdsValidAt2 = now;
  }

  // Reading pH Sensors with 1-minute delay logic
  if (now - lastPhCheck >= phInterval) {
    lastPhCheck = now;
    currentPhValue1 = readPhSensor(PH_PIN1, slope1);
    currentPhValue2 = readPhSensor(PH_PIN2, slope2);
    phSensorValid1 = true;
    phSensorValid2 = true;
    lastPhValidAt1 = now;
    lastPhValidAt2 = now;

    serviceAutomaticPhControl(1, currentPhValue1, now);
    serviceAutomaticPhControl(2, currentPhValue2, now);
  }

  const bool waterFull1 = digitalRead(WATER_SENSOR_PIN1) == HIGH;
  const bool waterFull2 = digitalRead(WATER_SENSOR_PIN2) == HIGH;
  if (waterFull1) {
    waterLowSince1 = 0;
  } else {
    manualWaterShortRuns1 = 0;
    if (waterLowSince1 == 0) waterLowSince1 = now;
    if (now - waterLowSince1 >= waterAutoDelay && !pumpRuns[0].active) {
      startPumpRun(1, 0, true);
    }
  }
  if (waterFull2) {
    waterLowSince2 = 0;
  } else {
    manualWaterShortRuns2 = 0;
    if (waterLowSince2 == 0) waterLowSince2 = now;
    if (now - waterLowSince2 >= waterAutoDelay && !pumpRuns[4].active) {
      startPumpRun(5, 0, true);
    }
  }

  servicePumpRuns();

  // Reading Temperature Sensors
  if (now - lastTempCheck >= tempInterval) {
    lastTempCheck = now;

    float reading1 = sensors1.getTempCByIndex(0);
    if (reading1 != DEVICE_DISCONNECTED_C && reading1 > -50.0) {
      currentTemperature1 = reading1;
      temperatureSensorValid1 = true;
      lastTemperatureValidAt1 = now;
    }
    sensors1.requestTemperatures();

    float reading2 = sensors2.getTempCByIndex(0);
    if (reading2 != DEVICE_DISCONNECTED_C && reading2 > -50.0) {
      currentTemperature2 = reading2;
      temperatureSensorValid2 = true;
      lastTemperatureValidAt2 = now;
    }
    sensors2.requestTemperatures();
  }

  // Syncing with Firebase Realtime Database
    if (firebaseInitialized && app.ready() && !remoteCommandPending &&
      pendingCommandCleanupWrites == 0 &&
      now - lastFirebasePush >= firebaseInterval) {
    lastFirebasePush = now;

    String liveTelemetryJson = "{";
    liveTelemetryJson += "\"tds1\":" + String(currentTdsValue1, 0) + ",";
    liveTelemetryJson += "\"ph1\":" + String(currentPhValue1, 2) + ",";
    liveTelemetryJson += "\"temp1\":" + String(currentTemperature1, 1) + ",";
    liveTelemetryJson += "\"water1\":\"" + String(digitalRead(WATER_SENSOR_PIN1) == HIGH ? "full" : "low") + "\",";
    liveTelemetryJson += "\"tds2\":" + String(currentTdsValue2, 0) + ",";
    liveTelemetryJson += "\"ph2\":" + String(currentPhValue2, 2) + ",";
    liveTelemetryJson += "\"temp2\":" + String(currentTemperature2, 1) + ",";
    liveTelemetryJson += "\"water2\":\"" + String(digitalRead(WATER_SENSOR_PIN2) == HIGH ? "full" : "low") + "\",";
    liveTelemetryJson += "\"actuators\":{";
    liveTelemetryJson += "\"waterPump1\":" + String(digitalRead(waterPump1Pin) == PUMP_ON ? "true" : "false") + ",";
    liveTelemetryJson += "\"phUp1\":" + String(digitalRead(phUp1Pin) == PUMP_ON ? "true" : "false") + ",";
    liveTelemetryJson += "\"phDown1\":" + String(digitalRead(phDown1Pin) == PUMP_ON ? "true" : "false") + ",";
    liveTelemetryJson += "\"nutrient1\":" + String(digitalRead(nutrientPump1Pin) == PUMP_ON ? "true" : "false") + ",";
    liveTelemetryJson += "\"waterPump2\":" + String(digitalRead(waterPump2Pin) == PUMP_ON ? "true" : "false") + ",";
    liveTelemetryJson += "\"phUp2\":" + String(digitalRead(phUp2Pin) == PUMP_ON ? "true" : "false") + ",";
    liveTelemetryJson += "\"phDown2\":" + String(digitalRead(phDown2Pin) == PUMP_ON ? "true" : "false") + ",";
    liveTelemetryJson += "\"nutrient2\":" + String(digitalRead(nutrientPump2Pin) == PUMP_ON ? "true" : "false") + ",";
    liveTelemetryJson += "\"growLight\":" + String(digitalRead(growLightPin) == LIGHT_ON ? "true" : "false") + "},";
    liveTelemetryJson += "\"wifiConnected\":true,";
    liveTelemetryJson += "\"firebaseReady\":true";
    liveTelemetryJson += "}";
    Database.update<object_t>(aClient, "/hydrodeck", object_t(liveTelemetryJson.c_str()));

    if (aClient.lastError().code() != 0) {
      Serial.print("Firebase live telemetry update failed, code: ");
      Serial.println(aClient.lastError().code());
    } else {
      Serial.println("Firebase update sent for dual sensor sets and actuators.");
    }
  }
}
