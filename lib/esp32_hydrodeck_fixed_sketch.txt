int getMedianNum(int values[], int length) {
  int bTab[length];
  for (byte i = 0; i < length; i++) {
    bTab[i] = values[i];
  }
  int i, j, bTemp;
  for (j = 0; j < length - 1; j++) {
    for (i = 0; i < length - j - 1; i++) {
      if (bTab[i] > bTab[i + 1]) {
        bTemp = bTab[i];
        bTab[i] = bTab[i + 1];
        bTab[i + 1] = bTemp;
      }
    }
  }
  if ((length & 1) > 0) {
    bTemp = bTab[(length - 1) / 2];
  } else {
    bTemp = (bTab[length / 2] + bTab[length / 2 - 1]) / 2;
  }
  return bTemp;
}

float readPhSensor(int pin, float sensorSlope) {
  int measurings = 0;
  int samples = 10;
  for (int i = 0; i < samples; i++) {
    measurings += analogRead(pin);
    delay(10);
  }
  float voltage = (v_ref / 4095.0f) * (float)measurings / (float)samples;
  float ph = 7.0 + ((2.5 - voltage) / sensorSlope);
  if (ph < 0.0) ph = 0.0;
  if (ph > 14.0) ph = 14.0;
  return ph;
}

// Actuator control functions
void setPump(int pumpNum, bool turnOn) {
  int pin = -1;
  switch (pumpNum) {
    case 1: pin = waterPump1Pin; break;
    case 2: pin = phUp1Pin; break;
    case 3: pin = phDown1Pin; break;
    case 4: pin = nutrientPump1Pin; break;
    case 5: pin = waterPump2Pin; break;
    case 6: pin = phUp2Pin; break;
    case 7: pin = phDown2Pin; break;
    case 8: pin = nutrientPump2Pin; break;
  }
  if (pin != -1) {
    digitalWrite(pin, turnOn ? PUMP_ON : PUMP_OFF);
  }
}

// Parse PH_ADJUST:<bed>:<direction> command format
void parsePhAdjustCommand(const String& command) {
  // Expected format: "PH_ADJUST:<bed>:<direction>"
  // Example: "PH_ADJUST:1:UP" or "PH_ADJUST:2:DOWN"
  if (!command.startsWith("PH_ADJUST:")) return;

  String payload = command.substring(10); // Remove "PH_ADJUST:" prefix
  int firstColon = payload.indexOf(':');
  if (firstColon == -1) return;

  String bedStr = payload.substring(0, firstColon);
  String direction = payload.substring(firstColon + 1);

  int bed = bedStr.toInt();
  if (bed < 1 || bed > 2) return;

  String actuatorName;
  if (direction == "UP") {
    actuatorName = (bed == 1) ? "phUp1" : "phUp2";
  } else if (direction == "DOWN") {
    actuatorName = (bed == 1) ? "phDown1" : "phDown2";
  } else {
    return; // Invalid direction
  }

  // Check if pH is actually in range that needs adjustment
  bool shouldActivate = false;
  if (bed == 1) {
    if (direction == "UP" && currentPhValue1 < minPhThreshold) shouldActivate = true;
    if (direction == "DOWN" && currentPhValue1 > maxPhThreshold) shouldActivate = true;
  } else { // bed == 2
    if (direction == "UP" && currentPhValue2 < minPhThreshold) shouldActivate = true;
    if (direction == "DOWN" && currentPhValue2 > maxPhThreshold) shouldActivate = true;
  }

  if (!shouldActivate) return;

  // Set durations based on bed
  unsigned long durationMs = (bed == 1 ? BED1_PH_PUMP_SECONDS : BED2_PH_PUMP_SECONDS) * 1000UL;

  // Find pump number
  int pumpNum = pumpNumberForActuator(actuatorName);
  if (pumpNum == 0) return;

  // Check if pump is already active
  if (pumpRuns[pumpNum - 1].active) return;

  // Check for conflicting pH pumps (can't have both up and down active at same time)
  if ((pumpNum == 2 || pumpNum == 3) && (pumpRuns[1].active || pumpRuns[2].active)) return;
  if ((pumpNum == 6 || pumpNum == 7) && (pumpRuns[5].active || pumpRuns[6].active)) return;

  // Start the pump run
  PumpRunState& run = pumpRuns[pumpNum - 1];
  run.active = true;
  run.stopWhenFull = false;
  run.stopAt = millis() + durationMs;
  setPump(pumpNum, true);
  publishActuatorState(actuatorName, true);
}

bool waterSensorFullForPump(int pumpNum) {
  return digitalRead(pumpNum == 1 ? WATER_SENSOR_PIN1 : WATER_SENSOR_PIN2) == HIGH;
}

bool wasCommandConsumed(const String& requestId) {
  String consumedIds = commandPreferences.getString("recentIds", "");
  const String token = "," + requestId + ",";
  return ("," + consumedIds + ",").indexOf(token) >= 0;
}

void rememberConsumedCommand(const String& requestId) {
  String consumedIds = commandPreferences.getString("recentIds", "");
  if (wasCommandConsumed(requestId)) return;
  if (consumedIds.length() > 0) consumedIds += ",";
  consumedIds += requestId;
  while (consumedIds.length() > 700) {
    const int separator = consumedIds.indexOf(',');
    if (separator < 0) break;
    consumedIds.remove(0, separator + 1);
  }
  commandPreferences.putString("recentIds", consumedIds);
}

bool startPumpRun(int pumpNum, unsigned long durationMs, bool stopWhenFull, const String& requestId = "") {
  if (pumpNum < 1 || pumpNum > 8 || pumpRuns[pumpNum - 1].active) return false;
  if ((pumpNum == 1 && pumpRuns[4].active) || (pumpNum == 5 && pumpRuns[0].active)) return false;
  if ((pumpNum == 2 || pumpNum == 3) && (pumpRuns[1].active || pumpRuns[2].active)) return false;
  if ((pumpNum == 6 || pumpNum == 7) && (pumpRuns[5].active || pumpRuns[6].active)) return false;
  PumpRunState& run = pumpRuns[pumpNum - 1];
  run.active = true;
  run.stopWhenFull = stopWhenFull;
  run.stopAt = durationMs == 0 ? 0 : millis() + durationMs;
  activeCommandIds[pumpNum - 1] = requestId;
  setPump(pumpNum, true);
  publishActuatorState(remoteActuatorNames[pumpNum - 1], true);
  publishCommandAck(requestId, "running");
  return true;
}

void serviceAutomaticPhControl(int bed, float phValue, unsigned long now) {
  const int upPump = bed == 1 ? 2 : 6;
  const int downPump = bed == 1 ? 3 : 7;
  const unsigned long pulseDuration =
      (bed == 1 ? BED1_PH_PUMP_SECONDS : BED2_PH_PUMP_SECONDS) * 1000UL;
  unsigned long& lowSince = bed == 1 ? bed1PhLowSince : bed2PhLowSince;
  unsigned long& highSince = bed == 1 ? bed1PhHighSince : bed2PhHighSince;

  if (phValue < minPhThreshold) {
    highSince = 0;
    if (lowSince == 0) lowSince = now;
    if (now - lowSince >= phDelayThreshold &&
        startPumpRun(upPump, pulseDuration, false)) {
      lowSince = now;
    }
  } else if (phValue > maxPhThreshold) {
    lowSince = 0;
    if (highSince == 0) highSince = now;
    if (now - highSince >= phDelayThreshold &&
        startPumpRun(downPump, pulseDuration, false)) {
      highSince = now;
    }
  } else {
    lowSince = 0;
    highSince = 0;
  }
}

void stopPumpRun(int pumpNum) {
  if (pumpNum < 1 || pumpNum > 8) return;
  PumpRunState& run = pumpRuns[pumpNum - 1];
  if (!run.active) return;
  setPump(pumpNum, false);
  const String requestId = activeCommandIds[pumpNum - 1];
  activeCommandIds[pumpNum - 1] = "";
  run.active = false;
  run.stopWhenFull = false;
  run.stopAt = 0;
  publishActuatorState(remoteActuatorNames[pumpNum - 1], false);
  publishCommandAck(requestId, "completed");
}

bool executePumpRequest(const String& actuator, bool enforceManualRules, const String& requestId = "") {
  const int pumpNum = pumpNumberForActuator(actuator);
  if (pumpNum == 0) return false;

  unsigned long durationMs = 0;
  bool stopWhenFull = false;
  int* manualShortRunCounter = nullptr;
  if (actuator == "phUp1" || actuator == "phDown1") {
    if ((actuator == "phUp1" && currentPhValue1 > maxPhThreshold) ||
      (actuator == "phDown1" && currentPhValue1 < minPhThreshold)) return false;
    durationMs = BED1_PH_PUMP_SECONDS * 1000UL;
  } else if (actuator == "phUp2" || actuator == "phDown2") {
    if ((actuator == "phUp2" && currentPhValue2 > maxPhThreshold) ||
      (actuator == "phDown2" && currentPhValue2 < minPhThreshold)) return false;
    durationMs = BED2_PH_PUMP_SECONDS * 1000UL;
  } else if (actuator == "nutrient1") {
    durationMs = BED1_NUTRIENT_PUMP_SECONDS * 1000UL;
  } else if (actuator == "nutrient2") {
    durationMs = BED2_NUTRIENT_PUMP_SECONDS * 1000UL;
  } else if (actuator == "waterPump1" || actuator == "waterPump2") {
    if (pumpRuns[pumpNum - 1].active) return false;
    const bool isFull = waterSensorFullForPump(pumpNum);
    if (isFull) {
      durationMs = FULL_WATER_MANUAL_PULSE_SECONDS * 1000UL;
      if (enforceManualRules) {
        int& count = actuator == "waterPump1" ? manualWaterShortRuns1 : manualWaterShortRuns2;
        if (count >= 2) return false;
        manualShortRunCounter = &count;
      }
    } else {
      stopWhenFull = true;
    }
  }
  const bool started = startPumpRun(pumpNum, durationMs, stopWhenFull, requestId);
  if (started && manualShortRunCounter != nullptr) (*manualShortRunCounter)++;
  return started;
}

void servicePumpRuns() {
  const unsigned long now = millis();
  for (int pumpNum = 1; pumpNum <= 8; pumpNum++) {
    PumpRunState& run = pumpRuns[pumpNum - 1];
    if (!run.active) continue;
    if (run.stopWhenFull && waterSensorFullForPump(pumpNum)) {
      stopPumpRun(pumpNum);
    } else if (run.stopAt != 0 && (long)(now - run.stopAt) >= 0) {
      stopPumpRun(pumpNum);
    }
  }
}


void processRemoteActuatorResult(AsyncResult& result) {
  remoteCommandPending = false;
  const String actuator = remoteActuatorNames[remoteCommandIndex];
  if (result.isError() || !result.available()) {
    remoteCommandIndex = (remoteCommandIndex + 1) % 9;
    return;
  }

  const String commandPath = "/hydrodeck/commands/" + actuator;
  const String payload = result.to<RealtimeDatabaseResult>().to<String>();
  String requestId = payload;
  bool growLightState = false;
  bool validGrowLightCommand = true;

  if (actuator == "growLight") {
    const int separator = payload.lastIndexOf('|');
    if (separator <= 0) {
      validGrowLightCommand = false;
    } else {
      requestId = payload.substring(0, separator);
      const String requestedState = payload.substring(separator + 1);
      validGrowLightCommand = requestedState == "0" || requestedState == "1";
      growLightState = requestedState == "1";
    }
  }

  if (payload.length() > 0) {
    Database.set<String>(aClient, commandPath, "");
    if (requestId.length() > 0) {
      if (wasCommandConsumed(requestId)) {
        publishCommandAck(requestId, "completed");
      } else {
        rememberConsumedCommand(requestId);
        if (actuator == "growLight") {
          if (validGrowLightCommand) {
            setGrowLight(growLightState);
            publishActuatorState(actuator, growLightState);
            publishCommandAck(requestId, "completed");
          } else {
            publishCommandAck(requestId, "failed");
          }
        } else if (!executePumpRequest(actuator, true, requestId)) {
          publishCommandAck(requestId, "failed");
        }
      }
    }
  }
  result.clear();
  remoteCommandIndex = (remoteCommandIndex + 1) % 9;
}

void requestRemoteActuatorCommand() {
  if (!firebaseInitialized || !app.ready() || remoteCommandPending || phLimitRequestPending) return;

  const String actuator = remoteActuatorNames[remoteCommandIndex];
  const String path = "/hydrodeck/commands/" + actuator;
  remoteCommandPending = true;
  Database.get(aClient, path, processRemoteActuatorResult, false, "remoteCommand");
}

void processPhLimitResult(AsyncResult& result) {
  phLimitRequestPending = false;
  if (!result.isError() && result.available()) {
    const float value = result.to<RealtimeDatabaseResult>().to<float>();
    if (value >= 0.0f && value <= 14.0f) {
      if (phLimitRequestIndex == 0 && value < maxPhThreshold) minPhThreshold = value;
      if (phLimitRequestIndex == 1 && value > minPhThreshold) maxPhThreshold = value;
    }
    result.clear();
  }
  phLimitRequestIndex = (phLimitRequestIndex + 1) % 2;
}

void requestPhLimit() {
  if (!firebaseInitialized || !app.ready() || remoteCommandPending || phLimitRequestPending) return;
  const String key = phLimitRequestIndex == 0 ? "minPh" : "maxPh";
  phLimitRequestPending = true;
  Database.get(aClient, "/hydrodeck/settings/" + key, processPhLimitResult, false, "phLimit");
}

void handleProvision() {
  if (!server.hasArg("ssid") || !server.hasArg("password")) {
    server.send(400, "text/plain", "Missing Wi-Fi credentials");
    return;
  }
  String newSsid = server.arg("ssid");
  String newPassword = server.arg("password");
  newSsid.trim();
  newPassword.trim();
  if (newSsid.isEmpty() || newSsid.length() >= sizeof(ssid) || newPassword.length() >= sizeof(password)) {
    server.send(400, "text/plain", "Invalid Wi-Fi credentials");
    return;
  }
  newSsid.toCharArray(ssid, sizeof(ssid));
  newPassword.toCharArray(password, sizeof(password));
  wifiPreferences.putString("ssid", ssid);
  wifiPreferences.putString("password", password);
  provisioningPending = true;
  wifiConnectInProgress = false;
  firebaseInitialized = false;
  server.send(200, "text/plain", "ESP32 is connecting to Wi-Fi");
}

void handleSettings() {

    server.send(400, "text/plain", "Missing pH thresholds");
    return;
  }
  const float minimum = server.arg("minPh").toFloat();
  const float maximum = server.arg("maxPh").toFloat();

    return;
  }
  minPhThreshold = minimum;
  maxPhThreshold = maximum;
  server.send(200, "text/plain", "OK");
}

void maintainProvisioningHotspot() {
  if (WiFi.getMode() == WIFI_AP_STA && WiFi.softAPIP() == apIp) return;

  const unsigned long now = millis();
  if (lastHotspotAttempt != 0 && now - lastHotspotAttempt < hotspotRetryInterval) return;
  lastHotspotAttempt = now;

  if (WiFi.getMode() != WIFI_AP_STA) {
    if (!WiFi.mode(WIFI_AP_STA)) {
      Serial.println("Failed to enable AP+STA mode; hotspot startup will retry.");
      return;
    }
    delay(100);
  }

  WiFi.setSleep(false);
  WiFi.softAPdisconnect(false);
  if (!WiFi.softAPConfig(apIp, apGateway, apSubnet)) {
    Serial.println("Failed to configure the ESP32 hotspot IP; startup will retry.");
    return;
  }

  const bool started = WiFi.softAP(AP_SSID, AP_PASSWORD, 1, false, 4);
  if (!started || WiFi.softAPIP() != apIp) {
    Serial.println("Failed to start ESP32-HYDRODECK hotspot; startup will retry.");
    return;
  }
  Serial.print("ESP32 hotspot is broadcasting: ");
  Serial.print(AP_SSID);
  Serial.print(" at ");
  Serial.println(WiFi.softAPIP());
}

// HTTP API Control Handlers
void handleControl() {
  // Handle traditional actuator control: /control?actuator=<name>&state=<0|1>
  if (server.hasArg("actuator") && server.hasArg("state")) {
    String actuator = server.arg("actuator");
    bool state = (server.arg("state") == "1" || server.arg("state") == "on");
    if (actuator == "growLight") {
      setGrowLight(state);
      publishActuatorState(actuator, state);
      server.send(200, "text/plain", "OK");
      return;
    }
    const int pumpNum = pumpNumberForActuator(actuator);
    if (pumpNum > 0) {
      const String requestId = server.arg("requestId");
      if (!state) {
        stopPumpRun(pumpNum);
        if (firebaseInitialized && app.ready()) {
          Database.set<String>(aClient, "/hydrodeck/commands/" + actuator, "");
        }
        server.send(200, "text/plain", "OK");
        return;
      }
      if (requestId.length() > 0 &&
          wasCommandConsumed(requestId)) {
        server.send(200, "text/plain", "Duplicate request ignored");
        return;
      }
      if (requestId.length() > 0) rememberConsumedCommand(requestId);
      if (!executePumpRequest(actuator, true, requestId)) {
        publishCommandAck(requestId, "failed");
        server.send(409, "text/plain", "Pump busy, threshold blocked, or water safety limit reached");
        return;
      }
      if (firebaseInitialized && app.ready()) {
        Database.set<String>(aClient, "/hydrodeck/commands/" + actuator, "");
      }
      server.send(200, "text/plain", "OK");
      return;
    }
  }

  // Handle command-based control: /control?command=<command>
  // Used for PH_ADJUST:<bed>:<direction> format from Flutter app
  if (server.hasArg("command")) {
    String command = server.arg("command");
    if (command.startsWith("PH_ADJUST:")) {
      parsePhAdjustCommand(command);
      server.send(200, "text/plain", "OK");
      return;
    }
    // Could add other command formats here if needed
  }

  server.send(400, "text/plain", "Bad Request");
}

void processFirebaseResult(AsyncResult& firebaseResult) {
  if (!firebaseResult.isResult()) return;
  if (firebaseResult.isError()) {
    Serial.print("Firebase error [");
    Serial.print(firebaseResult.uid());
    Serial.print("] code: ");
    Serial.print(firebaseResult.error().code());
    Serial.print(" message: ");
    Serial.println(firebaseResult.error().message());
  }
  if (firebaseResult.isEvent()) {
    Serial.print("Firebase event: ");
    Serial.println(firebaseResult.eventLog().message());
  }
  if (firebaseResult.isDebug()) {
    Serial.print("Firebase debug: ");
    Serial.println(firebaseResult.debug());
  }
}

void initializeFirebase() {
  if (firebaseInitialized || WiFi.status() != WL_CONNECTED) return;
  ssl_client.setInsecure();
  ssl_client.setConnectionTimeout(1000);
  ssl_client.setHandshakeTimeout(5);
  initializeApp(aClient, app, getAuth(user_auth), processFirebaseResult, "authTask");
  app.getApp<RealtimeDatabase>(Database);
  Database.url(DATABASE_URL);
  firebaseInitialized = true;
  Serial.println("Firebase authentication starting...");
}

void setup() {
  Serial.begin(115200);

  // Initialize Actuators (Default OFF)
  pinMode(waterPump1Pin, OUTPUT); digitalWrite(waterPump1Pin, PUMP_OFF);
  pinMode(phUp1Pin, OUTPUT); digitalWrite(phUp1Pin, PUMP_OFF);
  pinMode(phDown1Pin, OUTPUT); digitalWrite(phDown1Pin, PUMP_OFF);
  pinMode(nutrientPump1Pin, OUTPUT); digitalWrite(nutrientPump1Pin, PUMP_OFF);
  pinMode(waterPump2Pin, OUTPUT); digitalWrite(waterPump2Pin, PUMP_OFF);
  pinMode(phUp2Pin, OUTPUT); digitalWrite(phUp2Pin, PUMP_OFF);
  pinMode(phDown2Pin, OUTPUT); digitalWrite(phDown2Pin, PUMP_OFF);
  pinMode(nutrientPump2Pin, OUTPUT); digitalWrite(nutrientPump2Pin, PUMP_OFF);
  pinMode(growLightPin, OUTPUT); digitalWrite(growLightPin, LIGHT_OFF);

  // Initialize Sensors
  pinMode(WATER_SENSOR_PIN1, INPUT_PULLUP);
  pinMode(WATER_SENSOR_PIN2, INPUT_PULLUP);
  pinMode(PH_PIN1, INPUT);
  pinMode(PH_PIN2, INPUT);
  pinMode(TDS_PIN1, INPUT);
  pinMode(TDS_PIN2, INPUT);

  analogReadResolution(12);
  analogSetPinAttenuation(PH_PIN1, ADC_11db);
  analogSetPinAttenuation(PH_PIN2, ADC_11db);
  analogSetPinAttenuation(TDS_PIN1, ADC_11db);
  analogSetPinAttenuation(TDS_PIN2, ADC_11db);

  // Temperature Sensors Setup
  sensors1.begin();
  sensors1.requestTemperatures();
  sensors2.begin();
  sensors2.requestTemperatures();
  delay(750);

  float tempReading1 = sensors1.getTempCByIndex(0);
  if (tempReading1 != DEVICE_DISCONNECTED_C && tempReading1 > -50.0) {
    currentTemperature1 = tempReading1;
  }
  float tempReading2 = sensors2.getTempCByIndex(0);
  if (tempReading2 != DEVICE_DISCONNECTED_C && tempReading2 > -50.0) {
    currentTemperature2 = tempReading2;
  }
  sensors1.setWaitForConversion(false);
  sensors2.setWaitForConversion(false);

  // Wi-Fi and hotspot setup
  wifiPreferences.begin("wifi", false);
  commandPreferences.begin("pumpcmd", false);
  String savedSsid = wifiPreferences.getString("ssid", ssid);
  String savedPassword = wifiPreferences.getString("password", password);
  savedSsid.toCharArray(ssid, sizeof(ssid));
  savedPassword.toCharArray(password, sizeof(password));
  maintainProvisioningHotspot();

  WiFi.begin(ssid, password);
  wifiConnectInProgress = true;
  wifiConnectStarted = millis();
  Serial.println("Connecting to home Wi-Fi in the background; hotspot remains available.");

  server.on("/", HTTP_GET, handleRootIdent);
  server.on("/provision", HTTP_GET, handleProvision);
  server.on("/settings", HTTP_GET, handleSettings);
  server.on("/status", HTTP_GET, handleCombinedStatus);
  server.on("/control", HTTP_GET, handleControl); // Control via URL e.g. /control?actuator=phUp1&state=1 OR /control?command=PH_ADJUST:1:UP
  server.begin();
}

void loop() {
  app.loop();
  server.handleClient();
  unsigned long now = millis();

  maintainProvisioningHotspot();
  if (WiFi.status() == WL_CONNECTED && !mdnsStarted) {
    mdnsStarted = MDNS.begin("hydrodeck");
    if (mdnsStarted) Serial.println("mDNS: hydrodeck.local");
  }

  // Apply replacement Wi-Fi credentials without disabling the setup AP.
  if (provisioningPending) {
    if (!wifiConnectInProgress) {
      WiFi.disconnect(false, false);
      WiFi.begin(ssid, password);
      wifiConnectInProgress = true;
      wifiConnectStarted = now;
      Serial.println("Trying the newly provisioned Wi-Fi network...");
    } else if (WiFi.status() == WL_CONNECTED && WiFi.SSID() == String(ssid)) {
      provisioningPending = false;
      wifiConnectInProgress = false;
      initializeFirebase();
      Serial.println("Replacement Wi-Fi connected; setup hotspot remains available.");
    } else if (now - wifiConnectStarted >= wifiConnectTimeout) {
      wifiConnectInProgress = false;
      provisioningPending = false;
      lastWifiRetry = now;
      Serial.println("New Wi-Fi attempt timed out; setup hotspot remains available.");
    }
  } else if (WiFi.status() == WL_CONNECTED) {
    wifiConnectInProgress = false;
    initializeFirebase();
  } else {
    if (firebaseInitialized) {
      firebaseInitialized = false;
      Serial.println("Wi-Fi lost; Firebase will re-authenticate after reconnect.");
    }
    if (wifiConnectInProgress && now - wifiConnectStarted >= wifiConnectTimeout) {
      wifiConnectInProgress = false;
    }
    if (!wifiConnectInProgress && now - lastWifiRetry >= wifiRetryInterval) {
      lastWifiRetry = now;
      WiFi.begin(ssid, password);
      wifiConnectInProgress = true;
      wifiConnectStarted = now;
      Serial.println("Retrying saved Wi-Fi credentials...");
    }
  }

  if (now - lastRemoteCommandPoll >= remoteCommandInterval) {
    lastRemoteCommandPoll = now;
    requestRemoteActuatorCommand();
  }

  if (now - lastPhLimitPoll >= phLimitPollInterval) {
    lastPhLimitPoll = now;
    requestPhLimit();
  }

  // Sampling TDS Sensor 1 & Sensor 2
  static unsigned long sampleTime = 0;
  if (now - sampleTime > 40U) {
    sampleTime = now;
    analogBuffer1[analogBufferIndex1] = analogRead(TDS_PIN1);
    analogBufferIndex1 = (analogBufferIndex1 + 1) % SCOUNT;

    analogBuffer2[analogBufferIndex2] = analogRead(TDS_PIN2);
    analogBufferIndex2 = (analogBufferIndex2 + 1) % SCOUNT;
  }

  // Calculate TDS Values
  static unsigned long tdsTime = 0;
  if (now - tdsTime > 800U) {
    tdsTime = now;

    // Buffer 1
    for (int i = 0; i < SCOUNT; i++) analogBufferTemp1[i] = analogBuffer1[i];
    float avgV1 = getMedianNum(analogBufferTemp1, SCOUNT) * (float)VREF / 4096.0;
    avgV1 = constrain(avgV1, 0.0f, (float)VREF);
    float compCoeff1 = 1.0 + 0.02 * (currentTemperature1 - 25.0);
    float compV1 = avgV1 / compCoeff1;
    float tdsVal1 = (133.42 * compV1 * compV1 * compV1 - 255.86 * compV1 * compV1 + 857.39 * compV1) * 0.5;
    currentTdsValue1 = max(0.0f, tdsVal1) * tdsCalibrationFactor;

    // Buffer 2
    for (int i = 0; i < SCOUNT; i++) analogBufferTemp2[i] = analogBuffer2[i];
    float avgV2 = getMedianNum(analogBufferTemp2, SCOUNT) * (float)VREF / 4096.0;
    avgV2 = constrain(avgV2, 0.0f, (float)VREF);
    float compCoeff2 = 1.0 + 0.02 * (currentTemperature2 - 25.0);
    float compV2 = avgV2 / compCoeff2;
    float tdsVal2 = (133.42 * compV2 * compV2 * compV2 - 255.86 * compV2 * compV2 + 857.39 * compV2) * 0.5;
    currentTdsValue2 = max(0.0f, tdsVal2) * tdsCalibrationFactor;
  }

  // Reading pH Sensors with 10-second delay logic
  if (now - lastPhCheck >= phInterval) {
    lastPhCheck = now;
    currentPhValue1 = readPhSensor(PH_PIN1, slope1);
    currentPhValue2 = readPhSensor(PH_PIN2, slope2);

    serviceAutomaticPhControl(1, currentPhValue1, now);
    serviceAutomaticPhControl(2, currentPhValue2, now);
  }

  const bool waterFull1 = digitalRead(WATER_SENSOR_PIN1) == HIGH;
  const bool waterFull2 = digitalRead(WATER_SENSOR_PIN2) == HIGH;
  if (waterFull1) {
    waterLowSince1 = 0;
  } else {
    manualWaterShortRuns1 = 0;
    if (waterLowSince1 == 0) waterLowSince1 = now;
    if (now - waterLowSince1 >= waterAutoDelay && !pumpRuns[0].active) {
      startPumpRun(1, 0, true);
    }
  }
  if (waterFull2) {
    waterLowSince2 = 0;
  } else {
    manualWaterShortRuns2 = 0;
    if (waterLowSince2 == 0) waterLowSince2 = now;
    if (now - waterLowSince2 >= waterAutoDelay && !pumpRuns[4].active) {
      startPumpRun(5, 0, true);
    }
  }

  servicePumpRuns();

  // Reading Temperature Sensors
  if (now - lastTempCheck >= tempInterval) {
    lastTempCheck = now;

    float reading1 = sensors1.getTempCByIndex(0);
    if (reading1 != DEVICE_DISCONNECTED_C && reading1 > -50.0) currentTemperature1 = reading1;
    sensors1.requestTemperatures();

    float reading2 = sensors2.getTempCByIndex(0);
    if (reading2 != DEVICE_DISCONNECTED_C && reading2 > -50.0) currentTemperature2 = reading2;
    sensors2.requestTemperatures();
  }

  // Syncing with Firebase Realtime Database
  if (firebaseInitialized && app.ready() && now - lastFirebasePush >= firebaseInterval) {
    lastFirebasePush = now;

    Database.set<float>(aClient, "/hydrodeck/tds1", currentTdsValue1);
    Database.set<float>(aClient, "/hydrodeck/ph1", currentPhValue1);
    Database.set<float>(aClient, "/hydrodeck/temp1", currentTemperature1);
    Database.set<String>(aClient, "/hydrodeck/water1", digitalRead(WATER_SENSOR_PIN1) == HIGH ? "full" : "low");
    Database.set<float>(aClient, "/hydrodeck/tds2", currentTdsValue2);
    Database.set<float>(aClient, "/hydrodeck/ph2", currentPhValue2);
    Database.set<float>(aClient, "/hydrodeck/temp2", currentTemperature2);
    Database.set<String>(aClient, "/hydrodeck/water2", digitalRead(WATER_SENSOR_PIN2) == HIGH ? "full" : "low");
    String actuatorStateJson = "{";
    actuatorStateJson += "\"waterPump1\":" + String(digitalRead(waterPump1Pin) == PUMP_ON ? "true" : "false") + ",";
    actuatorStateJson += "\"phUp1\":" + String(digitalRead(phUp1Pin) == PUMP_ON ? "true" : "false") + ",";
    actuatorStateJson += "\"phDown1\":" + String(digitalRead(phDown1Pin) == PUMP_ON ? "true" : "false") + ",";
    actuatorStateJson += "\"nutrient1\":" + String(digitalRead(nutrientPump1Pin) == PUMP_ON ? "true" : "false") + ",";
    actuatorStateJson += "\"waterPump2\":" + String(digitalRead(waterPump2Pin) == PUMP_ON ? "true" : "false") + ",";
    actuatorStateJson += "\"phUp2\":" + String(digitalRead(phUp2Pin) == PUMP_ON ? "true" : "false") + ",";
    actuatorStateJson += "\"phDown2\":" + String(digitalRead(phDown2Pin) == PUMP_ON ? "true" : "false") + ",";
    actuatorStateJson += "\"nutrient2\":" + String(digitalRead(nutrientPump2Pin) == PUMP_ON ? "true" : "false") + ",";
    actuatorStateJson += "\"growLight\":" + String(digitalRead(growLightPin) == LIGHT_ON ? "true" : "false") + "}";
    Database.set<object_t>(aClient, "/hydrodeck/actuators", object_t(actuatorStateJson.c_str()));
    Database.set<bool>(aClient, "/hydrodeck/wifiConnected", true);
    Database.set<bool>(aClient, "/hydrodeck/firebaseReady", true);
    Database.set<object_t>(aClient, "/hydrodeck/lastSeen", object_t("{\".sv\":\"timestamp\"}"));

    if (aClient.lastError().code() != 0) {
      Serial.print("Firebase update failed, code: ");
      Serial.println(aClient.lastError().code());
    } else {
      Serial.println("Firebase update sent for dual sensor sets and actuators.");
    }
  }
}
