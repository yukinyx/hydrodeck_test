int getMedianNum(int values[], int length) {
  int bTab[length];
  for (byte i = 0; i < length; i++) {
    bTab[i] = values[i];
  }
  int i, j, bTemp;
  for (j = 0; j < length - 1; j++) {
    for (i = 0; i < length - j - 1; i++) {
      if (bTab[i] > bTab[i + 1]) {
        bTemp = bTab[i];
        bTab[i] = bTab[i + 1];
        bTab[i + 1] = bTemp;
      }
    }
  }
  if ((length & 1) > 0) {
    bTemp = bTab[(length - 1) / 2];
  } else {
    bTemp = (bTab[length / 2] + bTab[length / 2 - 1]) / 2;
  }
  return bTemp;
}

float readPhSensor(int pin, float sensorSlope) {
  int measurings = 0;
  int samples = 10;
  for (int i = 0; i < samples; i++) {
    measurings += analogRead(pin);
  }
  float voltage = (v_ref / 4095.0f) * (float)measurings / (float)samples;
  float ph = 7.0 + ((2.5 - voltage) / sensorSlope);
  if (ph < 0.0) ph = 0.0;
  if (ph > 14.0) ph = 14.0;
  return ph;
}

uint16_t packetCrc(const uint8_t* bytes, size_t length) {
  uint16_t crc = 0xffff;
  for (size_t index = 0; index < length; index++) {
    crc ^= bytes[index];
    for (uint8_t bit = 0; bit < 8; bit++) {
      crc = (crc & 1) ? (crc >> 1) ^ 0xa001 : crc >> 1;
    }
  }
  return crc;
}

template <typename Packet>
void preparePacket(Packet& packet, uint8_t type, uint8_t sender) {
  packet.header.magic = ESPNOW_MAGIC;
  packet.header.version = ESPNOW_VERSION;
  packet.header.type = type;
  packet.header.sender = sender;
  packet.header.payloadSize =
      sizeof(Packet) - sizeof(PacketHeader) - sizeof(packet.crc) -
      sizeof(packet.authTag);
  packet.header.sequence = ++espNowSequence;
}

template <typename Packet>
bool finalizePacket(Packet& packet) {
  packet.crc = packetCrc(
      reinterpret_cast<const uint8_t*>(&packet), offsetof(Packet, crc));
  uint8_t digest[32];
  const mbedtls_md_info_t* md = mbedtls_md_info_from_type(MBEDTLS_MD_SHA256);
  if (md == nullptr || mbedtls_md_hmac(
      md,
      reinterpret_cast<const unsigned char*>(RAIN_TOKEN), strlen(RAIN_TOKEN),
      reinterpret_cast<const unsigned char*>(&packet), offsetof(Packet, authTag),
      digest) != 0) {
    return false;
  }
  memcpy(packet.authTag, digest, sizeof(packet.authTag));
  return true;
}

template <typename Packet>
bool packetValid(const Packet& packet, size_t length, uint8_t type, uint8_t sender) {
  if (length != sizeof(Packet) || packet.header.magic != ESPNOW_MAGIC ||
      packet.header.version != ESPNOW_VERSION || packet.header.type != type ||
      packet.header.sender != sender ||
      packet.header.payloadSize !=
          sizeof(Packet) - sizeof(PacketHeader) - sizeof(packet.crc) -
          sizeof(packet.authTag)) {
    return false;
  }
  if (packet.crc != packetCrc(
      reinterpret_cast<const uint8_t*>(&packet), offsetof(Packet, crc))) {
    return false;
  }
  uint8_t digest[32];
  const mbedtls_md_info_t* md = mbedtls_md_info_from_type(MBEDTLS_MD_SHA256);
  if (md == nullptr || mbedtls_md_hmac(
      md,
      reinterpret_cast<const unsigned char*>(RAIN_TOKEN), strlen(RAIN_TOKEN),
      reinterpret_cast<const unsigned char*>(&packet), offsetof(Packet, authTag),
      digest) != 0) {
    return false;
  }
  return memcmp(packet.authTag, digest, sizeof(packet.authTag)) == 0;
}

bool sequenceIsNewer(uint16_t sequence, uint16_t previous) {
  return static_cast<int16_t>(sequence - previous) > 0;
}

void derivePeerKey(const char* purpose, uint8_t output[16]) {
  String input = String(purpose) + ":" + RAIN_TOKEN;
  uint8_t digest[32];
  mbedtls_sha256(
      reinterpret_cast<const unsigned char*>(input.c_str()),
      input.length(),
      digest,
      0);
  memcpy(output, digest, 16);
}

bool addEspNowPeer(const uint8_t mac[6], bool encrypted) {
  esp_now_peer_info_t peer = {};
  memcpy(peer.peer_addr, mac, 6);
  peer.channel = 0;
  peer.ifidx = WIFI_IF_STA;
  peer.encrypt = encrypted;
  if (encrypted) derivePeerKey("HYDRODECK-RAIN-LMK-v1", peer.lmk);
  const esp_err_t result = esp_now_add_peer(&peer);
  if (result == ESP_OK || result == ESP_ERR_ESPNOW_EXIST) return true;
  Serial.printf("ESP-NOW peer add failed mac=%02X:%02X:%02X:%02X:%02X:%02X err=%d channel=%d\n",
      mac[0], mac[1], mac[2], mac[3], mac[4], mac[5], result, WiFi.channel());
  return false;
}

bool sendEspNowPacket(const uint8_t mac[6], const void* packet, size_t length) {
  if (!espNowReady) return false;
  const esp_err_t result = esp_now_send(
      mac, reinterpret_cast<const uint8_t*>(packet), length);
  if (result != ESP_OK) {
    Serial.printf("ESP-NOW send enqueue failed err=%d channel=%d\n",
        result, WiFi.channel());
    return false;
  }
  return true;
}

void onEspNowReceive(
    const esp_now_recv_info_t* info,
    const uint8_t* data,
    int length) {
  if (info == nullptr || info->src_addr == nullptr || data == nullptr ||
      length <= 0 || length > static_cast<int>(ESPNOW_PACKET_MAX) ||
      espNowReceiveQueue == nullptr) return;
  ReceivedPacket packet = {};
  memcpy(packet.mac, info->src_addr, sizeof(packet.mac));
  packet.length = static_cast<uint8_t>(length);
  memcpy(packet.bytes, data, length);
  if (xQueueSend(espNowReceiveQueue, &packet, 0) != pdTRUE) {
    espNowQueueDrops.fetch_add(1, std::memory_order_relaxed);
  }
}

  if (pumpNum == 5) {

    digitalWrite(pin, turnOn ? PUMP_ON : PUMP_OFF);
  }
}

// Parse PH_ADJUST:<bed>:<direction> command format
void parsePhAdjustCommand(const String& command) {
  // Expected format: "PH_ADJUST:<bed>:<direction>"
  // Example: "PH_ADJUST:1:UP" or "PH_ADJUST:2:DOWN"
  if (!command.startsWith("PH_ADJUST:")) return;

  String payload = command.substring(10); // Remove "PH_ADJUST:" prefix
  int firstColon = payload.indexOf(':');
  if (firstColon == -1) return;

  String bedStr = payload.substring(0, firstColon);
  String direction = payload.substring(firstColon + 1);

  int bed = bedStr.toInt();
  if (bed < 1 || bed > 2) return;

  String actuatorName;
  if (direction == "UP") {
    actuatorName = (bed == 1) ? "phUp1" : "phUp2";
  } else if (direction == "DOWN") {
    actuatorName = (bed == 1) ? "phDown1" : "phDown2";
  } else {
    return; // Invalid direction
  }

  // Check if pH is actually in range that needs adjustment
  bool shouldActivate = false;
  if (bed == 1) {
    if (direction == "UP" && currentPhValue1 < minPhThreshold) shouldActivate = true;
    if (direction == "DOWN" && currentPhValue1 > maxPhThreshold) shouldActivate = true;
  } else { // bed == 2
    if (direction == "UP" && currentPhValue2 < minPhThreshold) shouldActivate = true;
    if (direction == "DOWN" && currentPhValue2 > maxPhThreshold) shouldActivate = true;
  }

  if (!shouldActivate) return;

  // Set durations based on bed
  unsigned long durationMs = bed == 1 ? BED1_PUMP_DURATION_MS : BED2_PUMP_DURATION_MS;

  // Find pump number
  int pumpNum = pumpNumberForActuator(actuatorName);
  if (pumpNum == 0) return;

  // Check if pump is already active
  if (pumpRuns[pumpNum - 1].active) return;

  // Check for conflicting pH pumps (can't have both up and down active at same time)
  if ((pumpNum == 2 || pumpNum == 3) && (pumpRuns[1].active || pumpRuns[2].active)) return;
  if ((pumpNum == 6 || pumpNum == 7) && (pumpRuns[5].active || pumpRuns[6].active)) return;

  // Start the pump run
  PumpRunState& run = pumpRuns[pumpNum - 1];
  run.active = true;
  run.hasDeadline = true;
  run.stopAt = millis() + durationMs;
  setPump(pumpNum, true);
  publishActuatorState(actuatorName, true);
}

String consumedCommandStatus(const String& requestId) {
  const String entries = commandPreferences.getString("recentIds", "");
  int start = 0;
  while (start < entries.length()) {
    const int separator = entries.indexOf(',', start);
    const int end = separator < 0 ? entries.length() : separator;
    const String entry = entries.substring(start, end);
    const int statusSeparator = entry.indexOf('=');
    const String storedId = statusSeparator < 0 ? entry : entry.substring(0, statusSeparator);
    if (storedId == requestId) {
      return statusSeparator < 0 ? "completed" : entry.substring(statusSeparator + 1);
    }
    if (separator < 0) break;
    start = separator + 1;
  }
  return "";
}

bool wasCommandConsumed(const String& requestId) {
  return consumedCommandStatus(requestId).length() > 0;
}

      (requestId.startsWith("-") && requestId.length() >= 16);
}

void rememberConsumedCommand(const String& requestId, const String& status = "accepted") {
  if (requestId.length() == 0) return;
  String entries = commandPreferences.getString("recentIds", "");
  String retained;
  int start = 0;
  while (start < entries.length()) {
    const int separator = entries.indexOf(',', start);
    const int end = separator < 0 ? entries.length() : separator;
    const String entry = entries.substring(start, end);
    const int statusSeparator = entry.indexOf('=');
    const String storedId = statusSeparator < 0 ? entry : entry.substring(0, statusSeparator);
    if (storedId != requestId && entry.length() > 0) {
      if (retained.length() > 0) retained += ",";
      retained += entry;
    }
    if (separator < 0) break;
    start = separator + 1;
  }
  if (retained.length() > 0) retained += ",";
  retained += requestId + "=" + status;
  while (retained.length() > 700) {
    const int separator = retained.indexOf(',');
    if (separator < 0) break;
    retained.remove(0, separator + 1);
  }
  commandPreferences.putString("recentIds", retained);
}

void publishActuatorState(const String& actuator, bool state) {
  if (!firebaseInitialized || !app.ready()) return;
  Database.set<bool>(aClient, "/hydrodeck/actuators/" + actuator, state);
}

void publishCommandAck(const String& requestId, const String& status) {
  rememberConsumedCommand(requestId, status);
  if (requestId.length() == 0 || !firebaseInitialized || !app.ready()) return;
  Database.set<String>(aClient, "/hydrodeck/commandAcks/" + requestId, status);
}

void publishCommandRejectionReason(const String& requestId, const String& reason) {
  if (requestId.length() == 0 || !firebaseInitialized || !app.ready()) return;
  Database.set<String>(
    aClient,
    "/hydrodeck/commandAckReasons/" + requestId,
    reason
  );
}

String pumpStartRejectionReason(int pumpNum) {
  if (pumpNum < 1 || pumpNum > 8) return "invalid_pump";
  if (isRemoteWaterPump(pumpNum) && !rainNodeOnline) return "rain_node_offline";
  if (pumpRuns[pumpNum - 1].active) return "pump_already_active";
  if ((pumpNum == 1 && pumpRuns[4].active) || (pumpNum == 5 && pumpRuns[0].active)) {
    return "other_bed_water_pump_active";
  }
  if ((pumpNum == 2 && pumpRuns[2].active) || (pumpNum == 3 && pumpRuns[1].active) ||
      (pumpNum == 6 && pumpRuns[6].active) || (pumpNum == 7 && pumpRuns[5].active)) {
    return "opposite_ph_pump_active";
  }
  const bool chemicalPump = pumpNum == 2 || pumpNum == 3 || pumpNum == 4 ||
      pumpNum == 6 || pumpNum == 7 || pumpNum == 8;
  if (chemicalPump) {
    for (int index = 0; index < 8; index++) {
      const int otherPump = index + 1;
      const bool otherIsChemical = otherPump == 2 || otherPump == 3 || otherPump == 4 ||
          otherPump == 6 || otherPump == 7 || otherPump == 8;
      if (otherPump != pumpNum && otherIsChemical && pumpRuns[index].active) {
        return "chemical_mutex_busy_by_" + String(remoteActuatorNames[index]);
      }
    }
  }
  return "";
}

bool startPumpRun(
    int pumpNum,
    unsigned long durationMs,
    const String& requestId = "",
    String* rejectionReason = nullptr) {
  const String reason = pumpStartRejectionReason(pumpNum);
  if (rejectionReason != nullptr) *rejectionReason = reason;
  if (reason.length() > 0) return false;
  PumpRunState& run = pumpRuns[pumpNum - 1];
  run.active = true;
  run.hasDeadline = durationMs > 0;
  run.awaitingStartConfirmation = isRemoteWaterPump(pumpNum);
  run.stopRequested = false;
  run.durationMs = durationMs;
  run.confirmationStartedAt = millis();
  run.stopAt = durationMs == 0 ? 0 : millis() + durationMs;
  activeCommandIds[pumpNum - 1] = requestId;
  setPump(pumpNum, true);
  if (isRemoteWaterPump(pumpNum)) {
    Serial.printf(
      "Remote pump start requested pump=%s requestId=%s durationMs=%lu\n",
      remoteActuatorNames[pumpNum - 1],
      requestId.c_str(),
      durationMs
    );
  } else {
    publishActuatorState(remoteActuatorNames[pumpNum - 1], true);
    publishCommandAck(requestId, "running");
  }
  return true;
}

void serviceAutomaticPhControl(int bed, float phValue, unsigned long now) {
  const int upPump = bed == 1 ? 2 : 6;
  const int downPump = bed == 1 ? 3 : 7;
  const unsigned long pulseDuration =
      bed == 1 ? BED1_PUMP_DURATION_MS : BED2_PUMP_DURATION_MS;
  unsigned long& lowSince = bed == 1 ? bed1PhLowSince : bed2PhLowSince;
  unsigned long& highSince = bed == 1 ? bed1PhHighSince : bed2PhHighSince;
  unsigned long& lastAttemptLogAt =
      bed == 1 ? bed1PhLastAttemptLogAt : bed2PhLastAttemptLogAt;

  if (phValue < minPhThreshold) {
    highSince = 0;
    if (lowSince == 0) lowSince = now;
    const unsigned long secondsBelowMin = (now - lowSince) / 1000UL;
    if (now - lowSince >= phDelayThreshold) {
      String rejectionReason;
      const bool started = startPumpRun(
        upPump,
        pulseDuration,
        "",
        &rejectionReason
      );
      if (started || lastAttemptLogAt == 0 ||
          now - lastAttemptLogAt >= phAutoRetryLogInterval) {
        Serial.printf(
          "Auto pH-up bed=%d pH=%.2f min=%.2f max=%.2f secondsBelowMin=%lu result=%s reason=%s\n",
          bed,
          phValue,
          minPhThreshold,
          maxPhThreshold,
          secondsBelowMin,
          started ? "accepted" : "rejected",
          started ? "none" : rejectionReason.c_str()
        );
        lastAttemptLogAt = now;
      }
      if (started) lowSince = now;
    }
  } else if (phValue > maxPhThreshold) {
    lowSince = 0;
    if (highSince == 0) highSince = now;
    const unsigned long secondsAboveMax = (now - highSince) / 1000UL;
    if (now - highSince >= phDelayThreshold) {
      String rejectionReason;
      const bool started = startPumpRun(
        downPump,
        pulseDuration,
        "",
        &rejectionReason
      );
      if (started || lastAttemptLogAt == 0 ||
          now - lastAttemptLogAt >= phAutoRetryLogInterval) {
        Serial.printf(
          "Auto pH-down bed=%d pH=%.2f min=%.2f max=%.2f secondsAboveMax=%lu result=%s reason=%s\n",
          bed,
          phValue,
          minPhThreshold,
          maxPhThreshold,
          secondsAboveMax,
          started ? "accepted" : "rejected",
          started ? "none" : rejectionReason.c_str()
        );
        lastAttemptLogAt = now;
      }
      if (started) highSince = now;
    }
  } else {
    if (phValue >= minPhThreshold + phControlHysteresis) lowSince = 0;
    if (phValue <= maxPhThreshold - phControlHysteresis) highSince = 0;
  }
}

void finishRemotePumpStop(int pumpNum) {
  if (pumpNum != 1 && pumpNum != 5) return;
  PumpRunState& run = pumpRuns[pumpNum - 1];
  const int index = pumpNum == 1 ? 0 : 1;
  const String requestId = activeCommandIds[pumpNum - 1];
  const String stopRequestId = pendingRemoteStopCommandIds[index];
  const String actuator = remoteActuatorNames[pumpNum - 1];
  setPump(pumpNum, false);
  if (index == 0) {
    rainPump1On = false;
    rainPumpStateKnown1 = true;
  } else {
    rainPump2On = false;
    rainPumpStateKnown2 = true;
  }
  activeCommandIds[pumpNum - 1] = "";
  run = PumpRunState();
  publishActuatorState(actuator, false);
  publishCommandAck(requestId, "completed");
  if (stopRequestId != requestId) publishCommandAck(stopRequestId, "completed");
  pendingRemoteStopCommandIds[index] = "";
  pendingRemoteStopStartedAt[index] = 0;
  Serial.printf(
    "Remote pump stopped pump=%s requestId=%s confirmed=1\n",
    actuator.c_str(),
    requestId.c_str()
  );
}

void rejectUnconfirmedRemotePumpStart(int pumpNum) {
  if (pumpNum != 1 && pumpNum != 5) return;
  PumpRunState& run = pumpRuns[pumpNum - 1];
  const String requestId = activeCommandIds[pumpNum - 1];
  const String actuator = remoteActuatorNames[pumpNum - 1];
  setPump(pumpNum, false);
  activeCommandIds[pumpNum - 1] = "";
  run = PumpRunState();
  publishCommandRejectionReason(requestId, "rain_node_no_confirm");
  publishCommandAck(requestId, "rejected");
  Serial.printf(
    "Remote pump rejected pump=%s requestId=%s reason=rain_node_no_confirm\n",
    actuator.c_str(),
    requestId.c_str()
  );
}

void finishRemotePumpFault(int pumpNum, const String& reason) {
  if (pumpNum != 1 && pumpNum != 5) return;
  const int index = pumpNum == 1 ? 0 : 1;
  PumpRunState& run = pumpRuns[pumpNum - 1];
  if (!run.active && pendingRemoteStopCommandIds[index].isEmpty()) return;
  const String requestId = activeCommandIds[pumpNum - 1];
  const String stopRequestId = pendingRemoteStopCommandIds[index];
  const String actuator = remoteActuatorNames[pumpNum - 1];
  const bool rejectedBeforeStart = run.active && run.awaitingStartConfirmation;
  setPump(pumpNum, false);
  activeCommandIds[pumpNum - 1] = "";
  run = PumpRunState();
  publishActuatorState(actuator, false);
  publishCommandRejectionReason(requestId, reason);
  publishCommandAck(requestId, rejectedBeforeStart ? "rejected" : "timed_out");
  if (stopRequestId != requestId) {
    publishCommandRejectionReason(stopRequestId, reason);
    publishCommandAck(stopRequestId, "timed_out");
  }
  pendingRemoteStopCommandIds[index] = "";
  pendingRemoteStopStartedAt[index] = 0;
  Serial.printf(
    "Remote pump %s pump=%s requestId=%s reason=%s\n",
    rejectedBeforeStart ? "rejected" : "stopped",
    actuator.c_str(),
    requestId.c_str(),
    reason.c_str()
  );
}

void requestRemotePumpStop(int pumpNum, const String& requestId) {
  const int index = pumpNum == 1 ? 0 : 1;
  PumpRunState& run = pumpRuns[pumpNum - 1];
  pendingRemoteStopCommandIds[index] = requestId;
  pendingRemoteStopStartedAt[index] = millis();
  const bool wasDesired = index == 0 ? rainDesiredPump1On : rainDesiredPump2On;
  setPump(pumpNum, false);
  if (!wasDesired) {
    remoteCommandCounter++;
    if (remoteCommandCounter == 0) remoteCommandCounter++;
    remotePumpCommandIds[index] = remoteCommandCounter;
    rainAcknowledgedCommandId[index] = 0;
    rainAcknowledgedResult[index] = 0;
    lastRainCommandSentAt[index] = 0;
  }
  if (run.active) run.stopRequested = true;
  Serial.printf(
    "Remote pump stop requested pump=%s requestId=%s command=%lu\n",
    remoteActuatorNames[pumpNum - 1],
    requestId.c_str(),
    static_cast<unsigned long>(remotePumpCommandIds[index])
  );
}

void stopPumpRun(int pumpNum, const String& requestId = "") {
  if (pumpNum < 1 || pumpNum > 8) return;
  if (isRemoteWaterPump(pumpNum)) {
    requestRemotePumpStop(pumpNum, requestId);
    return;
  }
  PumpRunState& run = pumpRuns[pumpNum - 1];
  if (!run.active) return;
  setPump(pumpNum, false);
  const String activeRequestId = activeCommandIds[pumpNum - 1];
  activeCommandIds[pumpNum - 1] = "";
  run.active = false;
  run.hasDeadline = false;
  run.awaitingStartConfirmation = false;
  run.stopRequested = false;
  run.durationMs = 0;
  run.stopAt = 0;
  publishActuatorState(remoteActuatorNames[pumpNum - 1], false);
  publishCommandAck(activeRequestId, "completed");
}
bool executePumpRequest(const String& actuator, bool requestedOn, const String& requestId = "") {
  const int pumpNum = pumpNumberForActuator(actuator);
  if (pumpNum == 0) return false;
  if (!requestedOn) {
    stopPumpRun(pumpNum, requestId);
    return true;
  }

  unsigned long durationMs = 0;
  if (actuator == "phUp1" || actuator == "phDown1") {
    if ((actuator == "phUp1" && currentPhValue1 > maxPhThreshold) ||
      (actuator == "phDown1" && currentPhValue1 < minPhThreshold)) return false;
    durationMs = BED1_PUMP_DURATION_MS;
  } else if (actuator == "phUp2" || actuator == "phDown2") {
    if ((actuator == "phUp2" && currentPhValue2 > maxPhThreshold) ||
      (actuator == "phDown2" && currentPhValue2 < minPhThreshold)) return false;
    durationMs = BED2_PUMP_DURATION_MS;
  } else if (actuator == "nutrient1") {
    durationMs = BED1_PUMP_DURATION_MS;
  } else if (actuator == "nutrient2") {
    durationMs = BED2_PUMP_DURATION_MS;
  } else if (isRemoteWaterPump(pumpNum)) {
    durationMs = 0;
  }
  return startPumpRun(pumpNum, durationMs, requestId);
}

void publishReportedRemotePumpState(int pumpNum, bool pumpIsOn, bool wasKnown) {
  PumpRunState& run = pumpRuns[pumpNum - 1];
  const String actuator = remoteActuatorNames[pumpNum - 1];
  const bool changed = pumpNum == 1
      ? (!wasKnown || rainPump1On != pumpIsOn)
      : (!wasKnown || rainPump2On != pumpIsOn);
  if (pumpNum == 1) {
    rainPump1On = pumpIsOn;
    rainPumpStateKnown1 = true;
  } else {
    rainPump2On = pumpIsOn;
    rainPumpStateKnown2 = true;
  }

  if (run.active && run.awaitingStartConfirmation && pumpIsOn) {
    run.awaitingStartConfirmation = false;
    if (run.hasDeadline) run.stopAt = millis() + run.durationMs;
    publishCommandAck(activeCommandIds[pumpNum - 1], "running");
    publishActuatorState(actuator, true);
    Serial.printf(
      "Remote pump confirmed pump=%s requestId=%s\n",
      actuator.c_str(),
      activeCommandIds[pumpNum - 1].c_str()
    );
    return;
  }
  if (!pumpIsOn && (run.stopRequested ||
      !pendingRemoteStopCommandIds[pumpNum == 1 ? 0 : 1].isEmpty())) {
    finishRemotePumpStop(pumpNum);
    return;
  }
  if (changed) publishActuatorState(actuator, pumpIsOn);
}

void serviceRemotePumpFeedback(unsigned long now) {
  for (const int pumpNum : {1, 5}) {
    PumpRunState& run = pumpRuns[pumpNum - 1];
    if (run.active && run.awaitingStartConfirmation &&
        now - run.confirmationStartedAt >= RAIN_START_CONFIRM_TIMEOUT_MS) {
      rejectUnconfirmedRemotePumpStart(pumpNum);
    }
  }
}

String reservoirLevelFromSensors(bool fullWet) {
  return fullWet ? "full" : "not_full";
}

void serviceRainNodeHealth(unsigned long now) {
  if (!rainNodeOnline) {
    for (const int pumpNum : {1, 5}) {
      const int index = pumpNum == 1 ? 0 : 1;
      if (!pendingRemoteStopCommandIds[index].isEmpty() &&
          now - pendingRemoteStopStartedAt[index] >= RAIN_NODE_TIMEOUT_MS) {
        finishRemotePumpFault(pumpNum, "rain_node_offline");
      }
    }
    return;
  }
  if (now - lastRainSeenAt <= RAIN_NODE_TIMEOUT_MS) return;
  rainNodeOnline = false;
  Serial.println("RAIN node offline");
  for (const int pumpNum : {1, 5}) {
    PumpRunState& run = pumpRuns[pumpNum - 1];
    const bool wasOn = pumpNum == 1 ? rainPump1On : rainPump2On;
    if (pumpNum == 1) rainPump1On = false;
    else rainPump2On = false;
    if (wasOn || (run.active && !run.awaitingStartConfirmation)) {
      publishActuatorState(remoteActuatorNames[pumpNum - 1], false);
    }
    if (!run.active) {
      if (!pendingRemoteStopCommandIds[pumpNum == 1 ? 0 : 1].isEmpty()) {
        finishRemotePumpFault(pumpNum, "rain_node_offline");
      }
      continue;
    }
    if (run.awaitingStartConfirmation) {
      rejectUnconfirmedRemotePumpStart(pumpNum);
    } else {
      finishRemotePumpFault(pumpNum, "rain_node_offline");
    }
  }
}

const char* rainFaultName(uint8_t fault) {
  switch (fault) {
    case FAULT_SENSOR_CONFLICT: return "sensor_conflict";
    case FAULT_PUMP_CONFLICT: return "pump_conflict";
    case FAULT_RESERVOIR_LOW: return "reservoir_low";
    case FAULT_MAX_RUN: return "max_run";
    case FAULT_STALE_COMMAND: return "stale_command";
    case FAULT_INVALID_COMMAND: return "invalid_command";
    default: return "";
  }
}

void sendMainBeacon() {
  if (!espNowReady) return;
  MainBeaconPacket packet = {};
  preparePacket(packet, MSG_MAIN_BEACON, DEVICE_MAIN);
  packet.bootId = mainBootId;
  const int currentChannel = WiFi.channel();
  packet.currentChannel = static_cast<uint8_t>(currentChannel > 0 ? currentChannel : 0);
  if (!finalizePacket(packet)) {
    Serial.println("ESP-NOW beacon authentication failed");
    return;
  }
  sendEspNowPacket(ESPNOW_BROADCAST_MAC, &packet, sizeof(packet));
}

unsigned long remotePumpRemainingMs(int pumpNum, unsigned long now) {
  const PumpRunState& run = pumpRuns[pumpNum - 1];
  if (isRemoteWaterPump(pumpNum) || !run.active || !run.hasDeadline ||
      run.stopRequested) return 0;
  if ((long)(now - run.stopAt) >= 0) return 1;
  return min(run.stopAt - now, RAIN_MAX_RUN_MS);
}

void sendRemotePumpCommand(int pumpNum, unsigned long now) {
  if (!espNowReady || !rainMacKnown || pumpNum != 1 && pumpNum != 5) return;
  const int index = pumpNum == 1 ? 0 : 1;
    if (rainAcknowledgedCommandId[index] == remotePumpCommandIds[index] &&
      rainAcknowledgedResult[index] >= ACK_ACCEPTED &&
      rainAcknowledgedResult[index] <= ACK_DUPLICATE) return;
  if (lastRainCommandSentAt[index] != 0 &&
      now - lastRainCommandSentAt[index] < RAIN_COMMAND_RETRY_MS) return;
  lastRainCommandSentAt[index] = now;
  PumpCommandPacket packet = {};
  preparePacket(packet, MSG_PUMP_COMMAND, DEVICE_MAIN);
  packet.mainBootId = mainBootId;
  packet.commandId = remotePumpCommandIds[index];
  packet.remainingMs = remotePumpRemainingMs(pumpNum, now);
  packet.pump = index == 0 ? PUMP1_ID : PUMP2_ID;
  packet.requestedOn = index == 0
      ? (rainDesiredPump1On ? 1 : 0)
      : (rainDesiredPump2On ? 1 : 0);
  if (!finalizePacket(packet)) {
    Serial.printf("ESP-NOW command authentication failed pump=%u\n",
        packet.pump);
    return;
  }
  if (!sendEspNowPacket(rainMac, &packet, sizeof(packet)) &&
      (!rainNodeOnline || now - lastRainCommandSentAt[index] >= 2000UL)) {
    Serial.printf("ESP-NOW command retry pending pump=%u command=%lu\n",
        packet.pump, static_cast<unsigned long>(packet.commandId));
  }
}

void adoptRainPeer(const uint8_t mac[6]) {
  if (rainMacKnown && memcmp(rainMac, mac, sizeof(rainMac)) == 0) return;
  if (rainNodeOnline && rainMacKnown) {
    Serial.println("ESP-NOW packet from an unexpected RAIN MAC rejected");
    return;
  }
  if (rainMacKnown) esp_now_del_peer(rainMac);
  rainMacKnown = false;
  memcpy(rainMac, mac, sizeof(rainMac));
  if (!addEspNowPeer(rainMac, true)) return;
  rainMacKnown = true;
  Serial.printf("ESP-NOW RAIN peer paired mac=%02X:%02X:%02X:%02X:%02X:%02X channel=%d\n",
      rainMac[0], rainMac[1], rainMac[2], rainMac[3], rainMac[4], rainMac[5],
      WiFi.channel());
}

void processRainStatus(const ReceivedPacket& received,
    const RainStatusPacket& report, unsigned long now) {
  const bool pumpMaskValid = report.pumpMask <= 3 && report.pumpMask != 3;
  const bool activePumpValid =
      (report.pumpMask == 0 && report.activePump == 0) ||
      (report.pumpMask == 1 && report.activePump == PUMP1_ID) ||
      (report.pumpMask == 2 && report.activePump == PUMP2_ID);
  if (report.bootId == 0 || report.fullWet > 1 ||
      !pumpMaskValid || !activePumpValid || report.fault > FAULT_INVALID_COMMAND ||
      report.currentChannel < 1 || report.currentChannel > 13 ||
      (report.pumpMask == 0 && report.activeCommandId != 0) ||
      (report.pumpMask != 0 && report.activeCommandId == 0)) {
    Serial.println("ESP-NOW RAIN packet rejected: invalid field values");
    return;
  }
  if (rainMacKnown && memcmp(rainMac, received.mac, sizeof(rainMac)) != 0 &&
      rainNodeOnline) {
    Serial.println("ESP-NOW status from unexpected RAIN peer rejected");
    return;
  }
  adoptRainPeer(received.mac);
  if (!rainMacKnown) return;

  const bool rainBootChanged =
      lastRainBootId != 0 && lastRainBootId != report.bootId;
  if (rainBootChanged) {
    Serial.printf("RAIN boot changed boot=%08lX; clearing old pump requests\n",
        static_cast<unsigned long>(report.bootId));
    for (const int pumpNum : {1, 5}) {
      PumpRunState& run = pumpRuns[pumpNum - 1];
      if (!run.active) {
        setPump(pumpNum, false);
      } else if (run.awaitingStartConfirmation) {
        rejectUnconfirmedRemotePumpStart(pumpNum);
      } else {
        finishRemotePumpFault(pumpNum, "rain_node_rebooted");
      }
    }
    lastRainSequenceKnown = false;
  }
  if (lastRainBootId == report.bootId && lastRainSequenceKnown &&
      !sequenceIsNewer(report.header.sequence, lastRainSequence)) {
    Serial.printf("ESP-NOW stale RAIN packet rejected seq=%u\n",
        report.header.sequence);
    return;
  }

  const bool wasOnline = rainNodeOnline;
  const String oldLevel = reservoirLevel;
  const bool previousFull = reservoirFullWet;
  const bool previousPump1Known = rainPumpStateKnown1;
  const bool previousPump2Known = rainPumpStateKnown2;
  if (!wasOnline || rainBootChanged) queueFreshRainSafetyStops();
  lastRainBootId = report.bootId;
  lastRainSequence = report.header.sequence;
  lastRainSequenceKnown = true;
  lastRainSeenAt = now;
  rainNodeOnline = true;
  reservoirFullWet = report.fullWet != 0;
  reservoirLevel = reservoirLevelFromSensors(reservoirFullWet);
  rainFaultCode = rainFaultName(report.fault);
  if (!wasOnline) {
    Serial.printf("RAIN ESP-NOW online channel=%u\n", report.currentChannel);
  }
  if (lastLoggedRainChannel != report.currentChannel) {
    lastLoggedRainChannel = report.currentChannel;
    Serial.printf("ESP-NOW channel active=%u\n", lastLoggedRainChannel);
  }
  if (oldLevel != reservoirLevel || previousFull != reservoirFullWet) {
    Serial.printf("Reservoir changed level=%s full=%u\n",
        reservoirLevel.c_str(), reservoirFullWet ? 1 : 0);
  }
  if (report.fault != FAULT_NONE) {
    Serial.printf("RAIN fault=%s\n", rainFaultName(report.fault));
  }
  if ((report.fault == FAULT_MAX_RUN ||
       report.fault == FAULT_RESERVOIR_LOW) && (report.pumpMask & 1) == 0) {
    finishRemotePumpFault(1, rainFaultName(report.fault));
  }
  if ((report.fault == FAULT_MAX_RUN ||
       report.fault == FAULT_RESERVOIR_LOW) && (report.pumpMask & 2) == 0) {
    finishRemotePumpFault(5, rainFaultName(report.fault));
  }
  publishReportedRemotePumpState(1, (report.pumpMask & 1) != 0,
      previousPump1Known);
  publishReportedRemotePumpState(5, (report.pumpMask & 2) != 0,
      previousPump2Known);
}

void processRainAck(const ReceivedPacket& received,
    const CommandAckPacket& ack) {
  if (!rainMacKnown || memcmp(rainMac, received.mac, sizeof(rainMac)) != 0 ||
      ack.rainBootId == 0 || ack.rainBootId != lastRainBootId ||
      ack.commandId == 0 || ack.result < ACK_ACCEPTED ||
      ack.result > ACK_DUPLICATE || ack.reason > FAULT_INVALID_COMMAND ||
      (ack.result != ACK_REJECTED && ack.reason != FAULT_NONE) ||
      (ack.result == ACK_REJECTED && ack.reason == FAULT_NONE)) {
    Serial.println("ESP-NOW ACK rejected: invalid peer, session, or fields");
    return;
  }
  const int index = ack.pump == PUMP1_ID ? 0 :
      (ack.pump == PUMP2_ID ? 1 : -1);
  if (index < 0 || ack.commandId != remotePumpCommandIds[index]) return;
  rainAcknowledgedCommandId[index] = ack.commandId;
  rainAcknowledgedResult[index] = ack.result;
  Serial.printf("ESP-NOW command ACK pump=%u command=%lu result=%u reason=%s\n",
      ack.pump, static_cast<unsigned long>(ack.commandId), ack.result,
      rainFaultName(ack.reason));
  if (ack.result == ACK_REJECTED) {
    const String reason = rainFaultName(ack.reason);
    if (reason.length() > 0) {
      finishRemotePumpFault(index == 0 ? 1 : 5, reason);
    }
  } else if (ack.result == ACK_ACCEPTED || ack.result == ACK_DUPLICATE) {
    lastRainCommandSentAt[index] = 0;
    const int pumpNum = index == 0 ? 1 : 5;
    PumpRunState& run = pumpRuns[pumpNum - 1];
    const bool requestedOn = index == 0
        ? rainDesiredPump1On : rainDesiredPump2On;
    if (requestedOn && run.active && run.awaitingStartConfirmation) {
      run.awaitingStartConfirmation = false;
      if (index == 0) {
        rainPump1On = true;
        rainPumpStateKnown1 = true;
      } else {
        rainPump2On = true;
        rainPumpStateKnown2 = true;
      }
      publishActuatorState(remoteActuatorNames[pumpNum - 1], true);
      publishCommandAck(activeCommandIds[pumpNum - 1], "running");
    } else if (!requestedOn &&
        (!pendingRemoteStopCommandIds[index].isEmpty() || run.stopRequested)) {
      finishRemotePumpStop(pumpNum);
    }
  }
}

void processEspNowPackets() {
  if (espNowReceiveQueue == nullptr) return;
  const uint32_t queueDrops = espNowQueueDrops.load(std::memory_order_relaxed);
  if (queueDrops != reportedEspNowQueueDrops) {
    reportedEspNowQueueDrops = queueDrops;
    Serial.printf("ESP-NOW receive queue overflow drops=%lu\n",
        static_cast<unsigned long>(queueDrops));
  }
  ReceivedPacket received;
  while (xQueueReceive(espNowReceiveQueue, &received, 0) == pdTRUE) {
    if (received.length < sizeof(PacketHeader) + sizeof(uint16_t)) continue;
    PacketHeader header;
    memcpy(&header, received.bytes, sizeof(header));
    if (header.magic != ESPNOW_MAGIC || header.version != ESPNOW_VERSION ||
        header.sender != DEVICE_RAIN) {
      Serial.println("ESP-NOW packet rejected: invalid header/version/sender");
      continue;
    }
    if (header.type == MSG_RAIN_HELLO || header.type == MSG_RAIN_STATUS) {
      RainStatusPacket report;
      memcpy(&report, received.bytes, sizeof(report));
      if (!packetValid(report, received.length, header.type, DEVICE_RAIN)) {
        Serial.println("ESP-NOW RAIN packet rejected: size/CRC");
        continue;
      }
      processRainStatus(received, report, millis());
    } else if (header.type == MSG_COMMAND_ACK) {
      CommandAckPacket ack;
      memcpy(&ack, received.bytes, sizeof(ack));
      if (!packetValid(ack, received.length, MSG_COMMAND_ACK, DEVICE_RAIN)) {
        Serial.println("ESP-NOW ACK rejected: size/CRC");
        continue;
      }
      processRainAck(received, ack);
    } else {
      Serial.printf("ESP-NOW unsupported RAIN packet type=%u\n", header.type);
    }
  }
}

bool initializeEspNow() {
  if (espNowReady) return true;
  if (espNowReceiveQueue == nullptr) {
    espNowReceiveQueue = xQueueCreate(8, sizeof(ReceivedPacket));
  }
  if (espNowReceiveQueue == nullptr) {
    Serial.println("ESP-NOW unavailable: receive queue is not allocated");
    return false;
  }
  const esp_err_t initResult = esp_now_init();
  if (initResult != ESP_OK) {
    Serial.printf("ESP-NOW init failed err=%d channel=%d; retrying\n",
        initResult, WiFi.channel());
    return false;
  }
  uint8_t pmk[16];
  derivePeerKey("HYDRODECK-RAIN-PMK-v1", pmk);
  const esp_err_t pmkResult = esp_now_set_pmk(pmk);
  if (pmkResult != ESP_OK ||
      !addEspNowPeer(ESPNOW_BROADCAST_MAC, false) ||
      esp_now_register_recv_cb(onEspNowReceive) != ESP_OK) {
    Serial.printf("ESP-NOW setup incomplete pmkErr=%d\n", pmkResult);
    esp_now_deinit();
    return false;
  }
  espNowReady = true;
  Serial.printf("ESP-NOW ready role=MAIN staMac=%s apMac=%s channel=%d\n",
      WiFi.macAddress().c_str(), WiFi.softAPmacAddress().c_str(),
      WiFi.channel());
  return true;
}

void serviceEspNow(unsigned long now) {
  processEspNowPackets();
  if (!espNowReady && now - lastEspNowRetryAt >= 2000UL) {
    lastEspNowRetryAt = now;
    initializeEspNow();
  }
  if (!espNowReady) return;
  if (now - lastBeaconSentAt >= RAIN_BEACON_INTERVAL_MS) {
    lastBeaconSentAt = now;
    sendMainBeacon();
  }
  sendRemotePumpCommand(1, now);
  sendRemotePumpCommand(5, now);
  if (now - lastLoggedRainChannelChangeAt >= 2000UL) {
    lastLoggedRainChannelChangeAt = now;
    const int channel = WiFi.channel();
    if (channel != lastLoggedRainChannel) {
      lastLoggedRainChannel = channel;
      Serial.printf("MAIN ESP-NOW/AP+STA channel now=%d\n", channel);
    }
  }
}

void servicePumpRuns() {
  const unsigned long now = millis();
  for (int pumpNum = 1; pumpNum <= 8; pumpNum++) {
    PumpRunState& run = pumpRuns[pumpNum - 1];
    if (!run.active) continue;
    if (isRemoteWaterPump(pumpNum) && run.stopRequested) continue;
    if (run.hasDeadline && (long)(now - run.stopAt) >= 0) {
      stopPumpRun(pumpNum);
    }
  }
}

void finishRemoteCommand(const String& previousStatus) {
  const String requestId = pendingRemoteCommandId;
  const String actuator = pendingRemoteActuator;
  const String commandValue = pendingRemoteCommandValue;

  if (previousStatus.length() > 0) {
    bool commandIsRunning = false;
    bool commandAwaitingRemoteStart = false;
    for (int index = 0; index < 8; index++) {
      if (activeCommandIds[index] == requestId) {
        commandIsRunning = true;
        commandAwaitingRemoteStart = pumpRuns[index].awaitingStartConfirmation;
      }
    }
    for (const String& stopRequestId : pendingRemoteStopCommandIds) {
      if (stopRequestId == requestId) commandIsRunning = true;
    }
    if (commandIsRunning) {
      if (!commandAwaitingRemoteStart) publishCommandAck(requestId, "running");
    } else {
      const bool wasTerminal = previousStatus == "completed" ||
          previousStatus == "rejected" || previousStatus == "timed_out";
      publishCommandAck(requestId, wasTerminal ? previousStatus : "completed");
    }
    Serial.printf(
      "Command requestId=%s actuator=%s result=%s duplicate=1 leftover_id_removed=1\n",
      requestId.c_str(),
      actuator.c_str(),
      commandIsRunning ? "running" : "completed"
    );
  } else {
    publishCommandAck(requestId, "accepted");
    if (actuator == "growLight") {
      if (commandValue == "0" || commandValue == "1") {
        const bool requestedState = commandValue == "1";
        setGrowLight(requestedState);
        publishActuatorState(actuator, requestedState);
        publishCommandAck(requestId, "completed");
        Serial.printf(
          "Command requestId=%s actuator=%s result=completed\n",
          requestId.c_str(),
          actuator.c_str()
        );
      } else {
        publishCommandAck(requestId, "rejected");
        Serial.printf(
          "Command requestId=%s actuator=%s result=rejected reason=invalid_value\n",
          requestId.c_str(),
          actuator.c_str()
        );
      }
    } else {
      const int pumpNum = pumpNumberForActuator(actuator);
      String rejectionReason;
      const bool waterPump = isRemoteWaterPump(pumpNum);
      const bool validValue = commandValue == "1" ||
        (waterPump && commandValue == "0");
      const bool requestedOn = commandValue == "1";
      const bool accepted = validValue &&
        executePumpRequest(actuator, requestedOn, requestId);
      if (!accepted) {
      rejectionReason = !validValue
        ? "invalid_value"
        : pumpStartRejectionReason(pumpNum);
        if (rejectionReason.length() == 0) rejectionReason = "threshold_or_manual_safety";
        if (rejectionReason == "rain_node_offline" ||
            rejectionReason == "rain_node_no_confirm") {
          publishCommandRejectionReason(requestId, rejectionReason);
        }
        publishCommandAck(requestId, "rejected");
      }
      Serial.printf(
        "Command requestId=%s actuator=%s result=%s reason=%s\n",
        requestId.c_str(),
        actuator.c_str(),
        accepted ? "accepted" : "rejected",
        accepted ? "none" : rejectionReason.c_str()
      );
    }
  }

  removeQueuedCommandChild(pendingRemoteCommandPath);
  pendingRemoteActuator = "";
  pendingRemoteCommandId = "";
  pendingRemoteCommandValue = "";
  pendingRemoteCommandPath = "";
  remoteCommandPending = false;
}

bool processQueuedCommand(
    const String& actuator,
    const String& requestId,
    const String& commandValue,
    const String& commandPath) {
  if (!isValidCommandRequestId(requestId)) {
    removeQueuedCommandChild(commandPath);
    Serial.printf(
      "Command requestId=%s actuator=%s result=removed reason=invalid_id\n",
      requestId.c_str(),
      actuator.c_str()
    );
    return true;
  }

  pendingRemoteActuator = actuator;
  pendingRemoteCommandId = requestId;
  pendingRemoteCommandValue = commandValue;
  pendingRemoteCommandPath = commandPath;
  const String previousStatus = consumedCommandStatus(requestId);
  finishRemoteCommand(previousStatus);
  return previousStatus.length() > 0;
}

void processRemoteActuatorResult(AsyncResult& result) {
  remoteCommandPending = false;
  if (result.isError() || !result.available()) {
    return;
  }

  const String payload = result.to<RealtimeDatabaseResult>().to<String>();
  DynamicJsonDocument commandDocument(2048);
  const DeserializationError parseError = deserializeJson(commandDocument, payload);
  if (!parseError && commandDocument.is<JsonObject>()) {
    JsonObject allActuatorQueues = commandDocument.as<JsonObject>();
    for (JsonPair actuatorEntry : allActuatorQueues) {
      const String actuator = actuatorEntry.key().c_str();
      bool knownActuator = false;
      for (int index = 0; index < 9; index++) {
        if (actuator == remoteActuatorNames[index]) knownActuator = true;
      }
      if (!knownActuator) continue;

      const String queuePath = "/hydrodeck/commands/" + actuator;
      JsonVariant queueValue = actuatorEntry.value();
      if (queueValue.is<JsonObject>()) {
        JsonObject commandMap = queueValue.as<JsonObject>();
        for (JsonPair command : commandMap) {
          const String requestId = command.key().c_str();
          String commandValue;
          if (command.value().is<const char*>()) {
            commandValue = command.value().as<const char*>();
          }
          const String commandPath = queuePath + "/" + requestId;
          const bool continueDraining = processQueuedCommand(
            actuator,
            requestId,
            commandValue,
            commandPath
          );
          if (!continueDraining) break;
        }
      } else if (queueValue.is<const char*>()) {
        const String legacyValue = queueValue.as<const char*>();
        if (legacyValue.length() == 0 || legacyValue == "null") continue;
        String requestId;
        String commandValue;
        if (actuator == "growLight") {
          const int separator = legacyValue.lastIndexOf('|');
          if (separator > 0) {
            requestId = legacyValue.substring(0, separator);
            commandValue = legacyValue.substring(separator + 1);
          }
        } else {
          requestId = legacyValue;
          commandValue = "1";
        }
        processQueuedCommand(actuator, requestId, commandValue, queuePath);
      }
    }
  }
  result.clear();
}

  Database.get(aClient, "/hydrodeck/commands", processRemoteActuatorResult, false, "remoteCommands");

String currentManilaIntervalKey(const struct tm& localTime) {
  char key[16];
  const int slotMinute = localTime.tm_min < 30 ? 0 : 30;
  snprintf(
    key,
    sizeof(key),
    "%04d%02d%02d_%02d%02d",
    localTime.tm_year + 1900,
    localTime.tm_mon + 1,
    localTime.tm_mday,
    localTime.tm_hour,
    slotMinute
  );
  return String(key);
}

const char* snapshotQueueKey(int slot) {
  return SNAPSHOT_QUEUE_KEYS[slot];
}

String snapshotIdentityFromJson(const String& snapshotJson) {
  DynamicJsonDocument document(1536);
  if (deserializeJson(document, snapshotJson)) return "";
  const char* intervalKey = document["intervalKey"] | "";
  const int batchNumber = document["batchNumber"] | 0;
  if (intervalKey[0] == '\0' || batchNumber <= 0) return "";
  return String(intervalKey) + "|" + String(batchNumber);
}

bool snapshotIdentityWasRecorded(const String& identity) {
  if (identity.length() == 0) return false;
  if (identity == lastIntervalSnapshotIdentity) return true;
  const String history = commandPreferences.getString("snapshotIds", ";");
  if (history.indexOf(";" + identity + ";") >= 0) return true;
  for (int slot = 0; slot < SNAPSHOT_QUEUE_LIMIT; slot++) {
    if (snapshotIdentityFromJson(
            commandPreferences.getString(snapshotQueueKey(slot), "")) == identity) {
      return true;
    }
  }
  return false;
}

void rememberSnapshotIdentity(const String& identity) {
  String history = commandPreferences.getString("snapshotIds", ";");
  const String token = ";" + identity + ";";
  if (history.indexOf(token) >= 0) return;
  if (!history.startsWith(";")) history = ";" + history;
  if (!history.endsWith(";")) history += ";";
  history += identity + ";";

  int separatorCount = 0;
  for (int index = 0; index < history.length(); index++) {
    if (history[index] == ';') separatorCount++;
  }
  while (separatorCount > 9) {
    const int first = history.indexOf(';');
    const int second = history.indexOf(';', first + 1);
    if (second < 0) break;
    history.remove(first + 1, second - first);
    separatorCount--;
  }
  commandPreferences.putString("snapshotIds", history);
}

bool enqueueIntervalSnapshot(const String& identity, const String& snapshotJson) {
  if (snapshotIdentityWasRecorded(identity)) return true;
  for (int slot = 0; slot < SNAPSHOT_QUEUE_LIMIT; slot++) {
    const char* key = snapshotQueueKey(slot);
    if (commandPreferences.getString(key, "").length() > 0) continue;
    if (commandPreferences.putString(key, snapshotJson) == 0) return false;
    lastIntervalSnapshotIdentity = identity;
    commandPreferences.putString("snapLast", identity);
    rememberSnapshotIdentity(identity);
    return true;
  }
  Serial.println("Snapshot queue full; newest scheduled snapshot dropped");
  return false;
}

void processQueuedSnapshotUpload(AsyncResult& result) {
  if (result.isDebug() || result.isEvent()) {
    result.clear();
    return;
  }
  if (!result.isResult()) return;
  if (result.uid() != pendingSnapshotUploadUid) {
    result.clear();
    return;
  }

  const int slot = pendingSnapshotUploadSlot;
  const String identity = pendingSnapshotUploadIdentity;
  const String payload = pendingSnapshotUploadPayload;
  const char* queueKey = slot >= 0 ? snapshotQueueKey(slot) : "";
  if (result.isError()) {
    Serial.printf("Snapshot upload failed identity=%s code=%d; retry queued\n",
        identity.c_str(), result.error().code());
    snapshotUploadPending = false;
    lastSnapshotUploadAttemptAt = millis();
  } else if (slot >= 0 && commandPreferences.getString(queueKey, "") == payload) {
    const bool removed = commandPreferences.remove(queueKey);
    if (removed || commandPreferences.getString(queueKey, "").length() == 0) {
      Serial.printf("Snapshot upload confirmed identity=%s\n", identity.c_str());
      lastSnapshotUploadAttemptAt = 0;
    } else {
      Serial.printf("Snapshot queue cleanup failed identity=%s; retry later\n",
          identity.c_str());
      lastSnapshotUploadAttemptAt = millis();
    }
    snapshotUploadPending = false;
  } else {
    Serial.printf("Snapshot queue item changed before ACK identity=%s\n",
        identity.c_str());
    snapshotUploadPending = false;
    lastSnapshotUploadAttemptAt = millis();
  }
  result.clear();
}

void serviceQueuedSnapshotUpload(unsigned long now) {
  if (!firebaseInitialized || !app.ready() || firebaseHeartbeatPending) return;
  if (snapshotUploadPending) {
    if (now - snapshotUploadStartedAt < SNAPSHOT_UPLOAD_TIMEOUT_MS) return;
    Serial.printf("Snapshot upload timed out identity=%s; retry queued\n",
        pendingSnapshotUploadIdentity.c_str());
    snapshotUploadPending = false;
    lastSnapshotUploadAttemptAt = now;
  }
  if (lastSnapshotUploadAttemptAt != 0 &&
      now - lastSnapshotUploadAttemptAt < SNAPSHOT_UPLOAD_RETRY_MS) return;

  int selectedSlot = -1;
  long oldestCapture = 0;
  String selectedPayload;
  String selectedIdentity;
  String selectedPath;
  for (int slot = 0; slot < SNAPSHOT_QUEUE_LIMIT; slot++) {
    const String payload = commandPreferences.getString(snapshotQueueKey(slot), "");
    if (payload.length() == 0) continue;
    DynamicJsonDocument document(1536);
    if (deserializeJson(document, payload)) {
      Serial.printf("Discarding malformed queued snapshot slot=%d\n", slot);
      commandPreferences.remove(snapshotQueueKey(slot));
      continue;
    }
    const String identity = snapshotIdentityFromJson(payload);
    const char* intervalKey = document["intervalKey"] | "";
    const int batchNumber = document["batchNumber"] | 0;
    if (identity.length() == 0 || intervalKey[0] == '\0' || batchNumber <= 0) {
      commandPreferences.remove(snapshotQueueKey(slot));
      continue;
    }
    const long capturedAt = document["capturedAt"].as<long>();
    if (selectedSlot < 0 || capturedAt < oldestCapture) {
      selectedSlot = slot;
      oldestCapture = capturedAt;
      selectedPayload = payload;
      selectedIdentity = identity;
      selectedPath = "/hydrodeck/intervalSnapshots/" + String(intervalKey) +
          "/sessions/" + String(batchNumber);
    }
  }
  if (selectedSlot < 0) return;

  snapshotUploadSequence++;
  pendingSnapshotUploadUid = "snapshotUpload_" + String(mainBootId, HEX) + "_" +
      String(snapshotUploadSequence);
  pendingSnapshotUploadIdentity = selectedIdentity;
  pendingSnapshotUploadPayload = selectedPayload;
  pendingSnapshotUploadSlot = selectedSlot;
  snapshotUploadPending = true;
  snapshotUploadStartedAt = now;
  lastSnapshotUploadAttemptAt = now;
  Database.set<object_t>(
    aClient,
    selectedPath,
    object_t(selectedPayload.c_str()),
    processQueuedSnapshotUpload,
    pendingSnapshotUploadUid
  );
}

void requestIntervalSnapshot() {
  if (activeBatchNumber <= 0) return;
  struct tm localTime;
  if (!getLocalTime(&localTime, 0) || localTime.tm_year < (2024 - 1900)) return;
  if (localTime.tm_min != 0 && localTime.tm_min != 30) return;

  const String intervalKey = currentManilaIntervalKey(localTime);
  const String intervalIdentity = intervalKey + "|" + String(activeBatchNumber);
  if (intervalIdentity == lastSnapshotScheduleAttemptIdentity) return;
  if (snapshotIdentityWasRecorded(intervalIdentity)) {
    lastSnapshotScheduleAttemptIdentity = intervalIdentity;
    return;
  }

  const unsigned long nowMs = millis();
  if (!temperatureSensorValid1 || !temperatureSensorValid2 ||
      !phSensorValid1 || !phSensorValid2 || !tdsSensorValid1 || !tdsSensorValid2 ||
      nowMs - lastTemperatureValidAt1 > 5000UL ||
      nowMs - lastTemperatureValidAt2 > 5000UL ||
      nowMs - lastPhValidAt1 > 5000UL || nowMs - lastPhValidAt2 > 5000UL ||
      nowMs - lastTdsValidAt1 > 5000UL || nowMs - lastTdsValidAt2 > 5000UL) return;

  lastSnapshotScheduleAttemptIdentity = intervalIdentity;
  const time_t capturedAt = time(nullptr);
  String snapshotJson = "{";
  snapshotJson += "\"intervalKey\":\"" + intervalKey + "\",";
  snapshotJson += "\"batchNumber\":" + String(activeBatchNumber) + ",";
  snapshotJson += "\"capturedAt\":" + String((long)capturedAt) + ",";
  snapshotJson += "\"source\":\"esp32\",";
  snapshotJson += "\"bed1\":{\"temp\":" + String(currentTemperature1, 1) + ",";
  snapshotJson += "\"ph\":" + String(currentPhValue1, 2) + ",";
  snapshotJson += "\"tds\":" + String(currentTdsValue1, 0) + ",";
  snapshotJson += "\"water\":\"" + String(digitalRead(WATER_SENSOR_PIN1) == HIGH ? "full" : "low") + "\"},";
  snapshotJson += "\"bed2\":{\"temp\":" + String(currentTemperature2, 1) + ",";
  snapshotJson += "\"ph\":" + String(currentPhValue2, 2) + ",";
  snapshotJson += "\"tds\":" + String(currentTdsValue2, 0) + ",";
  snapshotJson += "\"water\":\"" + String(digitalRead(WATER_SENSOR_PIN2) == HIGH ? "full" : "low") + "\"},";
  snapshotJson += "\"reservoir\":{\"level\":\"" + reservoirLevel + "\",\"full\":" +
      String(reservoirFullWet ? "true" : "false") + ",\"online\":" +
      String(rainNodeOnline ? "true" : "false") + "}}";

  if (!enqueueIntervalSnapshot(intervalIdentity, snapshotJson)) {
    Serial.printf("Snapshot identity=%s was not queued\n", intervalIdentity.c_str());
  }
}

void handleCombinedStatus() {
  const unsigned long now = millis();
  const bool heartbeatKnown = lastFirebaseHeartbeatSuccessAt != 0;
  const unsigned long heartbeatAge = heartbeatKnown
      ? now - lastFirebaseHeartbeatSuccessAt : 0;
  const bool heartbeatFresh = heartbeatKnown && heartbeatAge <= 10000UL;
  String json = "{";
  json += "\"reservoir\":{\"level\":\"" + reservoirLevel + "\",\"full\":" +
      String(reservoirFullWet ? "true" : "false") + ",\"online\":" +
      String(rainNodeOnline ? "true" : "false") + "},";
  json += "\"reservoirFull\":" + String(reservoirFullWet ? 1 : 0) + ",";

void maintainProvisioningHotspot() {
  if (WiFi.getMode() == WIFI_AP_STA && WiFi.softAPIP() == apIp) return;

  const unsigned long now = millis();
  if (lastHotspotAttempt != 0 && now - lastHotspotAttempt < hotspotRetryInterval) return;
  lastHotspotAttempt = now;

  if (WiFi.getMode() != WIFI_AP_STA) {
    if (!WiFi.mode(WIFI_AP_STA)) {
      Serial.println("Failed to enable AP+STA mode; hotspot startup will retry.");
      return;
    }
    delay(100);
  }

  WiFi.setSleep(false);
  WiFi.softAPdisconnect(false);
  if (!WiFi.softAPConfig(apIp, apGateway, apSubnet)) {
    Serial.println("Failed to configure the ESP32 hotspot IP; startup will retry.");
    return;
  }

  const bool started = WiFi.softAP(AP_SSID, AP_PASSWORD, 1, false, 4);
  if (!started || WiFi.softAPIP() != apIp) {
    Serial.println("Failed to start ESP32-HYDRODECK hotspot; startup will retry.");
    return;
  }
  Serial.print("ESP32 hotspot is broadcasting: ");
  Serial.print(AP_SSID);
  Serial.print(" at ");
  Serial.println(WiFi.softAPIP());
}

// HTTP API Control Handlers
void handleControl() {
  // Handle traditional actuator control: /control?actuator=<name>&state=<0|1>
  if (server.hasArg("actuator") && server.hasArg("state")) {
    String actuator = server.arg("actuator");
    const String stateValue = server.arg("state");
    bool state = (stateValue == "1" || stateValue == "on");
    if (actuator == "growLight") {
      const String requestId = server.arg("requestId");
      if (requestId.length() > 0 && !isValidCommandRequestId(requestId)) {
        server.send(400, "text/plain", "Invalid request ID");
        return;
      }
      if (requestId.length() > 0 && wasCommandConsumed(requestId)) {
        server.send(200, "text/plain", "Duplicate request ignored");
        return;
      }
      if (requestId.length() > 0) rememberConsumedCommand(requestId, "accepted");
      setGrowLight(state);
      publishActuatorState(actuator, state);
      if (requestId.length() > 0) publishCommandAck(requestId, "completed");
      server.send(200, "text/plain", "OK");
      return;
    }
    const int pumpNum = pumpNumberForActuator(actuator);
    if (pumpNum > 0) {
      if (isRemoteWaterPump(pumpNum) && stateValue != "0" &&
          stateValue != "1" && stateValue != "on" && stateValue != "off") {
        server.send(400, "text/plain", "Invalid water-pump state");
        return;
      }
      const String requestId = server.arg("requestId");
      if (requestId.length() > 0 && !isValidCommandRequestId(requestId)) {
        server.send(400, "text/plain", "Invalid request ID");
        return;
      }
      if (requestId.length() > 0 &&
          wasCommandConsumed(requestId)) {
        server.send(200, "text/plain", "Duplicate request ignored");
        return;
      }
      if (requestId.length() > 0) publishCommandAck(requestId, "accepted");
      if (!state) {
        if (isRemoteWaterPump(pumpNum)) {
          stopPumpRun(pumpNum, requestId);
        } else {
          stopPumpRun(pumpNum);
          if (requestId.length() > 0) publishCommandAck(requestId, "completed");
        }
        server.send(200, "text/plain", "OK");
        return;
      }
      if (!executePumpRequest(actuator, true, requestId)) {
        String rejectionReason = pumpStartRejectionReason(pumpNum);
        if (rejectionReason == "rain_node_offline" ||
            rejectionReason == "rain_node_no_confirm") {
          publishCommandRejectionReason(requestId, rejectionReason);
        }
        publishCommandAck(requestId, "rejected");
        server.send(409, "text/plain", rejectionReason.length() > 0
            ? rejectionReason
            : "threshold_or_manual_safety");
        return;
      }
      server.send(200, "text/plain", "OK");
      return;
    }
  }

  // Handle command-based control: /control?command=<command>
  // Used for PH_ADJUST:<bed>:<direction> format from Flutter app
  if (server.hasArg("command")) {
    String command = server.arg("command");
    if (command.startsWith("PH_ADJUST:")) {
      parsePhAdjustCommand(command);
      server.send(200, "text/plain", "OK");
      return;
    }
    // Could add other command formats here if needed
  }

  server.send(400, "text/plain", "Bad Request");
}

void processFirebaseResult(AsyncResult& firebaseResult) {
  if (!firebaseResult.isResult()) return;
  if (firebaseResult.isError()) {
    Serial.print("Firebase error code: ");
    Serial.print(firebaseResult.error().code());
    Serial.println();
    firebaseInitialized = false;
    lastFirebaseTelemetryJson = "";
  }
  if (firebaseResult.isEvent()) {
    Serial.println("Firebase event received.");
  }
  if (firebaseResult.isDebug()) {
    Serial.println("Firebase debug event received.");
  }
}

void onMainWifiEvent(WiFiEvent_t event, WiFiEventInfo_t info) {
  if (wifiDiagnosticQueue == nullptr) return;
  WifiDiagnosticEvent diagnostic = {};
  if (event == ARDUINO_EVENT_WIFI_STA_DISCONNECTED) {
    diagnostic.kind = 1;
    diagnostic.disconnectReason = info.wifi_sta_disconnected.reason;
  } else if (event == ARDUINO_EVENT_WIFI_STA_CONNECTED) {
    diagnostic.kind = 2;
  } else if (event == ARDUINO_EVENT_WIFI_STA_GOT_IP) {
    diagnostic.kind = 3;
  } else {
    return;
  }
  xQueueSend(wifiDiagnosticQueue, &diagnostic, 0);
}

void processMainWifiDiagnostics() {
  if (wifiDiagnosticQueue == nullptr) return;
  WifiDiagnosticEvent diagnostic;
  while (xQueueReceive(wifiDiagnosticQueue, &diagnostic, 0) == pdTRUE) {
    if (diagnostic.kind == 1) {
      const wifi_err_reason_t reason =
          static_cast<wifi_err_reason_t>(diagnostic.disconnectReason);
      Serial.printf("MAIN Wi-Fi disconnected reason=%u (%s) channel=%d\n",
          diagnostic.disconnectReason,
          WiFi.disconnectReasonName(reason), WiFi.channel());
      firebaseInitialized = false;
        lastFirebaseTelemetryJson = "";
    } else if (diagnostic.kind == 2) {
      Serial.printf("MAIN Wi-Fi associated channel=%d\n", WiFi.channel());
    } else if (diagnostic.kind == 3) {
      Serial.printf("MAIN Wi-Fi got IP channel=%d RSSI=%d\n",
          WiFi.channel(), WiFi.RSSI());
    }
  }
}

void processFirebaseHeartbeatResult(AsyncResult& result) {
  firebaseHeartbeatPending = false;
  if (result.isError()) {
    Serial.printf("Firebase heartbeat failed code=%d\n", result.error().code());
  } else {
    lastFirebaseHeartbeatSuccessAt = millis();
    Serial.printf("Firebase heartbeat succeeded uptimeMs=%lu\n",
        lastFirebaseHeartbeatSuccessAt);
  }
  result.clear();
}

void configureSafeOutput(int pin, int safeState) {
  digitalWrite(pin, safeState);
  pinMode(pin, OUTPUT);
  digitalWrite(pin, safeState);
}

void initializeFirebase() {
  if (firebaseInitialized || WiFi.status() != WL_CONNECTED) return;
  const unsigned long now = millis();
  if (lastFirebaseInitAttemptAt != 0 &&
      now - lastFirebaseInitAttemptAt < firebaseInitRetryInterval) return;
  lastFirebaseInitAttemptAt = now;
  ssl_client.useBuiltinCACertBundle();
  ssl_client.setConnectionTimeout(1000);
  ssl_client.setHandshakeTimeout(5);
  if (!ntpConfigured) {
    configTzTime("PHT-8", "pool.ntp.org", "time.nist.gov");
    ntpConfigured = true;
  }
  initializeApp(aClient, app, getAuth(user_auth), processFirebaseResult, "authTask");
  app.getApp<RealtimeDatabase>(Database);
  Database.url(DATABASE_URL);
  firebaseInitialized = true;
  Serial.printf("Firebase initialization started channel=%d RSSI=%d\n",
      WiFi.channel(), WiFi.RSSI());
}

void serviceConnectionDiagnostics(unsigned long now) {
  const bool firebaseReady = firebaseInitialized && app.ready();
  if (firebaseReady != lastLoggedFirebaseReady) {
    lastLoggedFirebaseReady = firebaseReady;
    Serial.printf("Firebase readiness changed ready=%u wifi=%u\n",
        firebaseReady ? 1 : 0, WiFi.status() == WL_CONNECTED ? 1 : 0);
  }
  if (now - lastConnectionDiagnosticAt < connectionDiagnosticInterval) return;
  lastConnectionDiagnosticAt = now;
  const unsigned long heartbeatAge = lastFirebaseHeartbeatSuccessAt == 0
      ? 0 : now - lastFirebaseHeartbeatSuccessAt;
  Serial.printf(
      "MAIN diagnostics wifi=%s channel=%d RSSI=%d firebase=%s heartbeatAgeMs=%lu RAIN=%s RAINChannel=%d\n",
      WiFi.status() == WL_CONNECTED ? "connected" : "disconnected",
      WiFi.channel(),
      WiFi.status() == WL_CONNECTED ? WiFi.RSSI() : 0,
      firebaseReady ? "ready" : "not_ready",
      heartbeatAge,
      rainNodeOnline ? "online" : "offline",
      lastLoggedRainChannel);
}

void setup() {
  Serial.begin(115200);
  wifiDiagnosticQueue = xQueueCreate(8, sizeof(WifiDiagnosticEvent));
  if (wifiDiagnosticQueue == nullptr) {
    Serial.println("Wi-Fi diagnostic event queue allocation failed");
  } else {
    WiFi.onEvent(onMainWifiEvent);
  }

  // Initialize Actuators (Default OFF)
  configureSafeOutput(phUp1Pin, PUMP_OFF);
  configureSafeOutput(phDown1Pin, PUMP_OFF);
  configureSafeOutput(nutrientPump1Pin, PUMP_OFF);
  configureSafeOutput(phUp2Pin, PUMP_OFF);
  configureSafeOutput(phDown2Pin, PUMP_OFF);
  configureSafeOutput(nutrientPump2Pin, PUMP_OFF);
  configureSafeOutput(growLightPin, LIGHT_OFF);

  // Initialize Sensors
  pinMode(WATER_SENSOR_PIN1, INPUT_PULLUP);
  pinMode(WATER_SENSOR_PIN2, INPUT_PULLUP);
  pinMode(PH_PIN1, INPUT);
  pinMode(PH_PIN2, INPUT);
  pinMode(TDS_PIN1, INPUT);
  pinMode(TDS_PIN2, INPUT);

  analogReadResolution(12);
  analogSetPinAttenuation(PH_PIN1, ADC_11db);
  analogSetPinAttenuation(PH_PIN2, ADC_11db);
  analogSetPinAttenuation(TDS_PIN1, ADC_11db);
  analogSetPinAttenuation(TDS_PIN2, ADC_11db);

  // Temperature Sensors Setup
  sensors1.begin();
  sensors1.requestTemperatures();
  sensors2.begin();
  sensors2.requestTemperatures();
  delay(750);

  float tempReading1 = sensors1.getTempCByIndex(0);
  if (tempReading1 != DEVICE_DISCONNECTED_C && tempReading1 > -50.0) {
    currentTemperature1 = tempReading1;
    temperatureSensorValid1 = true;
    lastTemperatureValidAt1 = millis();
  }
  float tempReading2 = sensors2.getTempCByIndex(0);
  if (tempReading2 != DEVICE_DISCONNECTED_C && tempReading2 > -50.0) {
    currentTemperature2 = tempReading2;
    temperatureSensorValid2 = true;
    lastTemperatureValidAt2 = millis();
  }
  sensors1.setWaitForConversion(false);
  sensors2.setWaitForConversion(false);

  // Wi-Fi and hotspot setup
  wifiPreferences.begin("wifi", false);
  commandPreferences.begin("pumpcmd", false);
  setGrowLight(commandPreferences.getBool("growLightOn", false));
  activeBatchNumber = commandPreferences.getInt("activeBatch", 0);
  lastIntervalSnapshotIdentity = commandPreferences.getString(
    "snapLast",
    commandPreferences.getString("lastSnapshotIdentity", "")
  );
  String savedSsid = wifiPreferences.getString("ssid", ssid);
  String savedPassword = wifiPreferences.getString("password", password);
  savedSsid.toCharArray(ssid, sizeof(ssid));
  savedPassword.toCharArray(password, sizeof(password));
  maintainProvisioningHotspot();

  mainBootId = esp_random();
  if (mainBootId == 0) mainBootId = 1;
  remoteCommandCounter = esp_random();
  remotePumpCommandIds[0] = ++remoteCommandCounter;
  remotePumpCommandIds[1] = ++remoteCommandCounter;
  espNowReceiveQueue = xQueueCreate(8, sizeof(ReceivedPacket));
  if (espNowReceiveQueue == nullptr) {
    Serial.println("ESP-NOW queue allocation failed; RAIN node unavailable");
  }
  initializeEspNow();

  WiFi.begin(ssid, password);
  wifiConnectInProgress = true;
  wifiConnectStarted = millis();
  Serial.println("Connecting to home Wi-Fi in the background; hotspot remains available.");

  server.on("/", HTTP_GET, handleRootIdent);
  server.on("/provision", HTTP_GET, handleProvision);
  server.on("/settings", HTTP_GET, handleSettings);
  server.on("/status", HTTP_GET, handleCombinedStatus);
  server.on("/control", HTTP_GET, handleControl); // Control via URL e.g. /control?actuator=phUp1&state=1 OR /control?command=PH_ADJUST:1:UP
  server.begin();
}

void loop() {
  app.loop();
  server.handleClient();
  unsigned long now = millis();
  processMainWifiDiagnostics();
  serviceConnectionDiagnostics(now);
  serviceRainNodeHealth(now);
  serviceRemotePumpFeedback(now);

  maintainProvisioningHotspot();
  serviceEspNow(now);
  if (WiFi.status() == WL_CONNECTED && !mdnsStarted) {
    mdnsStarted = MDNS.begin("hydrodeck");
    if (mdnsStarted) Serial.println("mDNS: hydrodeck.local");
  }

  // Apply replacement Wi-Fi credentials without disabling the setup AP.
  if (provisioningPending) {
    if (!wifiConnectInProgress) {
      WiFi.disconnect(false, false);
      WiFi.begin(ssid, password);
      wifiConnectInProgress = true;
      wifiConnectStarted = now;
      Serial.println("Trying the newly provisioned Wi-Fi network...");
    } else if (WiFi.status() == WL_CONNECTED && WiFi.SSID() == String(ssid)) {
      provisioningPending = false;
      wifiConnectInProgress = false;
      initializeFirebase();
      Serial.println("Replacement Wi-Fi connected; setup hotspot remains available.");
    } else if (now - wifiConnectStarted >= wifiConnectTimeout) {
      wifiConnectInProgress = false;
      provisioningPending = false;
      lastWifiRetry = now;
      Serial.println("New Wi-Fi attempt timed out; setup hotspot remains available.");
    }
  } else if (WiFi.status() == WL_CONNECTED) {
    wifiConnectInProgress = false;
    initializeFirebase();
  } else {
    if (firebaseInitialized) {
      firebaseInitialized = false;
      lastFirebaseTelemetryJson = "";
      Serial.println("Wi-Fi lost; Firebase will re-authenticate after reconnect.");
    }
    if (wifiConnectInProgress && now - wifiConnectStarted >= wifiConnectTimeout) {
      wifiConnectInProgress = false;
    }
    if (!wifiConnectInProgress && now - lastWifiRetry >= wifiRetryInterval) {
      lastWifiRetry = now;
      WiFi.begin(ssid, password);
      wifiConnectInProgress = true;
      wifiConnectStarted = now;
      Serial.println("Retrying saved Wi-Fi credentials...");
    }
  }

  if (now - lastRemoteCommandPoll >= remoteCommandInterval) {
    lastRemoteCommandPoll = now;
    requestRemoteActuatorCommand();
  }

  if (!remoteCommandPending && pendingCommandCleanupWrites == 0) {
    if (now - lastPhLimitPoll >= phLimitPollInterval) {
      lastPhLimitPoll = now;
      requestPhLimit();
    }

    if (now - lastActiveBatchPoll >= activeBatchPollInterval) {
      lastActiveBatchPoll = now;
      requestActiveBatchNumber();
    }
    if (firebaseInitialized && app.ready() && !firebaseHeartbeatPending &&
      !snapshotUploadPending &&
        now - lastFirebaseHeartbeat >= firebaseHeartbeatInterval) {
      lastFirebaseHeartbeat = now;
      firebaseHeartbeatStartedAt = now;
      firebaseHeartbeatPending = true;
      Database.set<object_t>(
        aClient,
        "/hydrodeck/lastSeen",
        object_t("{\".sv\":\"timestamp\"}"),
        processFirebaseHeartbeatResult,
        "deviceHeartbeat"
      );
    }
  }

  requestIntervalSnapshot();
  serviceQueuedSnapshotUpload(now);

  if (firebaseHeartbeatPending &&
      now - firebaseHeartbeatStartedAt >= firebaseHeartbeatInterval * 3UL) {
    firebaseHeartbeatPending = false;
    Serial.println("Firebase heartbeat callback timed out; retrying");
  }

  // Sampling TDS Sensor 1 & Sensor 2
  static unsigned long sampleTime = 0;
  if (now - sampleTime > 40U) {
    sampleTime = now;
    analogBuffer1[analogBufferIndex1] = analogRead(TDS_PIN1);
    analogBufferIndex1 = (analogBufferIndex1 + 1) % SCOUNT;

    analogBuffer2[analogBufferIndex2] = analogRead(TDS_PIN2);
    analogBufferIndex2 = (analogBufferIndex2 + 1) % SCOUNT;
  }

  // Calculate TDS Values
  static unsigned long tdsTime = 0;
  if (now - tdsTime > 800U) {
    tdsTime = now;

    // Buffer 1
    for (int i = 0; i < SCOUNT; i++) analogBufferTemp1[i] = analogBuffer1[i];
    float avgV1 = getMedianNum(analogBufferTemp1, SCOUNT) * (float)VREF / 4096.0;
    avgV1 = constrain(avgV1, 0.0f, (float)VREF);
    float compCoeff1 = 1.0 + 0.02 * (currentTemperature1 - 25.0);
    float compV1 = avgV1 / compCoeff1;
    float tdsVal1 = (133.42 * compV1 * compV1 * compV1 - 255.86 * compV1 * compV1 + 857.39 * compV1) * 0.5;
    currentTdsValue1 = max(0.0f, tdsVal1) * tdsCalibrationFactor;
    tdsSensorValid1 = true;
    lastTdsValidAt1 = now;

    // Buffer 2
    for (int i = 0; i < SCOUNT; i++) analogBufferTemp2[i] = analogBuffer2[i];
    float avgV2 = getMedianNum(analogBufferTemp2, SCOUNT) * (float)VREF / 4096.0;
    avgV2 = constrain(avgV2, 0.0f, (float)VREF);
    float compCoeff2 = 1.0 + 0.02 * (currentTemperature2 - 25.0);
    float compV2 = avgV2 / compCoeff2;
    float tdsVal2 = (133.42 * compV2 * compV2 * compV2 - 255.86 * compV2 * compV2 + 857.39 * compV2) * 0.5;
    currentTdsValue2 = max(0.0f, tdsVal2) * tdsCalibrationFactor;
    tdsSensorValid2 = true;
    lastTdsValidAt2 = now;
  }

  // Reading pH Sensors with 1-minute delay logic
  if (now - lastPhCheck >= phInterval) {
    lastPhCheck = now;
    currentPhValue1 = readPhSensor(PH_PIN1, slope1);
    currentPhValue2 = readPhSensor(PH_PIN2, slope2);
    phSensorValid1 = true;
    phSensorValid2 = true;
    lastPhValidAt1 = now;
    lastPhValidAt2 = now;

    serviceAutomaticPhControl(1, currentPhValue1, now);
    serviceAutomaticPhControl(2, currentPhValue2, now);
  }

  servicePumpRuns();

  // Reading Temperature Sensors
  if (now - lastTempCheck >= tempInterval) {
    lastTempCheck = now;

    float reading1 = sensors1.getTempCByIndex(0);
    if (reading1 != DEVICE_DISCONNECTED_C && reading1 > -50.0) {
      currentTemperature1 = reading1;
      temperatureSensorValid1 = true;
      lastTemperatureValidAt1 = now;
    }
    sensors1.requestTemperatures();

    float reading2 = sensors2.getTempCByIndex(0);
    if (reading2 != DEVICE_DISCONNECTED_C && reading2 > -50.0) {
      currentTemperature2 = reading2;
      temperatureSensorValid2 = true;
      lastTemperatureValidAt2 = now;
    }
    sensors2.requestTemperatures();
  }

  // Syncing with Firebase Realtime Database
    if (firebaseInitialized && app.ready() && !remoteCommandPending &&
      pendingCommandCleanupWrites == 0 &&
      now - lastFirebasePush >= firebaseInterval) {
    lastFirebasePush = now;

    String liveTelemetryJson = "{";
    liveTelemetryJson += "\"tds1\":" + String(currentTdsValue1, 0) + ",";
    liveTelemetryJson += "\"ph1\":" + String(currentPhValue1, 2) + ",";
    liveTelemetryJson += "\"temp1\":" + String(currentTemperature1, 1) + ",";
    liveTelemetryJson += "\"water1\":\"" + String(digitalRead(WATER_SENSOR_PIN1) == HIGH ? "full" : "low") + "\",";
    liveTelemetryJson += "\"tds2\":" + String(currentTdsValue2, 0) + ",";
    liveTelemetryJson += "\"ph2\":" + String(currentPhValue2, 2) + ",";
    liveTelemetryJson += "\"temp2\":" + String(currentTemperature2, 1) + ",";
    liveTelemetryJson += "\"water2\":\"" + String(digitalRead(WATER_SENSOR_PIN2) == HIGH ? "full" : "low") + "\",";
    liveTelemetryJson += "\"actuators\":{";
    liveTelemetryJson += "\"waterPump1\":" + String(rainPump1On ? "true" : "false") + ",";
    liveTelemetryJson += "\"phUp1\":" + String(digitalRead(phUp1Pin) == PUMP_ON ? "true" : "false") + ",";
    liveTelemetryJson += "\"phDown1\":" + String(digitalRead(phDown1Pin) == PUMP_ON ? "true" : "false") + ",";
    liveTelemetryJson += "\"nutrient1\":" + String(digitalRead(nutrientPump1Pin) == PUMP_ON ? "true" : "false") + ",";
    liveTelemetryJson += "\"waterPump2\":" + String(rainPump2On ? "true" : "false") + ",";
    liveTelemetryJson += "\"phUp2\":" + String(digitalRead(phUp2Pin) == PUMP_ON ? "true" : "false") + ",";
    liveTelemetryJson += "\"phDown2\":" + String(digitalRead(phDown2Pin) == PUMP_ON ? "true" : "false") + ",";
    liveTelemetryJson += "\"nutrient2\":" + String(digitalRead(nutrientPump2Pin) == PUMP_ON ? "true" : "false") + ",";
    liveTelemetryJson += "\"growLight\":" + String(digitalRead(growLightPin) == LIGHT_ON ? "true" : "false") + "},";
    liveTelemetryJson += "\"reservoir\":{\"level\":\"" + reservoirLevel + "\",\"full\":" +
      String(reservoirFullWet ? "true" : "false") + ",\"online\":" +
      String(rainNodeOnline ? "true" : "false") + "},";
    liveTelemetryJson += "\"wifiConnected\":" + String(WiFi.status() == WL_CONNECTED ? "true" : "false") + ",";
    liveTelemetryJson += "\"firebaseReady\":" + String(firebaseInitialized && app.ready() ? "true" : "false") + ",";
    const bool firebaseHeartbeatFresh = lastFirebaseHeartbeatSuccessAt != 0 &&
      now - lastFirebaseHeartbeatSuccessAt <= 10000UL;
    liveTelemetryJson += "\"firebaseHeartbeatFresh\":" +
      String(firebaseHeartbeatFresh ? "true" : "false");
    liveTelemetryJson += "}";
    if (liveTelemetryJson != lastFirebaseTelemetryJson) {
      Database.update<object_t>(
        aClient,
        "/hydrodeck",
        object_t(liveTelemetryJson.c_str())
      );
      if (aClient.lastError().code() != 0) {
        Serial.print("Firebase live telemetry update failed, code: ");
        Serial.println(aClient.lastError().code());
      } else {
        lastFirebaseTelemetryJson = liveTelemetryJson;
        Serial.println("Firebase changed telemetry update sent.");
      }
    }
  }
}
