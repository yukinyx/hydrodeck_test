uint16_t packetCrc(const uint8_t* bytes, size_t length) {
  uint16_t crc = 0xffff;
  for (size_t index = 0; index < length; index++) {
    crc ^= bytes[index];
    for (uint8_t bit = 0; bit < 8; bit++) {
      crc = (crc & 1) ? (crc >> 1) ^ 0xa001 : crc >> 1;
    }
  }
  return crc;
}

template <typename Packet>
void preparePacket(Packet& packet, uint8_t type, uint8_t sender) {
  packet.header.magic = ESPNOW_MAGIC;
  packet.header.version = ESPNOW_VERSION;
  packet.header.type = type;
  packet.header.sender = sender;
  packet.header.payloadSize =
      sizeof(Packet) - sizeof(PacketHeader) - sizeof(packet.crc) -
      sizeof(packet.authTag);
  packet.header.sequence = ++packetSequence;
}

template <typename Packet>
bool finalizePacket(Packet& packet) {
  packet.crc = packetCrc(
      reinterpret_cast<const uint8_t*>(&packet), offsetof(Packet, crc));
  uint8_t digest[32];
  const mbedtls_md_info_t* md = mbedtls_md_info_from_type(MBEDTLS_MD_SHA256);
  if (md == nullptr || mbedtls_md_hmac(
      md,
      reinterpret_cast<const unsigned char*>(RAIN_TOKEN), strlen(RAIN_TOKEN),
      reinterpret_cast<const unsigned char*>(&packet), offsetof(Packet, authTag),
      digest) != 0) {
    return false;
  }
  memcpy(packet.authTag, digest, sizeof(packet.authTag));
  return true;
}

template <typename Packet>
bool packetValid(const Packet& packet, size_t length, uint8_t type, uint8_t sender) {
  if (length != sizeof(Packet) || packet.header.magic != ESPNOW_MAGIC ||
      packet.header.version != ESPNOW_VERSION || packet.header.type != type ||
      packet.header.sender != sender ||
      packet.header.payloadSize !=
          sizeof(Packet) - sizeof(PacketHeader) - sizeof(packet.crc) -
          sizeof(packet.authTag)) {
    return false;
  }
  if (packet.crc != packetCrc(
      reinterpret_cast<const uint8_t*>(&packet), offsetof(Packet, crc))) {
    return false;
  }
  uint8_t digest[32];
  const mbedtls_md_info_t* md = mbedtls_md_info_from_type(MBEDTLS_MD_SHA256);
  if (md == nullptr || mbedtls_md_hmac(
      md,
      reinterpret_cast<const unsigned char*>(RAIN_TOKEN), strlen(RAIN_TOKEN),
      reinterpret_cast<const unsigned char*>(&packet), offsetof(Packet, authTag),
      digest) != 0) {
    return false;
  }
  return memcmp(packet.authTag, digest, sizeof(packet.authTag)) == 0;
}

bool sequenceIsNewer(uint16_t sequence, uint16_t previous) {
  return static_cast<int16_t>(sequence - previous) > 0;
}

bool commandIdIsNewer(uint32_t commandId, uint32_t previous) {
  return static_cast<int32_t>(commandId - previous) > 0;
}

void derivePeerKey(const char* purpose, uint8_t output[16]) {
  String input = String(purpose) + ":" + RAIN_TOKEN;
  uint8_t digest[32];
  mbedtls_sha256(
      reinterpret_cast<const unsigned char*>(input.c_str()),
      input.length(),
      digest,
      0);
  memcpy(output, digest, 16);
}

bool addPeer(const uint8_t mac[6], bool encrypted) {
  esp_now_peer_info_t peer = {};
  memcpy(peer.peer_addr, mac, 6);
  peer.channel = 0;
  peer.ifidx = WIFI_IF_STA;
  peer.encrypt = encrypted;
  if (encrypted) derivePeerKey("HYDRODECK-RAIN-LMK-v1", peer.lmk);
  const esp_err_t result = esp_now_add_peer(&peer);
  if (result == ESP_OK || result == ESP_ERR_ESPNOW_EXIST) return true;
  Serial.printf("ESP-NOW peer add failed mac=%02X:%02X:%02X:%02X:%02X:%02X err=%d\n",
      mac[0], mac[1], mac[2], mac[3], mac[4], mac[5], result);
  return false;
}

bool sendPacket(const uint8_t mac[6], const void* packet, size_t length) {
  if (!espNowReady) return false;
  const esp_err_t result = esp_now_send(
      mac, reinterpret_cast<const uint8_t*>(packet), length);
  if (result != ESP_OK) {
    Serial.printf("ESP-NOW send enqueue failed err=%d channel=%d\n",
        result, WiFi.channel());
    return false;
  }
  return true;
}

void onEspNowReceive(
    const esp_now_recv_info_t* info,
    const uint8_t* data,
    int length) {
  if (info == nullptr || info->src_addr == nullptr || data == nullptr ||
      length <= 0 || length > static_cast<int>(ESPNOW_PACKET_MAX) ||
      receivedPackets == nullptr) return;
  ReceivedPacket packet = {};
  memcpy(packet.mac, info->src_addr, sizeof(packet.mac));
  packet.length = static_cast<uint8_t>(length);
  memcpy(packet.bytes, data, length);
  if (xQueueSend(receivedPackets, &packet, 0) != pdTRUE) {
    receiveQueueDrops.fetch_add(1, std::memory_order_relaxed);
  }
}

void updateDebouncedSensor(
    int pin,
    bool& stableState,
    bool& candidateState,
    unsigned long& candidateAt,
    unsigned long now) {
  const bool reading = digitalRead(pin) == HIGH;
  if (reading != candidateState) {
    candidateState = reading;
    candidateAt = now;
  } else if (reading != stableState &&
             now - candidateAt >= RESERVOIR_DEBOUNCE_MS) {
    stableState = reading;
  }
}

void setPumpRelays(bool pump1Requested, bool pump2Requested, unsigned long now) {
  const bool previousPump1On = pump1On;
  const bool previousPump2On = pump2On;
  if (pump1Requested && pump2Requested) {
    faultCode = "pump_conflict";
    pump1Requested = false;
    pump2Requested = false;
  }

  const bool reservoirLow = reservoirLevel() == "not_full";
  if (BLOCK_PUMP_WHEN_RESERVOIR_LOW && reservoirLow) {
    if ((pump1Requested && !pump1On) || (pump2Requested && !pump2On)) {
      faultCode = "reservoir_low";
    }
    pump1Requested = pump1On;
    pump2Requested = pump2On;
  }

  if (pump1Requested && pump1MaxRunLocked) pump1Requested = false;
  if (pump2Requested && pump2MaxRunLocked) pump2Requested = false;
  if (pump1Requested && pump2On) {
    digitalWrite(WATER_PUMP2_PIN, PUMP_OFF);
    pump2On = false;
    pump2Cloud = false;
    pump2ActiveCommandId = "";
  }
  if (pump2Requested && pump1On) {
    digitalWrite(WATER_PUMP1_PIN, PUMP_OFF);
    pump1On = false;
    pump1Cloud = false;
    pump1ActiveCommandId = "";
  }

  if (pump1Requested != pump1On) {
    digitalWrite(WATER_PUMP1_PIN, pump1Requested ? PUMP_ON : PUMP_OFF);
    pump1On = pump1Requested;
    pump1Cloud = pump1On && startingFromCloud;
    if (pump1On) {
      pump1StartedAt = now;
      if (LOCAL_MAX_RUN_MS > 0) pump1MaxRunAt = now + LOCAL_MAX_RUN_MS;
      pump1ActiveCommandId = pump1CommandId;
    } else {
      pump1ActiveCommandId = "";
      if (!desiredPump1On) pump1MaxRunLocked = false;
    }
  }
  if (pump2Requested != pump2On) {
    digitalWrite(WATER_PUMP2_PIN, pump2Requested ? PUMP_ON : PUMP_OFF);
    pump2On = pump2Requested;
    pump2Cloud = pump2On && startingFromCloud;
    if (pump2On) {
      pump2StartedAt = now;
      if (LOCAL_MAX_RUN_MS > 0) pump2MaxRunAt = now + LOCAL_MAX_RUN_MS;
      pump2ActiveCommandId = pump2CommandId;
    } else {
      pump2ActiveCommandId = "";
      if (!desiredPump2On) pump2MaxRunLocked = false;
    }
  }
  if (pump1On != previousPump1On || pump2On != previousPump2On) {
    statusUpdatePending = true;
  }
}

void stopPumpForMaxRun(int pumpNumber) {
  if (LOCAL_MAX_RUN_MS == 0) return;
  faultCode = "max_run";
  if (pumpNumber == 1) {
    digitalWrite(WATER_PUMP1_PIN, PUMP_OFF);
    pump1On = false;
    pump1MaxRunLocked = true;
    pump1ActiveCommandId = "";
  } else {
    digitalWrite(WATER_PUMP2_PIN, PUMP_OFF);
    pump2On = false;
    pump2MaxRunLocked = true;
    pump2ActiveCommandId = "";
  }
  statusUpdatePending = true;
}

void servicePumpSafety(unsigned long now) {
  // Losing the MAIN unit stops only pumps that MAIN is controlling. Pumps
  // started from Firebase keep running while the Firebase link is alive.
  const bool mainLost = !mainSessionKnown || now - lastMainSeenAt >= MAIN_BEACON_TIMEOUT_MS;
  if (mainLost && ((pump1On && !pump1Cloud) || (pump2On && !pump2Cloud))) {
    stopMainControlledPumpsForRecovery(now);
  }

  // A Firebase-started pump needs a live Firebase link (command polls).
  if (((pump1On && pump1Cloud) || (pump2On && pump2Cloud)) &&
      now - lastCloudPollOkAt >= CLOUD_LINK_TIMEOUT_MS) {
    stopCloudControlledPumps("cloud_link_lost");
  }

  if (LOCAL_MAX_RUN_MS > 0) {
    if (pump1On && (long)(now - pump1MaxRunAt) >= 0) stopPumpForMaxRun(1);
    if (pump2On && (long)(now - pump2MaxRunAt) >= 0) stopPumpForMaxRun(2);
  }
}

void sendCommandAck(const uint8_t mac[6], uint32_t commandId, uint8_t pump,
    uint8_t result, uint8_t reason) {
  CommandAckPacket packet = {};
  preparePacket(packet, MSG_COMMAND_ACK, DEVICE_RAIN);
  packet.rainBootId = rainBootId;
  packet.commandId = commandId;
  packet.pump = pump;
  packet.result = result;
  packet.reason = reason;
  if (!finalizePacket(packet)) {
    Serial.printf("ESP-NOW ACK authentication failed command=%lu pump=%u\n",
        static_cast<unsigned long>(commandId), pump);
    return;
  }
  if (!sendPacket(mac, &packet, sizeof(packet))) {
    Serial.printf("ESP-NOW ACK send failed command=%lu pump=%u\n",
        static_cast<unsigned long>(commandId), pump);
  }
}

void queueWifiProvisionAck(uint32_t requestId, uint8_t status) {
  provisionAckRequestId = requestId;
  provisionAckStatus = status;
  provisionAckAttempts = 0;
  provisionAckLastSentAt = 0;
  provisionAckPending = true;
}

void serviceWifiProvisionAck(unsigned long now) {
  if (!provisionAckPending || !mainPeerKnown || !mainSessionKnown ||
      provisionAckAttempts >= RAIN_PROVISION_ACK_MAX_ATTEMPTS ||
      (provisionAckLastSentAt != 0 &&
       now - provisionAckLastSentAt < RAIN_PROVISION_ACK_RETRY_MS)) return;
  WifiProvisionAckPacket packet = {};
  preparePacket(packet, MSG_WIFI_PROVISION_ACK, DEVICE_RAIN);
  packet.requestId = provisionAckRequestId;
  packet.rainBootId = rainBootId;
  packet.status = provisionAckStatus;
  if (!finalizePacket(packet)) {
    Serial.println("ESP-NOW Wi-Fi provisioning ACK authentication failed");
    return;
  }
  provisionAckLastSentAt = now;
  provisionAckAttempts++;
  if (!sendPacket(mainMac, &packet, sizeof(packet)) &&
      provisionAckAttempts >= RAIN_PROVISION_ACK_MAX_ATTEMPTS) {
    Serial.println("RAIN Wi-Fi provisioning ACK could not be delivered");
  }
  if (provisionAckAttempts >= RAIN_PROVISION_ACK_MAX_ATTEMPTS) {
    provisionAckPending = false;
  }
}

  if (packet.requestId == 0 || packet.ssidLength == 0 ||
      packet.ssidLength > sizeof(packet.ssid) ||
      packet.passwordLength > sizeof(packet.password) ||
      (packet.passwordLength > 0 && packet.passwordLength < 8) ||
      memchr(packet.ssid, '\0', packet.ssidLength) != nullptr ||
      memchr(packet.password, '\0', packet.passwordLength) != nullptr) {
    return false;
  }
  for (size_t index = packet.ssidLength; index < sizeof(packet.ssid); index++) {
    if (packet.ssid[index] != 0) return false;
  }
  for (size_t index = packet.passwordLength;
       index < sizeof(packet.password); index++) {
    if (packet.password[index] != 0) return false;
  }
  return true;
}

void processWifiProvisioning(const ReceivedPacket& received,
    const WifiProvisionPacket& packet) {
  if (!mainSessionKnown || !mainPeerKnown ||
      memcmp(received.mac, mainMac, sizeof(mainMac)) != 0 ||
      (mainCommandSequenceKnown &&
       !sequenceIsNewer(packet.header.sequence, lastMainCommandSequence))) {
    Serial.println("ESP-NOW Wi-Fi provisioning rejected: unpaired MAIN or stale sequence");
    return;
  }
  mainCommandSequenceKnown = true;
  lastMainCommandSequence = packet.header.sequence;

  uint8_t status = WIFI_PROVISION_REJECTED;
  if (validWifiProvisionFields(packet)) {
    char requestedSsid[33] = {};
    char requestedPassword[64] = {};
    memcpy(requestedSsid, packet.ssid, packet.ssidLength);
    memcpy(requestedPassword, packet.password, packet.passwordLength);
    const bool sameSavedCredentials =
        savedSsid == requestedSsid && savedPassword == requestedPassword;
    const bool duplicateRequest = lastProvisionRequestKnown &&
        packet.requestId == lastProvisionRequestId;
    const uint32_t persistedRequestId = wifiPreferences.getUInt("provId", 0);
    if (duplicateRequest) {
      status = sameSavedCredentials
          ? lastProvisionRequestStatus : WIFI_PROVISION_REJECTED;
    } else if (persistedRequestId == packet.requestId &&
        !sameSavedCredentials) {
      status = WIFI_PROVISION_REJECTED;
    } else if (sameSavedCredentials) {
      status = WIFI_PROVISION_ACCEPTED;
      if (wifiPreferences.putUInt("provId", packet.requestId) ==
          sizeof(uint32_t) &&
          wifiPreferences.getUInt("provId", 0) == packet.requestId) {
        lastProvisionRequestId = packet.requestId;
        lastProvisionRequestKnown = true;
        lastProvisionRequestStatus = status;
      } else {
        status = WIFI_PROVISION_REJECTED;
      }
    } else {
        const String previousSsid = savedSsid;
        const String previousPassword = savedPassword;
        const size_t ssidStored =
            wifiPreferences.putString("ssid", requestedSsid);
        const size_t passwordStored =
            wifiPreferences.putString("password", requestedPassword);
        const size_t requestStored =
            wifiPreferences.putUInt("provId", packet.requestId);
        if (ssidStored == strlen(requestedSsid) &&
            passwordStored == strlen(requestedPassword) &&
            requestStored == sizeof(uint32_t) &&
            wifiPreferences.getString("ssid", "") == requestedSsid &&
            wifiPreferences.getString("password", "") == requestedPassword &&
            wifiPreferences.getUInt("provId", 0) == packet.requestId) {
          savedSsid = requestedSsid;
          savedPassword = requestedPassword;
          wifiCredentialsPending = true;
          wifiCredentialsConnectAfter = millis() + 5000UL;
          lastMainSeenAt = millis();
          status = WIFI_PROVISION_ACCEPTED;
          lastProvisionRequestId = packet.requestId;
          lastProvisionRequestKnown = true;
          lastProvisionRequestStatus = status;
        } else {
          wifiPreferences.putString("ssid", previousSsid);
          wifiPreferences.putString("password", previousPassword);
          wifiPreferences.putUInt("provId", persistedRequestId);
          Serial.println("RAIN Wi-Fi credentials could not be stored; previous values restored.");
        }
    }
  } else {
    Serial.println("ESP-NOW Wi-Fi provisioning rejected: malformed credentials");
  }

  queueWifiProvisionAck(packet.requestId, status);
  Serial.printf("RAIN Wi-Fi provisioning result=%s request=%08lX\n",
      status == WIFI_PROVISION_ACCEPTED ? "stored" : "rejected",
      static_cast<unsigned long>(packet.requestId));
}

void sendRainStatus(bool broadcast) {
  RainStatusPacket packet = {};
  preparePacket(packet, broadcast ? MSG_RAIN_HELLO : MSG_RAIN_STATUS,
      DEVICE_RAIN);
  packet.bootId = rainBootId;
  packet.uptimeMs = millis();
  packet.activeCommandId = pump1On ? activeCommandId[0] :
      (pump2On ? activeCommandId[1] : 0);
  const int currentChannel = WiFi.channel();
  packet.currentChannel = static_cast<uint8_t>(currentChannel > 0 ? currentChannel : 0);
  packet.fullWet = reservoirFullWet ? 1 : 0;
  packet.pumpMask = (pump1On ? 1 : 0) | (pump2On ? 2 : 0);
  packet.activePump = pump1On ? PUMP1_ID : (pump2On ? PUMP2_ID : 0);
  packet.fault = currentFaultCode();
  if (!finalizePacket(packet)) {
    Serial.println("ESP-NOW status authentication failed");
    return;
  }
  const uint8_t* destination = broadcast ? ESPNOW_BROADCAST_MAC : mainMac;
  if (!sendPacket(destination, &packet, sizeof(packet))) {
    Serial.printf("ESP-NOW status send failed broadcast=%u channel=%d\n",
        broadcast ? 1 : 0, WiFi.channel());
  }
}

void stopAllPumpsForRecovery(unsigned long now) {
  digitalWrite(WATER_PUMP1_PIN, PUMP_OFF);
  digitalWrite(WATER_PUMP2_PIN, PUMP_OFF);
  pump1On = false;
  pump2On = false;
  desiredPump1On = false;
  desiredPump2On = false;
  pump1ActiveCommandId = "";
  pump2ActiveCommandId = "";
  pump1Cloud = false;
  pump2Cloud = false;
  statusUpdatePending = true;
  mainSessionKnown = false;
  requireFreshOffCommand = true;
  if (mainPeerKnown) esp_now_del_peer(mainMac);
  mainPeerKnown = false;
  mainBeaconSequenceKnown = false;
  mainCommandSequenceKnown = false;
  lastMainSeenAt = now;
  Serial.println("RAIN safety: MAIN heartbeat lost; pumps OFF, awaiting fresh OFF commands");
}

  mainBeaconSequenceKnown = false;
  mainCommandSequenceKnown = false;
  lastMainSeenAt = now;

}

void processPumpCommand(const ReceivedPacket& received,
    const PumpCommandPacket& command, unsigned long now) {
  const uint8_t pump = command.pump;
  if (pump != PUMP1_ID && pump != PUMP2_ID) {
    sendCommandAck(received.mac, command.commandId, pump, ACK_REJECTED,
        FAULT_INVALID_COMMAND);
    Serial.printf("ESP-NOW command rejected reason=invalid_pump pump=%u\n", pump);
    return;
  }
  const uint8_t index = pump == PUMP1_ID ? 0 : 1;
  const bool pumpOn = index == 0 ? pump1On : pump2On;
  const uint32_t currentId = lastCommandId[index];

  if (!mainSessionKnown) {
    Serial.printf("ESP-NOW command deferred until authenticated MAIN beacon pump=%u\n",
        pump);
    return;
  }
  if (command.mainBootId != mainBootId ||
      memcmp(received.mac, mainMac, sizeof(mainMac)) != 0) {
    sendCommandAck(received.mac, command.commandId, pump, ACK_REJECTED,
        FAULT_STALE_COMMAND);
    Serial.printf("ESP-NOW command rejected reason=stale_session pump=%u\n", pump);
    return;
  }
  if (command.requestedOn > 1 ||
      (LOCAL_MAX_RUN_MS > 0 && command.remainingMs > LOCAL_MAX_RUN_MS)) {
    sendCommandAck(received.mac, command.commandId, pump, ACK_REJECTED,
        FAULT_INVALID_COMMAND);
    Serial.printf("ESP-NOW command rejected reason=invalid_fields pump=%u\n", pump);
    return;
  }
  if (command.commandId == 0) {
    sendCommandAck(received.mac, command.commandId, pump, ACK_REJECTED,
        FAULT_INVALID_COMMAND);
    Serial.printf("ESP-NOW command rejected reason=zero_command_id pump=%u\n", pump);
    return;
  }

  lastMainSeenAt = now;
  if (command.commandId == currentId && currentId != 0) {
    const uint8_t result = pumpOn == (command.requestedOn != 0)
        ? ACK_DUPLICATE : ACK_REJECTED;
    sendCommandAck(received.mac, command.commandId, pump, result,
        result == ACK_DUPLICATE ? FAULT_NONE : FAULT_STALE_COMMAND);
    if (result == ACK_DUPLICATE) Serial.printf(
        "ESP-NOW duplicate command ACKed without restart command=%lu pump=%u\n",
        static_cast<unsigned long>(command.commandId), pump);
    return;
  }
  if (currentId != 0 && !commandIdIsNewer(command.commandId, currentId)) {
    sendCommandAck(received.mac, command.commandId, pump, ACK_REJECTED,
        FAULT_STALE_COMMAND);
    Serial.printf("ESP-NOW stale command rejected command=%lu pump=%u\n",
        static_cast<unsigned long>(command.commandId), pump);
    return;
  }

  if (command.requestedOn == 0) {
    requireFreshOffCommand = false;
    faultCode = "";
    lastCommandId[index] = command.commandId;
    if (index == 0) {
      desiredPump1On = false;
      pump1MaxRunLocked = false;
    } else {
      desiredPump2On = false;
      pump2MaxRunLocked = false;
    }
    setPumpRelays(desiredPump1On, desiredPump2On, now);
    activeCommandId[index] = 0;
    sendCommandAck(received.mac, command.commandId, pump, ACK_ACCEPTED,
        FAULT_NONE);
    Serial.printf("ESP-NOW stop accepted pump=%u command=%lu\n", pump,
        static_cast<unsigned long>(command.commandId));
    return;
  }

  if (requireFreshOffCommand) {
    sendCommandAck(received.mac, command.commandId, pump, ACK_REJECTED,
        FAULT_STALE_COMMAND);
    Serial.printf("ESP-NOW start rejected reason=recovery_requires_fresh_off pump=%u\n", pump);
    return;
  }

  if (pumpOn) {
    sendCommandAck(received.mac, command.commandId, pump, ACK_REJECTED,
        FAULT_PUMP_CONFLICT);
    Serial.printf("ESP-NOW new command rejected while pump already on pump=%u\n", pump);
    return;
  }

  if ((pump == PUMP1_ID ? pump2On : pump1On) ||
      (pump == PUMP1_ID ? pump1MaxRunLocked : pump2MaxRunLocked)) {
    const uint8_t reason = (pump1MaxRunLocked || pump2MaxRunLocked)
        ? FAULT_MAX_RUN : FAULT_PUMP_CONFLICT;
    sendCommandAck(received.mac, command.commandId, pump, ACK_REJECTED, reason);
    Serial.printf("ESP-NOW start rejected pump=%u reason=%u\n", pump, reason);
    return;
  }
  if (BLOCK_PUMP_WHEN_RESERVOIR_LOW && reservoirLevel() == "not_full") {
    faultCode = "reservoir_low";
    sendCommandAck(received.mac, command.commandId, pump, ACK_REJECTED,
        FAULT_RESERVOIR_LOW);
    Serial.printf("ESP-NOW start rejected pump=%u reason=reservoir_low\n", pump);
    return;
  }

  lastCommandId[index] = command.commandId;
  activeCommandId[index] = command.commandId;
  if (index == 0) {
    desiredPump1On = true;
  } else {
    desiredPump2On = true;
  }
  faultCode = "";
  setPumpRelays(desiredPump1On, desiredPump2On, now);
  if (index == 0) {
    pump1ActiveCommandId = command.commandId;
  } else {
    pump2ActiveCommandId = command.commandId;
  }
  if (!(index == 0 ? pump1On : pump2On)) {
    const uint8_t reason = faultCode == "reservoir_low"
        ? FAULT_RESERVOIR_LOW : FAULT_PUMP_CONFLICT;
    sendCommandAck(received.mac, command.commandId, pump, ACK_REJECTED, reason);
    return;
  }
  sendCommandAck(received.mac, command.commandId, pump, ACK_ACCEPTED, FAULT_NONE);
  Serial.printf("ESP-NOW pump accepted pump=%u command=%lu\n",
      pump, static_cast<unsigned long>(command.commandId));
}

void processMainBeacon(const ReceivedPacket& received,
    const MainBeaconPacket& beacon, unsigned long now) {
  if (beacon.bootId == 0 || beacon.currentChannel < FIRST_DISCOVERY_CHANNEL ||
      beacon.currentChannel > LAST_DISCOVERY_CHANNEL) {
    Serial.println("ESP-NOW beacon rejected: invalid boot ID or channel");
    return;
  }
  const bool newSession = mainBootId != 0 && mainBootId != beacon.bootId;
  if (newSession) {
    stopAllPumpsForRecovery(now);
    lastCommandId[0] = 0;
    lastCommandId[1] = 0;
    mainBeaconSequenceKnown = false;
    mainCommandSequenceKnown = false;
    Serial.printf("MAIN session changed boot=%08lX; stale pumps cleared\n",
        static_cast<unsigned long>(beacon.bootId));
  } else if (mainBeaconSequenceKnown &&
      !sequenceIsNewer(beacon.header.sequence, lastMainBeaconSequence)) {
    return;
  }

  const bool firstContact = !mainSessionKnown || !mainPeerKnown || newSession;
  const bool peerChanged = mainPeerKnown &&
      memcmp(mainMac, received.mac, sizeof(mainMac)) != 0;
  if (peerChanged) {
    Serial.println("ESP-NOW MAIN peer changed; removing stale peer entry");
    esp_now_del_peer(mainMac);
    mainPeerKnown = false;
  }
  memcpy(mainMac, received.mac, sizeof(mainMac));
  if (!mainPeerKnown && !addPeer(mainMac, true)) {
    mainSessionKnown = false;
    Serial.println("ESP-NOW MAIN peer setup failed; continuing channel discovery");
    return;
  }
  mainBootId = beacon.bootId;
  mainSessionKnown = true;
  mainPeerKnown = true;
  mainBeaconSequenceKnown = true;
  lastMainBeaconSequence = beacon.header.sequence;
  lastMainSeenAt = now;
  if (firstContact || peerChanged) {
    Serial.printf("ESP-NOW MAIN peer paired channel=%u; sending discovery hello\n",
        beacon.currentChannel);
    sendRainStatus(true);
    statusUpdatePending = false;
    lastStatusSentAt = now;
  }
}

void processEspNowPackets(unsigned long now) {
  if (receivedPackets == nullptr) return;
  const uint32_t queueDrops = receiveQueueDrops.load(std::memory_order_relaxed);
  if (queueDrops != reportedReceiveQueueDrops) {
    reportedReceiveQueueDrops = queueDrops;
    Serial.printf("ESP-NOW receive queue overflow drops=%lu\n",
        static_cast<unsigned long>(queueDrops));
  }
  ReceivedPacket received;
  while (xQueueReceive(receivedPackets, &received, 0) == pdTRUE) {
    if (received.length < sizeof(PacketHeader) + sizeof(uint16_t)) continue;
    PacketHeader header;
    memcpy(&header, received.bytes, sizeof(header));
    if (header.magic != ESPNOW_MAGIC || header.version != ESPNOW_VERSION ||
        header.sender != DEVICE_MAIN) {
      Serial.println("ESP-NOW packet rejected: invalid header/version/sender");
      continue;
    }

    if (header.type == MSG_MAIN_BEACON) {
      MainBeaconPacket beacon;
      memcpy(&beacon, received.bytes, sizeof(beacon));
      if (!packetValid(beacon, received.length, MSG_MAIN_BEACON, DEVICE_MAIN)) {
        Serial.println("ESP-NOW beacon rejected: size/CRC/authentication");
        continue;
      }
      processMainBeacon(received, beacon, now);
      continue;
    }
    if (header.type == MSG_PUMP_COMMAND) {
      PumpCommandPacket command;
      memcpy(&command, received.bytes, sizeof(command));
      if (!packetValid(command, received.length, MSG_PUMP_COMMAND, DEVICE_MAIN)) {
        Serial.println("ESP-NOW command rejected: size/CRC/authentication");
        continue;
      }
      if (mainCommandSequenceKnown &&
          !sequenceIsNewer(command.header.sequence, lastMainCommandSequence)) {
        Serial.printf("ESP-NOW stale packet sequence rejected seq=%u\n",
            command.header.sequence);
        continue;
      }
      mainCommandSequenceKnown = true;
      lastMainCommandSequence = command.header.sequence;
      processPumpCommand(received, command, now);
      continue;
    }
    if (header.type == MSG_WIFI_PROVISION) {
      WifiProvisionPacket provision;
      memcpy(&provision, received.bytes, sizeof(provision));
      if (!packetValid(provision, received.length, MSG_WIFI_PROVISION,
              DEVICE_MAIN)) {
        Serial.println("ESP-NOW Wi-Fi provisioning rejected: size/CRC/authentication");
        continue;
      }
      processWifiProvisioning(received, provision);
    }
  }
}

void serviceChannelDiscovery(unsigned long now) {
  if (WiFi.status() == WL_CONNECTED || wifiAttemptInProgress) return;
  if (lastDiscoveryChannelChangeAt != 0 &&
      now - lastDiscoveryChannelChangeAt < DISCOVERY_CHANNEL_DWELL_MS) {
    return;
  }
  const uint8_t channel = nextDiscoveryChannel;
  nextDiscoveryChannel = channel >= LAST_DISCOVERY_CHANNEL
      ? FIRST_DISCOVERY_CHANNEL : channel + 1;
  lastDiscoveryChannelChangeAt = now;
  discoveryAttemptCount++;
  const esp_err_t result = esp_wifi_set_channel(channel, WIFI_SECOND_CHAN_NONE);
  if (result != ESP_OK) {
    Serial.printf("RAIN discovery attempt=%lu channel=%u failed=%d active=%d\n",
        static_cast<unsigned long>(discoveryAttemptCount), channel, result,
        WiFi.channel());
    return;
  }
  Serial.printf("RAIN discovery attempt=%lu listening=%u active=%d\n",
      static_cast<unsigned long>(discoveryAttemptCount), channel, WiFi.channel());
}

void serviceDiscovery(unsigned long now) {
  if (mainSessionKnown && now - lastMainSeenAt >= MAIN_BEACON_TIMEOUT_MS) {
    stopAllPumpsForRecovery(now);
  }
  if (!mainSessionKnown) serviceChannelDiscovery(now);
  serviceWifiProvisionAck(now);

  if (statusUpdatePending) {
    statusUpdatePending = false;
    lastStatusSentAt = now;
    sendRainStatus(!mainPeerKnown);
    return;
  }
  if (now - lastStatusSentAt < STATUS_INTERVAL_MS) return;
  lastStatusSentAt = now;
  sendRainStatus(!mainPeerKnown);
}

bool initializeEspNow() {
  if (espNowReady) return true;
  if (receivedPackets == nullptr) {
    receivedPackets = xQueueCreate(8, sizeof(ReceivedPacket));
  }
  if (receivedPackets == nullptr) {
    Serial.println("ESP-NOW unavailable: receive queue is not allocated");
    return false;
  }
  const esp_err_t initResult = esp_now_init();
  if (initResult != ESP_OK && initResult != ESP_ERR_ESPNOW_EXIST) {
    Serial.printf("ESP-NOW init failed err=%d channel=%d\n",
        initResult, WiFi.channel());
    return false;
  }
  uint8_t pmk[16];
  derivePeerKey("HYDRODECK-RAIN-PMK-v1", pmk);
  const esp_err_t pmkResult = esp_now_set_pmk(pmk);
  if (pmkResult != ESP_OK) {
    Serial.printf("ESP-NOW PMK setup failed err=%d\n", pmkResult);
    esp_now_deinit();
    return false;
  }
  if (!addPeer(ESPNOW_BROADCAST_MAC, false) ||
      esp_now_register_recv_cb(onEspNowReceive) != ESP_OK) {
    Serial.println("ESP-NOW callback/broadcast peer setup failed; retrying");
    esp_now_deinit();
    return false;
  }
  espNowReady = true;
  Serial.printf("ESP-NOW ready role=RAIN mac=%s channel=%d encryptedPeer=LMK\n",
      WiFi.macAddress().c_str(), WiFi.channel());
  return true;
}

void processRainFirebaseAuth(AsyncResult& result) {
  if (result.isError()) {
    Serial.printf("RAIN Firebase authentication failed code=%d\n",
        result.error().code());
  }
  result.clear();
}

void processRainFirebaseStatus(AsyncResult& result) {
  if (result.uid() != "rainStatus") {
    result.clear();
    return;
  }
  firebaseStatusWritePending = false;
  if (result.isError()) {
    Serial.printf("RAIN Firebase status write failed code=%d\n",
        result.error().code());
    noteFirebaseUploadFailure();
  } else {
    noteFirebaseUploadSuccess();
  }
  result.clear();
}

void initializeRainFirebase(unsigned long now) {
  if (firebaseInitialized || WiFi.status() != WL_CONNECTED) return;
  sslClient.useBuiltinCACertBundle();
  sslClient.setConnectionTimeout(1000);
  sslClient.setHandshakeTimeout(5);
  if (!ntpConfigured) {
    configTzTime("PHT-8", "pool.ntp.org", "time.nist.gov");
    ntpConfigured = true;
  }
  initializeApp(firebaseClient, firebaseApp, getAuth(userAuth),
      processRainFirebaseAuth, "rainAuth");
  firebaseApp.getApp<RealtimeDatabase>(firebaseDatabase);
  firebaseDatabase.url(DATABASE_URL);
  firebaseInitialized = true;
  lastFirebaseStatusWriteAt = now - FIREBASE_STATUS_INTERVAL_MS;
  Serial.printf("RAIN Firebase initialization started channel=%d RSSI=%d\n",
      WiFi.channel(), WiFi.RSSI());
}

void serviceRainFirebase(unsigned long now) {
  if (WiFi.status() != WL_CONNECTED || WiFi.SSID() != savedSsid) {
    firebaseInitialized = false;
    firebaseStatusWritePending = false;
    return;
  }
  initializeRainFirebase(now);
  if (!firebaseInitialized) return;
  if (!firebaseApp.ready()) return;

  if (firebaseStatusWritePending) {
    if (now - firebaseStatusWriteStartedAt < FIREBASE_RESULT_TIMEOUT_MS) return;
    firebaseStatusWritePending = false;
    noteFirebaseUploadFailure();
    Serial.println("RAIN Firebase status write timed out; retrying.");
  }
  if (now - lastFirebaseStatusWriteAt < FIREBASE_STATUS_INTERVAL_MS) return;
  lastFirebaseStatusWriteAt = now;
  firebaseStatusWriteStartedAt = now;
  firebaseStatusWritePending = true;
  String statusJson;
  statusJson.reserve(256);
  statusJson += "{";
  statusJson += "\"online\":true,";
  statusJson += "\"reservoir\":{\"level\":\"" + reservoirLevel() +
      "\",\"full\":" + String(reservoirFullWet ? "true" : "false") + "},";
  statusJson += "\"waterPump1\":" + String(pump1On ? "true" : "false") + ",";
  statusJson += "\"waterPump2\":" + String(pump2On ? "true" : "false") + ",";
  statusJson += "\"updatedAt\":{\".sv\":\"timestamp\"}";
  statusJson += "}";
  firebaseDatabase.set<object_t>(
      firebaseClient,
      "/hydrodeck/rainNodeStatus",
      object_t(statusJson.c_str()),
      processRainFirebaseStatus,
      "rainStatus"
  );
  if (firebaseClient.lastError().code() != 0) {
    firebaseStatusWritePending = false;
    noteFirebaseUploadFailure();
    Serial.printf("RAIN Firebase status enqueue failed code=%d\n",
        firebaseClient.lastError().code());
  }
}

void serviceRainWifi(unsigned long now) {
  if (wifiCredentialsPending) {
    if ((long)(now - wifiCredentialsConnectAfter) < 0) return;
    wifiCredentialsPending = false;
    WiFi.disconnect(false, false);
    firebaseInitialized = false;
    wifiAttemptInProgress = false;
    lastWifiAttemptAt = now;
    wifiAttemptStartedAt = now;
    WiFi.begin(savedSsid.c_str(), savedPassword.c_str());
    wifiAttemptInProgress = true;
    Serial.println("RAIN attempting newly stored Wi-Fi credentials.");
  }

  if (WiFi.status() == WL_CONNECTED && WiFi.SSID() == savedSsid) {
    wifiAttemptInProgress = false;
    wifiRetryExponent = 0;
    wifiRetryDelayMs = WIFI_RETRY_BASE_MS;
    initializeRainFirebase(now);
    return;
  }

  if (wifiAttemptInProgress &&
      now - wifiAttemptStartedAt >= WIFI_CONNECT_TIMEOUT_MS) {
    wifiAttemptInProgress = false;
    lastWifiAttemptAt = now;
    if (wifiRetryExponent < 3) wifiRetryExponent++;
    const unsigned long backoff =
        WIFI_RETRY_BASE_MS * (1UL << wifiRetryExponent);
    wifiRetryDelayMs = backoff > WIFI_RETRY_MAX_MS
        ? WIFI_RETRY_MAX_MS : backoff;
    Serial.printf("RAIN Wi-Fi attempt timed out; retry backoff=%lu ms\n",
        wifiRetryDelayMs);
  }

  if (savedSsid.isEmpty() || wifiAttemptInProgress ||
      now - lastWifiAttemptAt < wifiRetryDelayMs) return;
  lastWifiAttemptAt = now;
  wifiAttemptStartedAt = now;
  WiFi.begin(savedSsid.c_str(), savedPassword.c_str());
  wifiAttemptInProgress = true;
  Serial.println("RAIN retrying saved Wi-Fi credentials.");
}

void setup() {
  Serial.begin(115200);

  // Boot is always OFF; never restore a previous pump command after restart.
  configureSafeOutput(WATER_PUMP1_PIN, PUMP_OFF);
  configureSafeOutput(WATER_PUMP2_PIN, PUMP_OFF);
  pinMode(RES_FULL_PIN, INPUT_PULLUP);
  reservoirFullWet = digitalRead(RES_FULL_PIN) == HIGH;
  reservoirFullCandidate = reservoirFullWet;
  reservoirFullCandidateAt = millis();

  rainBootId = esp_random();
  if (rainBootId == 0) rainBootId = 1;
  receivedPackets = xQueueCreate(8, sizeof(ReceivedPacket));
  if (receivedPackets == nullptr) {
    Serial.println("ESP-NOW receive queue allocation failed; pumps remain OFF");
  }

  WiFi.mode(WIFI_STA);
  WiFi.setSleep(false);
  WiFi.setAutoReconnect(false);
  wifiPreferences.begin("wifi", false);
  cloudCommandPreferences.begin("raincmd", false);
  if (wifiPreferences.getBool("lowMemReboot", false)) {
    Serial.println("RAIN previous restart was triggered by the low-memory guard.");
    wifiPreferences.putBool("lowMemReboot", false);
  }
  savedSsid =wifiPreferences.getString("ssid", "");
  savedPassword = wifiPreferences.getString("password", "");
  lastProvisionRequestId = wifiPreferences.getUInt("provId", 0);
  lastProvisionRequestKnown = lastProvisionRequestId != 0;
  lastProvisionRequestStatus = WIFI_PROVISION_ACCEPTED;
  Serial.printf("HYDRO_RAIN_ACTIVE stationMac=%s wifiCredentials=%s\n",
      WiFi.macAddress().c_str(), savedSsid.isEmpty() ? "not_configured" : "stored");
  lastEspNowRetryAt = millis();
  initializeEspNow();
  if (!savedSsid.isEmpty() && savedSsid.length() <= 32 &&
      savedPassword.length() <= 63 &&
      (savedPassword.isEmpty() || savedPassword.length() >= 8)) {
    lastWifiAttemptAt = millis();
    wifiAttemptStartedAt = lastWifiAttemptAt;
    WiFi.begin(savedSsid.c_str(), savedPassword.c_str());
    wifiAttemptInProgress = true;
    Serial.println("RAIN connecting to saved Wi-Fi credentials in the background.");
  }
}

// Modem sleep (the ESP32 default) can drop the connection and delay responses.
// Re-apply "no sleep" periodically because a Wi-Fi restart can restore the default.
void keepWifiAwake() {
  WiFi.setSleep(false);
  wifi_ps_type_t powerSave = WIFI_PS_MIN_MODEM;
  if (esp_wifi_get_ps(&powerSave) != ESP_OK || powerSave != WIFI_PS_NONE) {
    esp_wifi_set_ps(WIFI_PS_NONE);
  }
}

// ---------------------------------------------------------------------------
// Direct Firebase pump control. RAIN reads /hydrodeck/commands/waterPump1 and
// /waterPump2 itself, so the water pumps work even when the MAIN unit is out
// of ESP-NOW range. RAIN is the only writer of /hydrodeck/actuators/waterPump1|2.
// ---------------------------------------------------------------------------
enum : uint8_t { CLOUD_OP_SET_STRING = 1, CLOUD_OP_SET_BOOL = 2, CLOUD_OP_REMOVE = 3 };

struct CloudWriteOp {
  uint8_t kind;
  bool boolValue;
  char path[96];
  char value[16];
};

struct CloudRun {
  bool active;
  bool runningAcked;
  String id;
};

const uint8_t CLOUD_WRITE_QUEUE_SIZE = 12;
CloudWriteOp cloudWriteQueue[CLOUD_WRITE_QUEUE_SIZE];
uint8_t cloudWriteHead = 0;
uint8_t cloudWriteCount = 0;
CloudRun cloudRun[2];
int8_t lastPublishedPump[2] = {-1, -1};
unsigned long lastActuatorPublishAt[2] = {0, 0};
DynamicJsonDocument cloudCommandDoc(768);

bool pushCloudWrite(uint8_t kind, const String& path, const char* value, bool boolValue) {
  if (cloudWriteCount >= CLOUD_WRITE_QUEUE_SIZE ||
      path.length() >= sizeof(cloudWriteQueue[0].path)) {
    Serial.println("RAIN cloud write queue full; dropped one write.");
    return false;
  }
  CloudWriteOp& op = cloudWriteQueue[(cloudWriteHead + cloudWriteCount) % CLOUD_WRITE_QUEUE_SIZE];
  op.kind = kind;
  op.boolValue = boolValue;
  strncpy(op.path, path.c_str(), sizeof(op.path) - 1);
  op.path[sizeof(op.path) - 1] = '\0';
  strncpy(op.value, value != nullptr ? value : "", sizeof(op.value) - 1);
  op.value[sizeof(op.value) - 1] = '\0';
  cloudWriteCount++;
  return true;
}

// The rejection reason must be written before the ack: the app reads it when
// it sees "rejected".
void queueCloudAck(const String& id, const char* status, const char* reason) {
  if (reason != nullptr && reason[0] != '\0') {
    pushCloudWrite(CLOUD_OP_SET_STRING, "/hydrodeck/commandAckReasons/" + id, reason, false);
  }
  pushCloudWrite(CLOUD_OP_SET_STRING, "/hydrodeck/commandAcks/" + id, status, false);
}

void queueCloudRemove(const String& path) {
  pushCloudWrite(CLOUD_OP_REMOVE, path, "", false);
}

void queueCloudActuator(int index, bool on) {
  pushCloudWrite(CLOUD_OP_SET_BOOL,
      String("/hydrodeck/actuators/") + (index == 0 ? "waterPump1" : "waterPump2"),
      "", on);
}

void processCloudWriteResult(AsyncResult& result) {
  if (result.uid() != "cloudWrite") {
    result.clear();
    return;
  }
  cloudWritePending = false;
  if (result.isError()) {
    Serial.printf("RAIN cloud write failed code=%d\n", result.error().code());
    noteFirebaseUploadFailure();
  } else {
    noteFirebaseUploadSuccess();
  }
  result.clear();
}

void serviceCloudWrites(unsigned long now) {
  if (cloudWritePending) {
    if (now - cloudWriteStartedAt < CLOUD_REQUEST_TIMEOUT_MS) return;
    cloudWritePending = false;
    noteFirebaseUploadFailure();
    Serial.println("RAIN cloud write timed out.");
  }
  if (cloudWriteCount == 0 || !firebaseInitialized || !firebaseApp.ready()) return;

  const CloudWriteOp op = cloudWriteQueue[cloudWriteHead];
  cloudWriteHead = (cloudWriteHead + 1) % CLOUD_WRITE_QUEUE_SIZE;
  cloudWriteCount--;
  cloudWritePending = true;
  cloudWriteStartedAt = now;
  if (op.kind == CLOUD_OP_SET_STRING) {
    firebaseDatabase.set<String>(firebaseClient, String(op.path), String(op.value),
        processCloudWriteResult, "cloudWrite");
  } else if (op.kind == CLOUD_OP_SET_BOOL) {
    firebaseDatabase.set<bool>(firebaseClient, String(op.path), op.boolValue,
        processCloudWriteResult, "cloudWrite");
  } else {
    firebaseDatabase.remove(firebaseClient, String(op.path),
        processCloudWriteResult, "cloudWrite");
  }
  if (firebaseClient.lastError().code() != 0) {
    cloudWritePending = false;
    noteFirebaseUploadFailure();
    Serial.printf("RAIN cloud write enqueue failed code=%d\n",
        firebaseClient.lastError().code());
  }
}

// ---- command bookkeeping (survives restarts so a command never runs twice) ----
String cloudConsumedStatus(const String& requestId) {
  const String entries = cloudCommandPreferences.getString("recentIds", "");
  int start = 0;
  while (start < static_cast<int>(entries.length())) {
    const int separator = entries.indexOf(',', start);
    const int end = separator < 0 ? entries.length() : separator;
    const String entry = entries.substring(start, end);
    const int statusSeparator = entry.indexOf('=');
    const String storedId = statusSeparator < 0 ? entry : entry.substring(0, statusSeparator);
    if (storedId == requestId) {
      return statusSeparator < 0 ? "completed" : entry.substring(statusSeparator + 1);
    }
    if (separator < 0) break;
    start = separator + 1;
  }
  return "";
}

void cloudRememberConsumed(const String& requestId, const char* status) {
  String entries = cloudCommandPreferences.getString("recentIds", "");
  String retained;
  int start = 0;
  while (start < static_cast<int>(entries.length())) {
    const int separator = entries.indexOf(',', start);
    const int end = separator < 0 ? entries.length() : separator;
    const String entry = entries.substring(start, end);
    const int statusSeparator = entry.indexOf('=');
    const String storedId = statusSeparator < 0 ? entry : entry.substring(0, statusSeparator);
    if (storedId != requestId && entry.length() > 0) {
      if (retained.length() > 0) retained += ",";
      retained += entry;
    }
    if (separator < 0) break;
    start = separator + 1;
  }
  if (retained.length() > 0) retained += ",";
  retained += requestId + "=" + status;
  while (retained.length() > 480) {
    const int separator = retained.indexOf(',');
    if (separator < 0) break;
    retained.remove(0, separator + 1);
  }
  cloudCommandPreferences.putString("recentIds", retained);
}

// App IDs are Firebase push IDs ("-..." 20 chars) or the cmd_* fallback.
bool validCloudRequestId(const String& id) {
  if (id.length() < 4 || id.length() > 48) return false;
  if (!(id.startsWith("cmd_") || (id.startsWith("-") && id.length() >= 16))) return false;
  for (size_t i = 0; i < id.length(); i++) {
    const char c = id[i];
    const bool ok = (c >= '0' && c <= '9') || (c >= 'a' && c <= 'z') ||
        (c >= 'A' && c <= 'Z') || c == '_' || c == '-';
    if (!ok) return false;
  }
  return true;
}

// Firebase push IDs start with a millisecond timestamp, so an old queued
// start command (for example from before a Wi-Fi outage) can be ignored.
bool cloudCommandTooOld(const String& id) {
  if (id.length() != 20 || id[0] != '-') return false;
  static const char kChars[] = "-0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZ_abcdefghijklmnopqrstuvwxyz";
  uint64_t createdMs = 0;
  for (int i = 0; i < 8; i++) {
    const char* position = strchr(kChars, id[i]);
    if (position == nullptr || id[i] == '\0') return false;
    createdMs = createdMs * 64 + static_cast<uint64_t>(position - kChars);
  }
  const time_t nowS = time(nullptr);
  if (nowS < 1600000000) return false;  // clock not synced yet: cannot tell
  const long ageS = static_cast<long>(nowS) - static_cast<long>(createdMs / 1000ULL);
  return ageS > CLOUD_COMMAND_MAX_AGE_S;
}

bool cloudStartPump(int index, const String& id, const char*& reason) {
  const bool pumpOn = index == 0 ? pump1On : pump2On;
  const bool otherOn = index == 0 ? pump2On : pump1On;
  const bool locked = index == 0 ? pump1MaxRunLocked : pump2MaxRunLocked;
  if (pumpOn) { reason = "pump_already_active"; return false; }
  if (otherOn) { reason = "other_bed_water_pump_active"; return false; }
  if (locked) { reason = "max_run_locked"; return false; }

  faultCode = "";
  if (index == 0) desiredPump1On = true; else desiredPump2On = true;
  startingFromCloud = true;
  setPumpRelays(desiredPump1On, desiredPump2On, millis());
  startingFromCloud = false;
  if (!(index == 0 ? pump1On : pump2On)) {
    if (index == 0) desiredPump1On = false; else desiredPump2On = false;
    reason = faultCode == "reservoir_low" ? "reservoir_low" : "start_blocked";
    return false;
  }
  if (index == 0) pump1ActiveCommandId = id; else pump2ActiveCommandId = id;
  return true;
}

void cloudStopPump(int index) {
  if (index == 0) desiredPump1On = false; else desiredPump2On = false;
  setPumpRelays(desiredPump1On, desiredPump2On, millis());
}

void rejectCloudCommand(const String& id, const String& commandPath, const char* reason) {
  cloudRememberConsumed(id, "rejected");
  queueCloudAck(id, "rejected", reason);
  queueCloudRemove(commandPath);
  Serial.printf("RAIN cloud command rejected id=%s reason=%s\n", id.c_str(), reason);
}

void processCloudCommand(int index, const String& id, const String& value) {
  const char* actuator = index == 0 ? "waterPump1" : "waterPump2";
  const String commandPath = String("/hydrodeck/commands/") + actuator + "/" + id;
  if (!validCloudRequestId(id)) {
    queueCloudRemove(commandPath);
    Serial.printf("RAIN cloud command removed reason=invalid_id actuator=%s\n", actuator);
    return;
  }

  const String previous = cloudConsumedStatus(id);
  if (previous.length() > 0) {
    // Already handled (leftover after a failed cleanup or a restart): never run it again.
    if (!(cloudRun[index].active && cloudRun[index].id == id)) {
      const bool terminal = previous == "completed" || previous == "rejected" ||
          previous == "timed_out";
      queueCloudAck(id, terminal ? previous.c_str() : "completed", "");
    }
    queueCloudRemove(commandPath);
    Serial.printf("RAIN cloud command duplicate ignored id=%s\n", id.c_str());
    return;
  }

  const bool isStart = value == "1";
  const bool isStop = value == "0";
  if (!isStart && !isStop) {
    rejectCloudCommand(id, commandPath, "invalid_value");
    return;
  }
  if (isStart && cloudCommandTooOld(id)) {
    rejectCloudCommand(id, commandPath, "stale_command");
    return;
  }

  cloudRememberConsumed(id, "accepted");
  queueCloudAck(id, "accepted", "");

  if (isStop) {
    cloudStopPump(index);
    cloudRememberConsumed(id, "completed");
    queueCloudAck(id, "completed", "");
    queueCloudRemove(commandPath);
    Serial.printf("RAIN cloud stop id=%s actuator=%s\n", id.c_str(), actuator);
    return;
  }

  const char* reason = "";
  if (cloudStartPump(index, id, reason)) {
    cloudRun[index].active = true;
    cloudRun[index].runningAcked = false;
    cloudRun[index].id = id;
    cloudRememberConsumed(id, "running");
    queueCloudRemove(commandPath);
    Serial.printf("RAIN cloud start id=%s actuator=%s\n", id.c_str(), actuator);
  } else {
    rejectCloudCommand(id, commandPath, reason);
  }
}

void processCloudCommandsResult(AsyncResult& result) {
  if (result.uid() != "cloudCommands") {
    result.clear();
    return;
  }
  cloudPollPending = false;
  if (result.isError()) {
    Serial.printf("RAIN cloud command poll failed code=%d\n", result.error().code());
    noteFirebaseUploadFailure();
    result.clear();
    return;
  }
  // A poll that reached Firebase keeps the link alive even when the queue is empty.
  lastCloudPollOkAt = millis();
  noteFirebaseUploadSuccess();
  if (!result.available()) {
    result.clear();
    return;
  }

  const String payload = result.to<RealtimeDatabaseResult>().to<String>();
  result.clear();
  if (payload.length() < 3) return;  // "null" or empty: nothing queued

  DynamicJsonDocument filter(64);
  filter["waterPump1"] = true;
  filter["waterPump2"] = true;
  cloudCommandDoc.clear();
  const DeserializationError parseError = deserializeJson(
      cloudCommandDoc, payload, DeserializationOption::Filter(filter));
  if (parseError || !cloudCommandDoc.is<JsonObject>()) return;

  uint8_t handled = 0;  // a couple per poll keeps the write queue short
  for (int index = 0; index < 2 && handled < 2; index++) {
    JsonObject queue = cloudCommandDoc[index == 0 ? "waterPump1" : "waterPump2"].as<JsonObject>();
    if (queue.isNull()) continue;
    for (JsonPair entry : queue) {
      if (handled >= 2) break;
      const String id = entry.key().c_str();
      const String value = entry.value().is<const char*>()
          ? String(entry.value().as<const char*>()) : String("");
      processCloudCommand(index, id, value);
      handled++;
    }
  }
}

void serviceCloudCommands(unsigned long now) {
  if (cloudPollPending) {
    if (now - cloudPollStartedAt < CLOUD_REQUEST_TIMEOUT_MS) return;
    cloudPollPending = false;
    noteFirebaseUploadFailure();
    Serial.println("RAIN cloud command poll timed out.");
  }
  if (!firebaseInitialized || !firebaseApp.ready()) return;
  // Let queued acks/cleanups finish first so a command is never seen twice.
  if (cloudWritePending || cloudWriteCount > 0) return;
  if (lastCloudPollAt != 0 && now - lastCloudPollAt < REMOTE_COMMAND_INTERVAL_MS) return;
  lastCloudPollAt = now;
  cloudPollStartedAt = now;
  cloudPollPending = true;
  firebaseDatabase.get(firebaseClient, "/hydrodeck/commands",
      processCloudCommandsResult, false, "cloudCommands");
  if (firebaseClient.lastError().code() != 0) {
    cloudPollPending = false;
    noteFirebaseUploadFailure();
    Serial.printf("RAIN cloud command poll enqueue failed code=%d\n",
        firebaseClient.lastError().code());
  }
}

// Publishes the pump's real state (RAIN owns the relays) and the running /
// completed acks for Firebase-started runs.
void serviceCloudPumpFeedback(unsigned long now) {
  const bool firebaseReady = firebaseInitialized && firebaseApp.ready();
  for (int index = 0; index < 2; index++) {
    const bool on = index == 0 ? pump1On : pump2On;
    const bool cloud = index == 0 ? pump1Cloud : pump2Cloud;
    CloudRun& run = cloudRun[index];
    if (run.active) {
      if (on && cloud) {
        if (!run.runningAcked) {
          queueCloudAck(run.id, "running", "");
          run.runningAcked = true;
        }
      } else {
        // Ended: stop command, MAIN OFF, safety stop, or reservoir/relay fault.
        queueCloudAck(run.id, "completed", "");
        cloudRememberConsumed(run.id, "completed");
        run.active = false;
        run.id = "";
      }
    }
    if (firebaseReady &&
        (lastPublishedPump[index] != (on ? 1 : 0) ||
         now - lastActuatorPublishAt[index] >= ACTUATOR_RESYNC_MS)) {
      queueCloudActuator(index, on);
      lastPublishedPump[index] = on ? 1 : 0;
      lastActuatorPublishAt[index] = now;
    }
  }
}

// Switch both pumps off, flag the next boot, and restart. The pumps stay off
// after boot until the main unit sends a fresh command.
void restartForMemory(const char* reason) {
  Serial.printf("RAIN MEMORY GUARD: %s; switching pumps off and restarting.\n", reason);
  digitalWrite(WATER_PUMP1_PIN, PUMP_OFF);
  digitalWrite(WATER_PUMP2_PIN, PUMP_OFF);
  wifiPreferences.putBool("lowMemReboot", true);
  Serial.flush();
  delay(100);
  ESP.restart();
}

// Restart cleanly when memory is nearly exhausted instead of letting Wi-Fi,
// TLS and ESP-NOW fail one after another.
void serviceMemoryGuard(unsigned long now) {
  static unsigned long lastCheckAt = 0;
  static unsigned long lastLowCheckAt = 0;
  static unsigned long lastHeapLogAt = 0;
  static uint8_t lowChecks = 0;
  if (now - lastCheckAt < 1000UL) return;
  lastCheckAt = now;

  const uint32_t freeHeap = ESP.getFreeHeap();
  const uint32_t largestBlock = ESP.getMaxAllocHeap();
  if (now - lastHeapLogAt >= HEAP_LOG_INTERVAL_MS) {
    lastHeapLogAt = now;
    Serial.printf("RAIN heap free=%u minFree=%u largestBlock=%u\n",
        static_cast<unsigned>(freeHeap),
        static_cast<unsigned>(ESP.getMinFreeHeap()),
        static_cast<unsigned>(largestBlock));
  }

  // Fast path: uploads failing or stuck for 10 s while memory is nearly full.
  if (now >= UPLOAD_FAIL_BOOT_GRACE_MS && WiFi.status() == WL_CONNECTED) {
    const bool uploadsFailing = firebaseUploadFailingSince != 0 &&
        now - firebaseUploadFailingSince >= UPLOAD_FAIL_RESTART_MS;
    const bool uploadsStuck =
        (firebaseStatusWritePending &&
            now - firebaseStatusWriteStartedAt >= UPLOAD_FAIL_RESTART_MS) ||
        (cloudPollPending && now - cloudPollStartedAt >= UPLOAD_FAIL_RESTART_MS) ||
        (cloudWritePending && now - cloudWriteStartedAt >= UPLOAD_FAIL_RESTART_MS);
    const bool memoryNearFull =
        freeHeap < NEAR_FULL_FREE_BYTES || largestBlock < NEAR_FULL_LARGEST_BLOCK_BYTES;
    if ((uploadsFailing || uploadsStuck) && memoryNearFull) {
      Serial.printf("RAIN uploads failing/stuck >%lus with heapFree=%u heapLargestBlock=%u\n",
          UPLOAD_FAIL_RESTART_MS / 1000UL,
          static_cast<unsigned>(freeHeap), static_cast<unsigned>(largestBlock));
      restartForMemory("Firebase uploads failing while memory is nearly full");
    }
  }

  // Slow path: memory critically low for several checks in a row.
  if (now - lastLowCheckAt < LOW_HEAP_CHECK_INTERVAL_MS) return;
  lastLowCheckAt = now;
  if (now < LOW_HEAP_BOOT_GRACE_MS) return;
  if (freeHeap >= LOW_HEAP_FREE_BYTES && largestBlock >= LOW_HEAP_LARGEST_BLOCK_BYTES) {
    lowChecks = 0;
    return;
  }
  lowChecks++;
  Serial.printf("RAIN LOW MEMORY check %u/%u heapFree=%u heapLargestBlock=%u\n",
      lowChecks, LOW_HEAP_CONSECUTIVE_CHECKS,
      static_cast<unsigned>(freeHeap), static_cast<unsigned>(largestBlock));
  if (lowChecks < LOW_HEAP_CONSECUTIVE_CHECKS) return;
  restartForMemory("memory critically low");
}

void loop() {
  const unsigned long now = millis();
  serviceMemoryGuard(now);
  static unsigned long lastWifiPowerCheck = 0;
  if (now - lastWifiPowerCheck >= 30000UL) {
    lastWifiPowerCheck = now;
    keepWifiAwake();
  }
  const bool previousReservoirFull = reservoirFullWet;
  updateDebouncedSensor(
    RES_FULL_PIN,
    reservoirFullWet,
    reservoirFullCandidate,
    reservoirFullCandidateAt,
    now
  );
  if (reservoirFullWet != previousReservoirFull) statusUpdatePending = true;
  servicePumpSafety(now);
  processEspNowPackets(now);
  firebaseApp.loop();
  serviceRainWifi(now);
  serviceRainFirebase(now);
  serviceCloudCommands(now);
  serviceCloudPumpFeedback(now);
  serviceCloudWrites(now);
  if (!espNowReady && now - lastEspNowRetryAt >= ESPNOW_INIT_RETRY_MS) {
    lastEspNowRetryAt = now;
    Serial.println("RAIN retrying ESP-NOW initialization");
    initializeEspNow();
  }
  serviceDiscovery(now);
}
